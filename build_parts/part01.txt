def log(msg: str) -> None:
    print(f"[stremio-tv] {msg}", flush=True)


def load_json(path: Path) -> Any:
    with path.open("r", encoding="utf-8") as f:
        return json.load(f)


def write_json(path: Path, obj: Any) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("w", encoding="utf-8") as f:
        json.dump(obj, f, ensure_ascii=False, separators=(",", ":"))


def download(url: str, destination: Path, timeout: int = 90) -> None:
    destination.parent.mkdir(parents=True, exist_ok=True)
    req = urllib.request.Request(
        url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "*/*",
            "Accept-Encoding": "identity",
        },
    )
    log(f"Downloading {url}")
    try:
        with urllib.request.urlopen(req, timeout=timeout) as resp, destination.open("wb") as out:
            shutil.copyfileobj(resp, out)
    except urllib.error.URLError as exc:
        raise RuntimeError(f"Failed to download {url}: {exc}") from exc


def strip_accents(value: str) -> str:
    return "".join(
        c for c in unicodedata.normalize("NFKD", value) if not unicodedata.combining(c)
    )


def norm(value: str) -> str:
    value = strip_accents(value).lower()
    value = value.replace("&", " and ")
    value = QUALITY_RE.sub(" ", value)
    value = re.sub(r"\b(?:channel|network|tv|television)\b", " ", value)
    return re.sub(r"[^a-z0-9]+", "", value)


def norm_id(value: str) -> str:
    return re.sub(r"[^a-z0-9]+", "", strip_accents(value).lower())


def clean_display_name(value: str) -> str:
    value = NOISE_RE.sub(" ", value)
    value = QUALITY_RE.sub(" ", value)
    value = SPACE_RE.sub(" ", value).strip(" -|•")
    return value or "Unnamed Channel"


def extract_quality(name: str, tvg_id: str = "") -> str | None:
    """Return the quality declared by a playlist, without probing the video."""
    # Free-TV/IPTV uses a circled S marker for some SD entries.
    if "Ⓢ" in name:
        return "SD"
    m = QUALITY_RE.search(name)
    if m:
        return m.group(1).upper().replace("2160P", "4K")

    # IPTV-org often stores a generic quality variant in the tvg-id suffix.
    if "@" in tvg_id:
        suffix = tvg_id.rsplit("@", 1)[1]
        m = QUALITY_RE.search(suffix)
        if m:
            return m.group(1).upper().replace("2160P", "4K")
    return None


def quality_height(stream: dict[str, Any]) -> int:
    measured_height = stream.get("measured_height")
    try:
        measured_height = int(measured_height or 0)
    except (TypeError, ValueError):
        measured_height = 0
    if measured_height > 0:
        return measured_height

    quality = (stream.get("measured_quality") or stream.get("quality") or "").upper()
    return {
        "4K": 2160, "UHD": 2160, "2160P": 2160,
        "1440P": 1440,
        "1080P": 1080, "FHD": 1080,
        "720P": 720, "HD": 720,
        "576P": 576, "540P": 540, "480P": 480, "SD": 480,
        "360P": 360, "240P": 240, "144P": 144,
    }.get(quality, 0)


def quality_label(stream: dict[str, Any], fallback: str = "Unknown") -> str:
    """Return measured rendition quality before an unverified source label."""
    measured = str(stream.get("measured_quality") or "").strip()
    if measured:
        return measured
    declared = str(stream.get("quality") or "").strip()
    return declared or fallback


def quality_label_for_height(height: int) -> str:
    if height >= 2160:
        return "4K"
    return f"{height}P" if height > 0 else "Unknown"


def is_hd_stream(stream: dict[str, Any]) -> bool:
    return quality_height(stream) >= 720


def is_sd_stream(stream: dict[str, Any]) -> bool:
    height = quality_height(stream)
    return 0 < height < 720

def parse_extinf(line: str) -> tuple[dict[str, str], str]:
    """Split EXTINF on the first comma outside quoted attributes."""
    body = line[len("#EXTINF:") :]
    in_quotes = False
    split_at = -1
    for i, ch in enumerate(body):
        if ch == '"':
            in_quotes = not in_quotes
        elif ch == "," and not in_quotes:
            split_at = i
            break
    if split_at >= 0:
        info, name = body[:split_at], body[split_at + 1 :]
    else:
        info, name = body, "Unnamed Channel"
    attrs = {k.lower(): v for k, v in ATTR_RE.findall(info)}
    return attrs, name.strip()


def parse_m3u(text: str, source: dict[str, Any]) -> list[dict[str, Any]]:
    entries: list[dict[str, Any]] = []
    pending: dict[str, Any] | None = None
    pending_headers: dict[str, str] = {}

    source_name = source.get("name", "IPTV")
    family = source.get("family", source_name)
    priority = int(source.get("priority", 50))
    is_philly = bool(source.get("philly"))

    for raw in text.splitlines():
        line = raw.strip()
        if not line:
            continue
        if line.startswith("#EXTINF:"):
            attrs, name = parse_extinf(line)
            tvg_id = attrs.get("tvg-id", "").strip()
            declared_quality = extract_quality(f"{name} {attrs.get('tvg-name', '')}", tvg_id)
            # Free-TV/IPTV documents that non-HD entries are marked with Ⓢ.
            # If an entry from that project is unmarked, treat it as generic HD.
            if declared_quality is None and family == "free-tv" and "Ⓢ" not in name:
                declared_quality = "HD"
            pending = {
                "name_raw": name,
                "name": clean_display_name(name),
                "quality": declared_quality,
                "tvg_id": tvg_id,
                "logo": attrs.get("tvg-logo", "").strip(),
                "group": attrs.get("group-title", "").strip(),
                "language": attrs.get("tvg-language", "").strip(),
                "country": attrs.get("tvg-country", "").strip(),
                "source": source_name,
                "family": family,
                "priority": priority,
                "philly": is_philly,
            }
            if source.get("content_kind"):
                pending["content_kind"] = source["content_kind"]
            if source.get("content_note"):
                pending["content_note"] = source["content_note"]
            pending_headers = {}
            continue
        if pending is None:
            continue
        if line.startswith("#EXTVLCOPT:http-user-agent="):
            pending_headers["User-Agent"] = line.split("=", 1)[1].strip()
            continue
        if line.startswith("#EXTVLCOPT:http-referrer=") or line.startswith("#EXTVLCOPT:http-referer="):
            pending_headers["Referer"] = line.split("=", 1)[1].strip()
            continue
        if line.startswith("#EXTVLCOPT:http-origin="):
            pending_headers["Origin"] = line.split("=", 1)[1].strip()
            continue
        if line.startswith("#EXTHTTP:"):
            try:
                obj = json.loads(line.split(":", 1)[1].strip())
                headers = obj.get("http-user-agent") or obj.get("headers") or obj
                if isinstance(headers, dict):
                    for k, v in headers.items():
                        if isinstance(v, str):
                            pending_headers[str(k)] = v
            except Exception:
                pass
            continue
        if line.startswith("#"):
            continue

        candidate_url = line
        inline_headers: dict[str, str] = {}
        if "|" in candidate_url and re.match(r"^https?://", candidate_url, re.I):
            candidate_url, header_blob = candidate_url.split("|", 1)
            for key, value in urllib.parse.parse_qsl(header_blob, keep_blank_values=True):
                normalized_key = key.strip().lower()
                if normalized_key == "user-agent":
                    inline_headers["User-Agent"] = value
                elif normalized_key in {"referer", "referrer"}:
                    inline_headers["Referer"] = value
                elif normalized_key == "origin":
                    inline_headers["Origin"] = value

        if re.match(r"^(?:https?|rtmp|rtmps|ftp)://", candidate_url, re.I):
            item = dict(pending)
            item["url"] = candidate_url
            headers = dict(pending_headers)
            headers.update(inline_headers)
            item["headers"] = headers
            entries.append(item)
        pending = None
        pending_headers = {}

    return entries
