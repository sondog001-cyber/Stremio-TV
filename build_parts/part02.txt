
    Example: FoxNewsChannel.us@SD and @HD should be one channel, while
    Fox.us@WTXFTV must remain distinct from other FOX affiliates.
    """
    value = (value or "").strip()
    if "@" not in value:
        return value.lower()
    base, suffix = value.rsplit("@", 1)
    suffix_norm = re.sub(r"[^a-z0-9]+", "", suffix.lower())
    if suffix_norm in _GENERIC_VARIANTS:
        return base.lower()
    return value.lower()

def playlist_key(entry: dict[str, Any]) -> str:
    tvg_id = entry.get("tvg_id", "").strip()
    if tvg_id:
        return "id:" + canonical_tvg_id(tvg_id)
    return "name:" + norm(entry.get("name", ""))


def entry_matches_pattern(entry: dict[str, Any], pattern: str) -> bool:
    """Conservative allow-list matching for a stream entry.

    Patterns that look like tvg-ids (contain a dot, e.g. ``HGTV.us``) are
    matched as channel IDs, not arbitrary substrings.  Human labels such as
    ``FOX 29`` are matched against display metadata.
    """
    pattern = (pattern or "").strip()
    if not pattern:
        return False

    if "." in pattern:
        entry_id = (entry.get("tvg_id") or "").strip()
        if not entry_id:
            return False
        return canonical_tvg_id(entry_id) == canonical_tvg_id(pattern)

    target = " ".join(
        [
            entry.get("tvg_id", ""),
            entry.get("name", ""),
            entry.get("name_raw", ""),
            entry.get("group", ""),
        ]
    )
    p = norm(pattern)
    return bool(p) and p in norm(target)

def _desired_id(pattern: str, curation: dict[str, Any]) -> str:
    return (curation.get("canonical_id_aliases") or {}).get(pattern, pattern)


def _logical_channel_key(value: str) -> str:
    """Normalize branding while preserving meaningful short names such as USA/HGTV."""
    value = (value or "").strip()
    value = re.sub(r"\.us(?:@.*)?$", "", value, flags=re.I)
    value = re.sub(r"([a-z0-9])([A-Z])", r"\1 \2", value)
    value = value.replace("&", " and ")
    value = QUALITY_RE.sub(" ", value)
    value = re.sub(r"\b(?:east|west|eastern|western|feed|adaptive)\b", " ", value, flags=re.I)
    value = re.sub(r"\b(?:channel|network|television)\b", " ", value, flags=re.I)
    tokens = re.findall(r"[A-Za-z0-9]+", value.lower())
    while len(tokens) > 1 and tokens[-1] in {"us", "usa"}:
        tokens.pop()
    return "".join(tokens)


def canonicalize_approved_sources(entries: list[dict[str, Any]], config: dict[str, Any]) -> list[dict[str, Any]]:
    """Map user-approved unverified playlists onto our curated logical channel IDs.

    This lets TVPass-style slug IDs and alternate network branding merge as
    backup streams instead of creating duplicate Stremio channels.
    """
    curation = config.get("curation") or {}
    approved = set(curation.get("approved_unverified_families") or [])
    alias_map = curation.get("approved_source_aliases") or {}
    if not approved:
        return entries

    desired_raw = [
        p for key in ("philly_allow", "national_allow")
        for p in curation.get(key, [])
        if "." in (p or "")
    ]
    desired: list[str] = []
    seen: set[str] = set()
    for p in desired_raw:
        target = _desired_id(p, curation)
        marker = canonical_tvg_id(target)
        if marker not in seen:
            seen.add(marker)
            desired.append(target)

    desired_by_exact = {canonical_tvg_id(p): p for p in desired}
    desired_by_key: dict[str, list[str]] = defaultdict(list)
    alias_by_key: dict[str, list[str]] = defaultdict(list)
    for p in desired:
        key = _logical_channel_key(p)
        if key:
            desired_by_key[key].append(p)
        for alias in alias_map.get(p, []):
            akey = _logical_channel_key(alias)
            if akey:
                alias_by_key[akey].append(p)

    out: list[dict[str, Any]] = []
    for entry in entries:
        if entry.get("family") not in approved:
            out.append(entry)
            continue

        target: str | None = None
        entry_id = (entry.get("tvg_id") or "").strip()
        if entry_id:
            target = desired_by_exact.get(canonical_tvg_id(entry_id))

        if target is None:
            keys = {
                _logical_channel_key(entry.get("name", "")),
                _logical_channel_key(entry.get("name_raw", "")),
                _logical_channel_key(entry_id),
            }
            keys.discard("")
            candidates: list[str] = []
            for key in keys:
                candidates.extend(desired_by_key.get(key, []))
                candidates.extend(alias_by_key.get(key, []))
            candidates = list(dict.fromkeys(candidates))
            if len(candidates) == 1:
                target = candidates[0]

        if target:
            item = dict(entry)
            item["source_tvg_id"] = entry_id
            item["tvg_id"] = target
            out.append(item)
        else:
            out.append(entry)
    return out


def build_allowlist_coverage(entries: list[dict[str, Any]], config: dict[str, Any]) -> list[dict[str, Any]]:
    """Report whether each exact tvg-id allow-list entry exists in any source.

    Human-label aliases are intentionally excluded because they are often duplicate
    spellings for the same channel. This report lets us find channels with zero
    candidate URLs, which needs-sources.json cannot see because it only covers
    channels that were actually built.
    """
    curation = config.get("curation") or {}
    patterns: list[tuple[str, str]] = []
    for kind, key in (("Philadelphia", "philly_allow"), ("National", "national_allow")):
        for pattern in curation.get(key, []):
            if "." in (pattern or ""):
                patterns.append((kind, _desired_id(pattern, curation)))

    rows: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    for kind, pattern in patterns:
        marker = (kind, canonical_tvg_id(pattern))
        if marker in seen:
            continue
        seen.add(marker)
        matches = [e for e in entries if entry_matches_pattern(e, pattern)]
        rows.append({
            "group": kind,
            "pattern": pattern,
            "found": bool(matches),
            "candidate_count": len({e.get("url") for e in matches if e.get("url")}),
            "sources": sorted({e.get("source") for e in matches if e.get("source")}),
            "qualities": sorted({e.get("quality") or "Unknown" for e in matches}),
        })
    return rows


def curate_entries(entries: list[dict[str, Any]], config: dict[str, Any]) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Apply the Philadelphia expanded-basic style allow-list.

    Every source is checked against the Philadelphia allow-list first.  This is
    important because a backup WPVI/WTXF stream may appear in a national/raw
    source rather than the dedicated Philadelphia playlist.
    """
    curation = config.get("curation") or {}
    if not curation.get("enabled"):
        return entries, []

    philly_allow = curation.get("philly_allow", [])
    national_allow = curation.get("national_allow", [])
    kept: list[dict[str, Any]] = []
    dropped: list[dict[str, Any]] = []
    for entry in entries:
        if any(entry_matches_pattern(entry, pat) for pat in philly_allow):
            entry = dict(entry)
            entry["philly"] = True
            kept.append(entry)
        elif any(entry_matches_pattern(entry, pat) for pat in national_allow):
            kept.append(entry)
        else:
            dropped.append(entry)
    return kept, dropped

def classify_channel(name: str, group: str) -> str:
    hay = f"{name} {group}".lower()

    mappings: list[tuple[str, tuple[str, ...]]] = [
        ("Home & Lifestyle", ("hgtv", "magnolia", "food", "cooking", "diy", "home", "lifestyle", "travel channel", "tlc")),
        ("News", ("news", "weather", "cnn", "msnbc", "cnbc", "fox business", "bloomberg", "c-span", "cspan")),
        ("Sports", ("sport", "espn", "nfl", "nba", "nhl", "mlb", "golf", "tennis", "racing", "fubo sports", "stadium")),
        ("Kids", ("kids", "children", "nickelodeon", "nick jr", "cartoon", "disney junior", "pbs kids")),
        ("Movies", ("movie", "cinema", "film", "movies", "hallmark", "paramount movie")),
        ("Documentary", ("documentary", "discovery", "history", "science", "smithsonian", "nat geo", "national geographic", "nature")),
        ("Music", ("music", "mtv", "vevo", "radio", "concert")),
        ("Entertainment", ("entertainment", "comedy", "reality", "drama", "general", "series", "family", "classic tv")),
    ]
    for category, words in mappings:
        if any(word in hay for word in words):
            return category
    return "Other"


def is_favorite(channel: dict[str, Any], favorite_patterns: list[str]) -> bool:
    hay = norm(channel.get("name", "")) + norm_id(channel.get("tvg_id", ""))
    return any(norm(pat) and norm(pat) in hay for pat in favorite_patterns)


def stream_priority(stream: dict[str, Any]) -> tuple[int, int, int, int, int, str]:
    """Lower tuple is better.

    For local-TV replacement candidates, content identity is more important than
    source reputation: a passing full-linear feed should beat a news-only FAST
    channel. Unknown existing sources remain between those two classes.
    """
    url = stream.get("url", "")
    content_kind = (stream.get("content_kind") or "unknown").lower()
    content_rank = {
        "full-linear": 0,
        "unknown": 1,
        "news-fast": 2,
        "event": 3,
    }.get(content_kind, 1)
    source_rank = int(stream.get("priority", 50))
    philly_rank = 0 if stream.get("philly") else 1
    https_rank = 0 if url.lower().startswith("https://") else 1
    # Negative height means higher declared resolution sorts first.
    return (content_rank, source_rank, philly_rank, -quality_height(stream), https_rank, url)


def _choose_prefer_new_family(candidates: list[dict[str, Any]], selected: list[dict[str, Any]]) -> dict[str, Any] | None:
    used_families = {s.get("family") for s in selected if s.get("family")}
    for candidate in candidates:
        if candidate.get("family") not in used_families:
            return candidate
    return candidates[0] if candidates else None
