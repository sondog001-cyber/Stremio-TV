
    Example: FoxNewsChannel.us@SD and @HD should be one channel, while
    Fox.us@WTXFTV must remain distinct from other FOX affiliates.
    """
    value = (value or "").strip()
    if "@" not in value:
        return value.lower()
    base, suffix = value.rsplit("@", 1)
    suffix_norm = re.sub(r"[^a-z0-9]+", "", suffix.lower())
    if suffix_norm in _GENERIC_VARIANTS:
        return base.lower()
    return value.lower()

def playlist_key(entry: dict[str, Any]) -> str:
    tvg_id = entry.get("tvg_id", "").strip()
    if tvg_id:
        return "id:" + canonical_tvg_id(tvg_id)
    return "name:" + norm(entry.get("name", ""))


def entry_matches_pattern(entry: dict[str, Any], pattern: str) -> bool:
    """Conservative allow-list matching for a stream entry.

    Patterns that look like tvg-ids (contain a dot, e.g. ``HGTV.us``) are
    matched as channel IDs, not arbitrary substrings.  Human labels such as
    ``FOX 29`` are matched against display metadata.
    """
    pattern = (pattern or "").strip()
    if not pattern:
        return False

    if "." in pattern:
        entry_id = (entry.get("tvg_id") or "").strip()
        if not entry_id:
            return False
        return canonical_tvg_id(entry_id) == canonical_tvg_id(pattern)

    target = " ".join(
        [
            entry.get("tvg_id", ""),
            entry.get("name", ""),
            entry.get("name_raw", ""),
            entry.get("group", ""),
        ]
    )
    p = norm(pattern)
    return bool(p) and p in norm(target)

def _desired_id(pattern: str, curation: dict[str, Any]) -> str:
    return (curation.get("canonical_id_aliases") or {}).get(pattern, pattern)


def _logical_channel_key(value: str) -> str:
    """Normalize branding while preserving meaningful short names such as USA/HGTV."""
    value = (value or "").strip()
    value = re.sub(r"\.us(?:@.*)?$", "", value, flags=re.I)
    value = re.sub(r"([a-z0-9])([A-Z])", r"\1 \2", value)
    value = value.replace("&", " and ")
    value = QUALITY_RE.sub(" ", value)
    value = re.sub(r"\b(?:east|west|eastern|western|feed|adaptive)\b", " ", value, flags=re.I)
    # Preserve meaningful service-type words such as "Channel" and "Network".
    # Dropping them caused false identities such as NFL Channel -> NFL Network.
    value = re.sub(r"\b(?:television)\b", " ", value, flags=re.I)
    tokens = re.findall(r"[A-Za-z0-9]+", value.lower())
    while len(tokens) > 1 and tokens[-1] in {"us", "usa"}:
        tokens.pop()
    return "".join(tokens)


def canonicalize_approved_sources(entries: list[dict[str, Any]], config: dict[str, Any]) -> list[dict[str, Any]]:
    """Map user-approved unverified playlists onto our curated logical channel IDs.

    This lets TVPass-style slug IDs and alternate network branding merge as
    backup streams instead of creating duplicate Stremio channels.
    """
    curation = config.get("curation") or {}
    approved = set(curation.get("approved_unverified_families") or [])
    alias_map = curation.get("approved_source_aliases") or {}
    canonical_aliases = {
        canonical_tvg_id(source_id): target_id
        for source_id, target_id in (curation.get("canonical_id_aliases") or {}).items()
    }
    explicitly_remapped: list[dict[str, Any]] = []
    for entry in entries:
        source_id = str(entry.get("tvg_id") or "").strip()
        target_id = canonical_aliases.get(canonical_tvg_id(source_id)) if source_id else None
        if target_id and canonical_tvg_id(source_id) != canonical_tvg_id(target_id):
            item = dict(entry)
            item["source_tvg_id"] = source_id
            item["tvg_id"] = target_id
            explicitly_remapped.append(item)
        else:
            explicitly_remapped.append(entry)
    entries = explicitly_remapped
    if not approved:
        return entries

    desired_raw = [
        p for key in ("philly_allow", "national_allow")
        for p in curation.get(key, [])
        if "." in (p or "")
    ]
    desired: list[str] = []
    seen: set[str] = set()
    for p in desired_raw:
        target = _desired_id(p, curation)
        marker = canonical_tvg_id(target)
        if marker not in seen:
            seen.add(marker)
            desired.append(target)

    desired_by_exact = {canonical_tvg_id(p): p for p in desired}
    desired_by_key: dict[str, list[str]] = defaultdict(list)
    alias_by_key: dict[str, list[str]] = defaultdict(list)
    for p in desired:
        key = _logical_channel_key(p)
        if key:
            desired_by_key[key].append(p)
        for alias in alias_map.get(p, []):
            akey = _logical_channel_key(alias)
            if akey:
                alias_by_key[akey].append(p)

    out: list[dict[str, Any]] = []
    for entry in entries:
        if entry.get("family") not in approved:
            out.append(entry)
            continue

        target: str | None = None
        entry_id = (entry.get("tvg_id") or "").strip()
        if entry_id:
            target = desired_by_exact.get(canonical_tvg_id(entry_id))

        if target is None:
            keys = {
                _logical_channel_key(entry.get("name", "")),
                _logical_channel_key(entry.get("name_raw", "")),
                _logical_channel_key(entry_id),
            }
            keys.discard("")
            candidates: list[str] = []
            for key in keys:
                candidates.extend(desired_by_key.get(key, []))
                candidates.extend(alias_by_key.get(key, []))
            candidates = list(dict.fromkeys(candidates))
            if len(candidates) == 1:
                target = candidates[0]

        if target:
            item = dict(entry)
            item["source_tvg_id"] = entry_id
            item["tvg_id"] = target
            out.append(item)
        else:
            out.append(entry)
    return out


def enforce_exact_identity(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Reject known lookalikes before probing or allowlist coverage."""
    rules = (config.get("curation") or {}).get("identity_rules") or {}
    if not rules:
        return entries, []

    rules_by_id = {
        canonical_tvg_id(tvg_id): (tvg_id, rule)
        for tvg_id, rule in rules.items()
    }
    kept: list[dict[str, Any]] = []
    rejected: list[dict[str, Any]] = []

    for entry in entries:
        target_id = (entry.get("tvg_id") or "").strip()
        rule_tuple = rules_by_id.get(canonical_tvg_id(target_id))
        if not rule_tuple:
            kept.append(entry)
            continue

        _rule_id, rule = rule_tuple
        accept_keys = {
            _logical_channel_key(x)
            for x in (rule.get("accept") or [])
            if _logical_channel_key(x)
        }
        reject_keys = {
            _logical_channel_key(x)
            for x in (rule.get("reject") or [])
            if _logical_channel_key(x)
        }

        source_id = (entry.get("source_tvg_id") or "").strip()
        identity_values = [
            entry.get("name", ""),
            entry.get("name_raw", ""),
            source_id,
        ]
        identity_keys = {
            _logical_channel_key(x)
            for x in identity_values
            if _logical_channel_key(x)
        }

        reject_hit = sorted(identity_keys & reject_keys)
        if reject_hit:
            rejected.append({
                "target_tvg_id": target_id,
                "name": entry.get("name"),
                "name_raw": entry.get("name_raw"),
                "source_tvg_id": source_id,
                "source": entry.get("source"),
                "family": entry.get("family"),
                "url": entry.get("url"),
                "reason": "explicit-lookalike-reject",
                "matched_identity_keys": reject_hit,
            })
            continue

        remapped = bool(source_id) and canonical_tvg_id(source_id) != canonical_tvg_id(target_id)
        if remapped and accept_keys and not (identity_keys & accept_keys):
            rejected.append({
                "target_tvg_id": target_id,
                "name": entry.get("name"),
                "name_raw": entry.get("name_raw"),
                "source_tvg_id": source_id,
                "source": entry.get("source"),
                "family": entry.get("family"),
                "url": entry.get("url"),
                "reason": "remapped-entry-did-not-match-exact-identity",
                "accepted_aliases": rule.get("accept") or [],
            })
            continue

        kept.append(entry)

    if rejected:
        log(f"Exact identity gate: rejected {len(rejected):,} lookalike/remapped candidates")
    return kept, rejected


def apply_stream_quarantine(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Remove streams with confirmed playback defects before health probing.

    HTTP/media reachability cannot detect every decoder-visible defect.  This
    list is intentionally exact and channel-scoped so one bad endpoint can be
    suppressed without blacklisting an entire source family or host.
    """
    rules = (config.get("curation") or {}).get("stream_quarantine") or []
    if not rules:
        return entries, []

    normalized_rules: list[dict[str, Any]] = []
    for rule in rules:
        if not isinstance(rule, dict):
            continue
        url = str(rule.get("url") or "").strip()
        if not url:
            continue
        tvg_id = str(rule.get("tvg_id") or "").strip()
        normalized_rules.append({
            **rule,
            "url": url,
            "tvg_id": tvg_id,
            "_canonical_tvg_id": canonical_tvg_id(tvg_id) if tvg_id else "",
        })

    kept: list[dict[str, Any]] = []
    quarantined: list[dict[str, Any]] = []
    for entry in entries:
        entry_url = str(entry.get("url") or "").strip()
        entry_id = canonical_tvg_id(str(entry.get("tvg_id") or ""))
        matched = next((
            rule for rule in normalized_rules
            if entry_url == rule["url"]
            and (not rule["_canonical_tvg_id"] or entry_id == rule["_canonical_tvg_id"])
        ), None)
        if matched is None:
            kept.append(entry)
            continue

        quarantined.append({
            "name": entry.get("name"),
            "tvg_id": entry.get("tvg_id"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "url": entry_url,
            "reason": matched.get("reason") or "Manual playback-defect quarantine",
            "observed_at": matched.get("observed_at"),
            "evidence": matched.get("evidence"),
        })

    if quarantined:
        log(f"Playback-defect quarantine: removed {len(quarantined):,} candidate stream(s)")
    return kept, quarantined


def build_allowlist_coverage(entries: list[dict[str, Any]], config: dict[str, Any]) -> list[dict[str, Any]]:
    """Report whether each exact tvg-id allow-list entry exists in any source.

    Human-label aliases are intentionally excluded because they are often duplicate
    spellings for the same channel. This report lets us find channels with zero
    candidate URLs, which needs-sources.json cannot see because it only covers
    channels that were actually built.
    """
    curation = config.get("curation") or {}
    patterns: list[tuple[str, str]] = []
    for kind, key in (("Philadelphia", "philly_allow"), ("National", "national_allow")):
        for pattern in curation.get(key, []):
            if "." in (pattern or ""):
                patterns.append((kind, _desired_id(pattern, curation)))

    rows: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    for kind, pattern in patterns:
        marker = (kind, canonical_tvg_id(pattern))
        if marker in seen:
            continue
        seen.add(marker)
        matches = [e for e in entries if entry_matches_pattern(e, pattern)]
        rows.append({
            "group": kind,
            "pattern": pattern,
            "found": bool(matches),
            "candidate_count": len({e.get("url") for e in matches if e.get("url")}),
            "sources": sorted({e.get("source") for e in matches if e.get("source")}),
            "qualities": sorted({quality_label(e) for e in matches}),
            "declared_qualities": sorted({e.get("quality") or "Unknown" for e in matches}),
            "measured_qualities": sorted({e.get("measured_quality") for e in matches if e.get("measured_quality")}),
        })
    return rows


def curate_entries(entries: list[dict[str, Any]], config: dict[str, Any]) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Apply the Philadelphia expanded-basic style allow-list.

    Every source is checked against the Philadelphia allow-list first.  This is
    important because a backup WPVI/WTXF stream may appear in a national/raw
    source rather than the dedicated Philadelphia playlist.
    """
    curation = config.get("curation") or {}
    if not curation.get("enabled"):
        return entries, []

    philly_allow = curation.get("philly_allow", [])
    national_allow = curation.get("national_allow", [])
    kept: list[dict[str, Any]] = []
    dropped: list[dict[str, Any]] = []
    for entry in entries:
        if any(entry_matches_pattern(entry, pat) for pat in philly_allow):
            entry = dict(entry)
            entry["philly"] = True
            kept.append(entry)
        elif any(entry_matches_pattern(entry, pat) for pat in national_allow):
            kept.append(entry)
        else:
            dropped.append(entry)
    return kept, dropped

def classify_channel(name: str, group: str) -> str:
    hay = f"{name} {group}".lower()

    mappings: list[tuple[str, tuple[str, ...]]] = [
        ("Home & Lifestyle", ("hgtv", "magnolia", "food", "cooking", "diy", "home", "lifestyle", "travel channel", "tlc")),
        ("News", ("news", "weather", "cnn", "msnbc", "cnbc", "fox business", "bloomberg", "c-span", "cspan")),
        ("Sports", ("sport", "espn", "nfl", "nba", "nhl", "mlb", "golf", "tennis", "racing", "fubo sports", "stadium")),
        ("Kids", ("kids", "children", "nickelodeon", "nick jr", "cartoon", "disney junior", "pbs kids")),
        ("Movies", ("movie", "cinema", "film", "movies", "hallmark", "paramount movie")),
        ("Documentary", ("documentary", "discovery", "history", "science", "smithsonian", "nat geo", "national geographic", "nature")),
        ("Music", ("music", "mtv", "vevo", "radio", "concert")),
        ("Entertainment", ("entertainment", "comedy", "reality", "drama", "general", "series", "family", "classic tv")),
    ]
    for category, words in mappings:
        if any(word in hay for word in words):
            return category
    return "Other"


def is_favorite(channel: dict[str, Any], favorite_patterns: list[str]) -> bool:
    hay = norm(channel.get("name", "")) + norm_id(channel.get("tvg_id", ""))
    return any(norm(pat) and norm(pat) in hay for pat in favorite_patterns)


def channel_preference_rules(config: dict[str, Any], channel_id: str) -> list[dict[str, str]]:
    """Return exact-channel source rules, falling back to the default family order."""
    settings = config.get("stream_selection") or {}
    configured = settings.get("channel_preferences") or {}
    raw_rules: Any = None
    for configured_id, value in configured.items():
        if canonical_tvg_id(configured_id) == canonical_tvg_id(channel_id):
            raw_rules = value
            break

    if raw_rules is None:
        families: list[str] = []
        primary = str(settings.get("primary_family") or "").strip()
        if primary:
            families.append(primary)
        families.extend(str(item).strip() for item in settings.get("fallback_families", []) if str(item).strip())
        raw_rules = families
    elif isinstance(raw_rules, dict):
        raw_rules = raw_rules.get("sources") or []

    rules: list[dict[str, str]] = []
    for raw in raw_rules or []:
        if isinstance(raw, str):
            rules.append({"family": raw, "label": raw})
        elif isinstance(raw, dict):
            rule = {str(key): str(value) for key, value in raw.items() if value is not None}
            if rule:
                rule.setdefault("label", rule.get("family") or rule.get("source_contains") or "preferred")
                rules.append(rule)
    return rules


def _stream_preference_match(
    stream: dict[str, Any],
    preference_rules: list[dict[str, str]],
) -> tuple[int, str]:
    if not preference_rules:
        return 0, "stability-default"
    for index, rule in enumerate(preference_rules):
        family = rule.get("family")
        content_kind = rule.get("content_kind")
        source_contains = rule.get("source_contains")
        url_contains = rule.get("url_contains")
        if family and str(stream.get("family") or "").lower() != family.lower():
            continue
        if content_kind and str(stream.get("content_kind") or "unknown").lower() != content_kind.lower():
            continue
        if source_contains and source_contains.lower() not in str(stream.get("source") or "").lower():
            continue
        if url_contains and url_contains.lower() not in str(stream.get("url") or "").lower():
            continue
        return index, rule.get("label") or family or "preferred"
    return len(preference_rules) + 1, "unlisted-source"


def stream_effective_url(stream: dict[str, Any]) -> str:
    """Return the post-redirect media URL when probing exposed one."""
    return str(stream.get("effective_url") or stream.get("url") or "").strip()


def stream_provider_host(stream: dict[str, Any]) -> str:
    """Collapse CDN subdomains into one provider identity for real redundancy."""
    url = stream_effective_url(stream)
    try:
        host = (urllib.parse.urlsplit(url).hostname or "").lower().rstrip(".")
    except ValueError:
        return ""
    if not host:
        return ""
    try:
        ipaddress.ip_address(host)
        return host
    except ValueError:
        pass
    labels = [label for label in host.split(".") if label]
    if len(labels) <= 2:
        return host
    multipart_suffixes = {"co.uk", "com.au", "co.nz", "com.br", "co.jp"}
    suffix = ".".join(labels[-2:])
    width = 3 if suffix in multipart_suffixes and len(labels) >= 3 else 2
    return ".".join(labels[-width:])


def stream_priority(
    stream: dict[str, Any],
    preference_rules: list[dict[str, str]] | None = None,
) -> tuple[int, int, int, int, int, int, int, int, int, str]:
    """Lower tuple is better.

    Content identity remains the first guardrail, then cross-build stability
    determines which currently-passing URL should become Primary. A fresh URL
    that only passed this build (Quarantine) cannot outrank an established
    Stable/Backup URL of the same content class.
    """
    url = stream.get("url", "")
    content_kind = (stream.get("content_kind") or "unknown").lower()
    content_rank = {
        "full-linear": 0,
        "unknown": 1,
        "news-fast": 2,
        "event": 3,
    }.get(content_kind, 1)

    stability = stream.get("stability") or {}
    stability_rank = int(stability.get("rank", 2))
    # More successful builds break ties inside the same class.
    stability_passes_rank = -int(stability.get("passes", 0))
    stability_samples_rank = -int(stability.get("samples", 0))
    preference_rank, _preference_label = _stream_preference_match(stream, preference_rules or [])

    source_rank = int(stream.get("priority", 50))
    philly_rank = 0 if stream.get("philly") else 1
    https_rank = 0 if url.lower().startswith("https://") else 1
    # Negative height means higher declared resolution sorts first.
    return (
        content_rank,
        stability_rank,
        stability_passes_rank,
        stability_samples_rank,
        preference_rank,
        source_rank,
        philly_rank,
        -quality_height(stream),
        https_rank,
        url,
    )


def _choose_prefer_new_family(candidates: list[dict[str, Any]], selected: list[dict[str, Any]]) -> dict[str, Any] | None:
    used_families = {s.get("family") for s in selected if s.get("family")}
    for candidate in candidates:
        if candidate.get("family") not in used_families:
            return candidate
    return candidates[0] if candidates else None


def _choose_backup_candidate(
    candidates: list[dict[str, Any]],
    selected: list[dict[str, Any]],
    *,
    prefer_hosts: bool,
    prefer_families: bool,
) -> dict[str, Any] | None:
    pool = candidates
    if prefer_hosts:
        used_hosts = {stream_provider_host(item) for item in selected if stream_provider_host(item)}
        independent = [item for item in pool if stream_provider_host(item) not in used_hosts]
        if independent:
            pool = independent
    return _choose_prefer_new_family(pool, selected) if prefer_families else (pool[0] if pool else None)


def select_streams(
    streams: list[dict[str, Any]],
    max_streams: int,
    config: dict[str, Any],
    channel_id: str = "",
) -> list[dict[str, Any]]:
    """Pick at most 3 links, targeting two HD and one SD fallback.

    Selection is based on the quality label declared by the upstream playlist;
    it does not decode/probe every HLS rendition during the static build.
    """
    if not channel_id:
        channel_id = next((str(stream.get("tvg_id") or "") for stream in streams if stream.get("tvg_id")), "")
    preference_rules = channel_preference_rules(config, channel_id)
    priority_key = lambda stream: stream_priority(stream, preference_rules)

    unique: list[dict[str, Any]] = []
    seen_urls: set[str] = set()
    for s in sorted(streams, key=priority_key):
        url = stream_effective_url(s)
        if not url or url in seen_urls:
            continue
        seen_urls.add(url)
        s["effective_url"] = url
        s["provider_host"] = stream_provider_host(s)
        unique.append(s)

    target = config.get("stream_selection") or {}
    target_hd = min(int(target.get("target_hd", 2)), max_streams)
    target_sd = min(int(target.get("target_sd", 1)), max(0, max_streams - target_hd))
    prefer_families = bool(target.get("prefer_distinct_families", True))
    prefer_hosts = bool(target.get("prefer_distinct_provider_hosts", True))

    selected: list[dict[str, Any]] = []
    if unique:
        primary = dict(unique[0])
        primary["slot"] = "HD 1" if is_hd_stream(primary) else "SD" if is_sd_stream(primary) else "Primary"
        selected.append(primary)
        unique[0] = primary

    def remaining(predicate):
        chosen = {id(x) for x in selected}
        return [s for s in unique if id(s) not in chosen and predicate(s)]

    current_hd = sum(1 for stream in selected if is_hd_stream(stream))
    for hd_index in range(current_hd, target_hd):
        candidates = remaining(is_hd_stream)
        if not candidates:
            break
        pick = _choose_backup_candidate(
            candidates,
            selected,
            prefer_hosts=prefer_hosts,
            prefer_families=prefer_families,
        )
        item = dict(pick)
        item["slot"] = f"HD {hd_index + 1}"
        selected.append(item)
        # Replace the original object in unique with the copied object identity
        # tracking by URL rather than id for subsequent passes.
        for i, u in enumerate(unique):
            if u.get(
