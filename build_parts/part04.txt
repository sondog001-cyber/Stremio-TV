                return epg_id, "fuzzy-name", best_score

    return None, "unmatched", 0.0


def parse_xmltv_time(value: str) -> dt.datetime | None:
    value = (value or "").strip()
    if not value:
        return None
    # XMLTV often uses: YYYYMMDDHHMMSS ±HHMM
    formats = ["%Y%m%d%H%M%S %z", "%Y%m%d%H%M %z", "%Y%m%d%H%M%S", "%Y%m%d%H%M"]
    for fmt in formats:
        try:
            parsed = dt.datetime.strptime(value, fmt)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=UTC)
            return parsed.astimezone(UTC)
        except ValueError:
            continue
    return None


def iso_z(value: dt.datetime) -> str:
    return value.astimezone(UTC).isoformat(timespec="seconds").replace("+00:00", "Z")


def date_range_for_program(start: dt.datetime, end: dt.datetime) -> list[str]:
    if end <= start:
        return []
    cursor = start.date()
    last = (end - dt.timedelta(microseconds=1)).date()
    result: list[str] = []
    while cursor <= last:
        result.append(cursor.isoformat())
        cursor += dt.timedelta(days=1)
    return result


def build_program_index(
    epg_path: Path,
    epg_to_channels: dict[str, list[str]],
) -> tuple[dict[str, dict[str, list[dict[str, Any]]]], set[str]]:
    schedules: dict[str, dict[str, list[dict[str, Any]]]] = defaultdict(lambda: defaultdict(list))
    all_dates: set[str] = set()
    matched_programmes = 0

    with gzip.open(epg_path, "rb") as f:
        for event, elem in ET.iterparse(f, events=("end",)):
            if elem.tag != "programme":
                if elem.tag == "channel":
                    elem.clear()
                continue

            epg_id = elem.attrib.get("channel", "")
            channel_ids = epg_to_channels.get(epg_id)
            if not channel_ids:
                elem.clear()
                continue

            start = parse_xmltv_time(elem.attrib.get("start", ""))
            end = parse_xmltv_time(elem.attrib.get("stop", ""))
            if not start or not end or end <= start:
                elem.clear()
                continue

            title = extract_text(elem, "title") or "Untitled"
            desc = extract_text(elem, "desc")
            categories = [
                (x.text or "").strip()
                for x in elem.findall("category")
                if (x.text or "").strip()
            ]
            icon_elem = elem.find("icon")
            thumbnail = icon_elem.attrib.get("src", "") if icon_elem is not None else ""
            date_text = extract_text(elem, "date")
            runtime_min = max(1, round((end - start).total_seconds() / 60))
            start_iso = iso_z(start)
            end_iso = iso_z(end)

            for channel_id in channel_ids:
                programme: dict[str, Any] = {
                    "id": f"{channel_id}:epg:{start_iso}",
                    "title": title,
                    "released": start_iso,
                    "startTime": start_iso,
                    "endTime": end_iso,
                    "runtime": f"{runtime_min} min",
                }
                if desc:
                    programme["overview"] = desc
                if thumbnail:
                    programme["thumbnail"] = thumbnail
                if categories:
                    programme["genres"] = categories[:8]
                if date_text[:4].isdigit():
                    programme["releaseInfo"] = date_text[:4]

                for day in date_range_for_program(start, end):
                    schedules[channel_id][day].append(programme)
                    all_dates.add(day)
            matched_programmes += 1
            elem.clear()

    for channel_days in schedules.values():
        for day, programmes in channel_days.items():
            programmes.sort(key=lambda p: p["startTime"])

    log(f"Matched {matched_programmes:,} XMLTV programmes to playlist channels")
    return schedules, all_dates

_PUBLIC_TEXT_CACHE: dict[tuple[str, int], tuple[str, str]] = {}


def _fetch_public_text(url: str, timeout: int = 12, byte_limit: int = 2_000_000) -> tuple[str, str]:
    """Fetch a public page/API response without credentials or session cookies."""
    cache_key = (url, byte_limit)
    if cache_key in _PUBLIC_TEXT_CACHE:
        return _PUBLIC_TEXT_CACHE[cache_key]
    req = urllib.request.Request(
        url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "text/html,application/json,text/plain,*/*",
            "Accept-Encoding": "identity",
        },
    )
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        final_url = resp.geturl()
        body = resp.read(byte_limit)
    result = (body.decode("utf-8", errors="ignore"), final_url)
    _PUBLIC_TEXT_CACHE[cache_key] = result
    return result


def _extract_public_hls_urls(text: str, base_url: str = "") -> list[str]:
    """Extract plainly published HLS URLs from public text/JSON.

    This intentionally does not copy cookies, bearer tokens, DRM data, or attempt
    to manufacture signed URLs. Query strings that are already present in public
    page source are preserved and still have to pass the health checker.
    """
    normalized = html.unescape(text or "")
    normalized = normalized.replace("\\u002F", "/").replace("\\/", "/")
    normalized = normalized.replace("\\u0026", "&")

    found: list[str] = []
    absolute = re.findall(
        r'https?://[^\s\\"\'<>]+?\.m3u8(?:\?[^\s\\"\'<>]*)?',
        normalized,
        flags=re.I,
    )
    for url in absolute:
        found.append(url.rstrip("),.;"))

    if base_url:
        relative = re.findall(
            r'(?:(?:src|url|file|stream|hls)[\"\'=: ]+)(/[^\s\\"\'<>]+?\.m3u8(?:\?[^\s\\"\'<>]*)?)',
            normalized,
            flags=re.I,
        )
        for path in relative:
            found.append(urllib.parse.urljoin(base_url, path.rstrip("),.;")))

    return list(dict.fromkeys(found))


def _exact_allowlist_ids(config: dict[str, Any], scope: str = "all") -> list[str]:
    """Return deduplicated exact IDs for all, Philly, or national scope."""
    curation = config.get("curation") or {}
    local_settings = (config.get("discovery") or {}).get("philly_local_sources") or {}
    ids: list[str] = []
    seen: set[str] = set()
    keys = {
        "all": ("philly_allow", "national_allow"),
        "philly": ("philly_allow",),
        "national": ("national_allow",),
    }.get(scope)
    if keys is None:
        raise ValueError(f"Unknown allowlist scope: {scope}")
    for key in keys:
        patterns = curation.get(key, [])
        if scope == "philly" and key == "philly_allow" and local_settings.get("target_ids"):
            patterns = local_settings["target_ids"]
        for pattern in patterns:
            if "." not in (pattern or ""):
                continue
            target = _desired_id(pattern, curation)
            marker = canonical_tvg_id(target)
            if marker in seen:
                continue
            seen.add(marker)
            ids.append(target)
    return ids


def _target_ids_for_source_hunt(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
    scope: str = "all",
) -> list[str]:
    """Target only current candidate gaps plus channels missing last build.

    A source can disappear before health probing, so the previous deployed channel
    state is included alongside exact IDs that have no candidate in this build.
    The union means a recovered URL is actively hunted without feeding every
    unrelated upstream channel into the health checker.
    """
    in_scope = {
        canonical_tvg_id(item): item
        for item in _exact_allowlist_ids(config, scope=scope)
    }
    targets: dict[str, str] = {}
    coverage_entries = canonicalize_approved_sources(existing_entries, config)
    for row in build_allowlist_coverage(coverage_entries, config):
        if not row.get("found"):
            pattern = str(row.get("pattern") or "")
            if pattern and canonical_tvg_id(pattern) in in_scope:
                targets[canonical_tvg_id(pattern)] = pattern

    previous = _load_previous_channel_state(config)
    for record in (previous.get("channels") or {}).values():
        if not isinstance(record, dict) or record.get("present"):
            continue
        pattern = str(record.get("pattern") or "")
        if pattern and canonical_tvg_id(pattern) in in_scope:
            targets[canonical_tvg_id(pattern)] = pattern

    allow_order = _exact_allowlist_ids(config, scope=scope)
    return [item for item in allow_order if canonical_tvg_id(item) in targets]


def _filter_entries_to_exact_targets(
    entries: list[dict[str, Any]],
    target_ids: list[str],
    config: dict[str, Any],
) -> list[dict[str, Any]]:
    """Canonicalize a source and retain only the requested exact logical IDs."""
    wanted = {canonical_tvg_id(item) for item in target_ids}
    canonicalized = canonicalize_approved_sources(entries, config)
    return [
        entry
        for entry in canonicalized
        if canonical_tvg_id(entry.get("tvg_id", "")) in wanted
    ]


def _parse_aria_markdown(text: str, source: dict[str, Any]) -> tuple[list[dict[str, Any]], int]:
    """Parse aria-tv's small US channel table without loading its global M3U."""
    entries: list[dict[str, Any]] = []
    scanned = 0
    row_re = re.compile(
        r"^\|\s*\d+\s*\|\s*(?P<name>[^|]+?)\s*\|\s*\[>\]\((?P<url>[^)]+)\)\s*\|(?P<rest>.*)$",
        re.I,
    )
    for raw in text.splitlines():
        match = row_re.match(raw.strip())
        if not match:
            continue
        scanned += 1
        name = html.unescape(match.group("name").strip())
        url = html.unescape(match.group("url").strip())
        rest = match.group("rest")
        fields = [field.strip() for field in rest.split("|") if field.strip()]
        status = fields[-1].lower() if fields else ""
        if status == "not-working" or not re.match(r"^https?://", url, re.I):
            continue
        logo_match = re.search(r'<img[^>]+src="([^"]+)"', rest, re.I)
        entries.append({
            "name_raw": name,
            "name": clean_display_name(name),
            "quality": extract_quality(name),
            "tvg_id": "",
            "logo": html.unescape(logo_match.group(1)) if logo_match else "",
            "group": "",
            "language": "English",
            "country": "US",
            "source": source.get("name", "aria-tv US"),
            "family": source.get("family", "aria-tv"),
            "priority": int(source.get("priority", 8)),
            "philly": bool(source.get("philly", False)),
            "url": url,
            "headers": {},
            "source_status": status or "unknown",
        })
    return entries, scanned


def discover_targeted_source_families(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[str]]:
    """Scan high-value public source families only for remaining exact IDs."""
    settings = (config.get("discovery") or {}).get("targeted_source_families") or {}
    if not settings.get("enabled", True):
        return [], [], []

    target_ids = _target_ids_for_source_hunt(config, existing_entries, scope="national")
    sources = settings.get("sources") or []
    if not target_ids or not sources:
        return [], [{
            "kind": "targeted-source-summary",
            "scope": "national",
            "status": "nothing-to-scan",
            "target_ids": target_ids,
            "source_count": len(sources),
        }], target_ids

    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    timeout = int(settings.get("timeout_seconds", 20))
    byte_limit = int(settings.get("byte_limit", 8_000_000))
    default_cap = max(1, int(settings.get("max_candidates_per_target", 4)))

    for source in sources:
        url = str(source.get("url") or "").strip()
        fmt = str(source.get("format") or "m3u").lower()
        row: dict[str, Any] = {
            "kind": "targeted-source-family",
            "scope": "national",
            "name": source.get("name"),
            "family": source.get("family"),
            "url": url,
            "format": fmt,
            "target_count": len(target_ids),
            "target_ids": target_ids,
            "status": "pending",
            "records_scanned": 0,
            "candidates_found": 0,
            "matched_ids": [],
        }
        try:
            text, final_url = _fetch_public_text(
                url,
                timeout=int(source.get("timeout_seconds", timeout)),
                byte_limit=int(source.get("byte_limit", byte_limit)),
            )
            source_info = dict(source)
            if fmt == "aria-markdown":
                parsed, scanned = _parse_aria_markdown(text, source_info)
            else:
                parsed = parse_m3u(text, source_info)
                scanned = len(parsed)
            matched = _filter_entries_to_exact_targets(parsed, target_ids, config)
            accepted: list[dict[str, Any]] = []
            per_target: dict[str, int] = defaultdict(int)
            source_cap = max(1, int(source.get("max_candidates_per_target", default_cap)))
            for entry in matched:
                target_marker = canonical_tvg_id(entry.get("tvg_id", ""))
                if per_target[target_marker] >= source_cap:
                    continue
                marker = (target_marker, entry.get("url", ""))
                if marker in seen:
                    continue
                seen.add(marker)
                per_target[target_marker] += 1
                entry["targeted_scan"] = True
                discovered.append(entry)
                accepted.append(entry)

            row["status"] = "ok"
            row["final_url"] = final_url
            row["records_scanned"] = scanned
            row["matches_before_cap"] = len(matched)
            row["candidates_found"] = len(accepted)
            row["matched_ids"] = sorted({entry.get("tvg_id") for entry in accepted if entry.get("tvg_id")})
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    rows.insert(0, {
        "kind": "targeted-source-summary",
        "scope": "national",
        "status": "complete",
        "target_ids": target_ids,
        "source_count": len(sources),
        "candidates_found": len(discovered),
    })
    log(
        f"Targeted source-family scan: {len(target_ids)} exact IDs across "
        f"{len(sources)} sources; {len(discovered)} candidates retained"
    )
    return discovered, rows, target_ids


def discover_philly_local_sources(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[str]]:
    """Hunt Philadelphia station sources independently of national cable."""
    settings = (config.get("discovery") or {}).get("philly_local_sources") or {}
    if not settings.get("enabled", True):
        return [], [], []

    target_ids = _target_ids_for_source_hunt(config, existing_entries, scope="philly")
    sources = settings.get("playlists") or []
    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    timeout = int(settings.get("timeout_seconds", 20))
    byte_limit = int(settings.get("byte_limit", 8_000_000))
    default_cap = max(1, int(settings.get("max_candidates_per_target", 6)))

    for source in sources:
        url = str(source.get("url") or "").strip()
        row: dict[str, Any] = {
            "kind": "philly-local-playlist",
            "scope": "philly-local",
            "name": source.get("name"),
            "family": source.get("family"),
            "url": url,
            "target_count": len(target_ids),
            "target_ids": target_ids,
            "status": "pending",
            "records_scanned": 0,
            "candidates_found": 0,
            "matched_ids": [],
        }
        if not target_ids:
            row["status"] = "nothing-to-scan"
            rows.append(row)
            continue
        try:
            text, final_url = _fetch_public_text(
                url,
                timeout=int(source.get("timeout_seconds", timeout)),
                byte_limit=int(source.get("byte_limit", byte_limit)),
            )
            parsed = parse_m3u(text, dict(source))
            matched = _filter_entries_to_exact_targets(parsed, target_ids, config)
            accepted: list[dict[str, Any]] = []
            per_target: dict[str, int] = defaultdict(int)
            source_cap = max(1, int(source.get("max_candidates_per_target", default_cap)))
            for entry in matched:
                target_marker = canonical_tvg_id(entry.get("tvg_id", ""))
                if per_target[target_marker] >= source_cap:
                    continue
                marker = (target_marker, entry.get("url", ""))
                if marker in seen:
                    continue
                seen.add(marker)
                per_target[target_marker] += 1
                entry["philly"] = True
                entry["philly_local_scan"] = True
                discovered.append(entry)
                accepted.append(entry)
            row["status"] = "ok"
            row["final_url"] = final_url
            row["records_scanned"] = len(parsed)
            row["matches_before_cap"] = len(matched)
            row["candidates_found"] = len(accepted)
            row["matched_ids"] = sorted({entry.get("tvg_id") for entry in accepted if entry.get("tvg_id")})
        except Exception as exc:

            row["error"] = str(exc)[:240]
        rows.append(row)

        config,
        items=settings.get("official_pages") or [],
        target_ids=target_ids,
        row_kind="philly-official-page",
    )
    discovered.extend(official_entries)
    rows.extend(official_rows)
    rows.insert(0, {
        "kind": "philly-local-summary",
        "scope": "philly-local",
        "status": "complete" if target_ids else "nothing-to-scan",
        "target_ids": target_ids,
        "source_count": len(sources),
        "official_page_count": len(settings.get("official_pages") or []),
        "candidates_found": len(discovered),
    })
    log(
        f"Philadelphia local scan: {len(target_ids)} exact IDs across "
        f"{len(sources)} playlists and {len(settings.get('official_pages') or [])} official pages; "
        f"{len(discovered)} candidates retained"
    )
    return discovered, rows, target_ids


def discover_official_page_streams(
    config: dict[str, Any],
    items: list[dict[str, Any]] | None = None,
    target_ids: list[str] | None = None,
    row_kind: str = "official-page",
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    settings = items if items is not None else (config.get("discovery") or {}).get("official_pages") or []
    wanted = {canonical_tvg_id(item) for item in target_ids} if target_ids is not None else None

    for item in settings:
        page_url = item.get("page_url")
        tvg_id = item.get("tvg_id")
        if not page_url or not tvg_id:
            continue
        if wanted is not None and canonical_tvg_id(tvg_id) not in wanted:
            continue

        row: dict[str, Any] = {
            "kind": row_kind,
            "scope": "philly-local" if row_kind.startswith("philly-") else "shared",
            "name": item.get("name") or tvg_id,
            "tvg_id": tvg_id,
            "page_url": page_url,
            "status": "pending",
            "candidates": [],
        }
        try:
            page_text, final_url = _fetch_public_text(
                page_url,
                timeout=int(item.get("timeout_seconds", 12)),
            )
            urls = _extract_public_hls_urls(page_text, final_url)

            # Guard against unrelated clip/ad manifests on a station page. If the
            # page publishes several HLS URLs, keep only URLs whose nearby source
            # context looks like a live/linear stream.
            accepted: list[str] = []
            normalized = html.unescape(page_text).replace("\\u002F", "/").replace("\\/", "/")
            for url in urls:
                idx = normalized.find(url)
                context = normalized[max(0, idx - 500): idx + len(url) + 500].lower() if idx >= 0 else ""
                if len(urls) == 1 or any(word in context for word in ("live", "livestream", "linear", "streaming")):
                    accepted.append(url)

            candidate_mode = str(item.get("candidate_mode") or "emit").lower()
            emitted = accepted[: int(item.get("max_candidates", 4))]
            if candidate_mode == "research-only":
                emitted = []

            for url in emitted:
                discovered.append({
                    "name_raw": item.get("name") or tvg_id,
                    "name": item.get("name") or tvg_id,
                    "quality": item.get("quality"),
                    "tvg_id": tvg_id,
                    "logo": item.get("logo", ""),
                    "group": item.get("group", "Local"),
                    "language": "English",
                    "country": "US",
                    "source": item.get("source") or f"Official page: {urllib.parse.urlsplit(page_url).netloc}",
                    "family": "official-page",
                    "priority": int(item.get("priority", -10)),
                    "philly": bool(item.get("philly", True)),
                    "url": url,
                    "headers": {},
                    "source_type": "official-public-page",
                    "content_kind": item.get("content_kind", "unknown"),
                    "content_note": item.get("content_note", ""),
                })
            row["candidates"] = accepted[: int(item.get("max_candidates", 4))]
            row["candidates_emitted"] = len(emitted)
            if candidate_mode == "research-only":
                row["status"] = (
                    "research-only-public-hls-found"
                    if row["candidates"]
                    else "research-only-no-public-hls"
                )
            else:
                row["status"] = "found" if row["candidates"] else "no-public-hls-in-page-source"
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    return discovered, rows


def discover_iptv_org_passed_issues(
    config: dict[str, Any],
    target_ids: list[str] | None = None,
    scope: str = "shared",
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Inspect fresh IPTV-org add/remove issues for remaining exact IDs.

    Additions that passed IPTV-org's check can become candidates. Removal issues
    are diagnostics only and never reintroduce their broken URL. Every addition
    still goes through our own double HLS/media-segment probe.
    """
    settings = (config.get("discovery") or {}).get("iptv_org_issues") or {}
    if not settings.get("enabled", True):
        return [], []

    targets = settings.get("targets") or []
    if target_ids is not None:
        wanted = {canonical_tvg_id(item) for item in target_ids}
        targets = [
            target
            for target in targets
            if canonical_tvg_id(target.get("tvg_id", "")) in wanted
        ]
    if not targets:
        return [], []

    params = urllib.parse.urlencode({
        "state": "all",
        "sort": "updated",
        "direction": "desc",
        "per_page": int(settings.get("per_page", 100)),
    })
    api_url = "https://api.github.com/repos/iptv-org/iptv/issues?" + params
    rows: list[dict[str, Any]] = []
    discovered: list[dict[str, Any]] = []

    try:
        payload_text, _ = _fetch_public_text(api_url, timeout=int(settings.get("timeout_seconds", 20)))
        payload = json.loads(payload_text)
        issues = payload if isinstance(payload, list) else []
    except Exception as exc:
        return [], [{
            "kind": "iptv-org-issues",
            "scope": scope,
            "status": "failed",
            "error": str(exc)[:240],
            "api_url": api_url,
        }]

    for issue in issues:
        labels = {
            str(label.get("name") or "")
            for label in (issue.get("labels") or [])
            if isinstance(label, dict)
        }
        change_kind = "add" if "streams:add" in labels else "remove" if "streams:remove" in labels else ""
        if not change_kind or "check:passed" not in labels:
            continue
        if any(label.startswith("rejected:") for label in labels):
            continue

        title = issue.get("title") or ""
        body = issue.get("body") or ""
        combined = f"{title}\n{body}"
        combined_norm = norm(combined)

        matched_target: dict[str, Any] | None = None
        for target in targets:
            terms = [target.get("tvg_id", "")] + list(target.get("aliases") or [])
            if any(term and (norm_id(term) in norm_id(combined) or norm(term) in combined_norm) for term in terms):
                matched_target = target
                break
        if not matched_target:
            continue

        issue_number = issue.get("number")
        issue_url = issue.get("html_url")
        if change_kind == "remove":
            rows.append({
                "kind": "iptv-org-issue",
                "scope": scope,
                "status": "removal-signal",
                "change": "remove",
                "issue": issue_number,
                "issue_url": issue_url,
                "tvg_id": matched_target["tvg_id"],
                "name": matched_target.get("name"),
            })
            continue

        urls = _extract_public_hls_urls(combined)
        urls = [u for u in urls if ".m3u8" in u.lower()]
        if not urls:
            continue

        for url in urls[: int(settings.get("max_urls_per_issue", 3))]:
            discovered.append({
                "name_raw": matched_target.get("name") or matched_target["tvg_id"],
                "name": matched_target.get("name") or matched_target["tvg_id"],
                "quality": None,
                "tvg_id": matched_target["tvg_id"],
                "logo": "",
                "group": matched_target.get("group", ""),
                "language": "English",
                "country": "US",
                "source": f"IPTV-org check:passed issue #{issue_number}",
                "family": "iptv-org-issue",
                "priority": int(settings.get("priority", 6)),
                "philly": bool(matched_target.get("philly", False)),
                "url": url,
                "headers": {},
                "source_type": "public-third-party",
                "content_kind": matched_target.get("content_kind", "unknown"),
                "discovery_issue": issue_url,
            })
        rows.append({
            "kind": "iptv-org-issue",
            "scope": scope,
            "status": "candidate-found",
            "change": "add",
            "issue": issue_number,
            "issue_url": issue_url,
            "tvg_id": matched_target["tvg_id"],
            "name": matched_target.get("name"),
            "urls": urls[: int(settings.get("max_urls_per_issue", 3))],
        })

    return discovered, rows


def discover_seed_candidates(
    config: dict[str, Any],
    target_ids: list[str] | None = None,
    scope: str = "shared",
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Load public research leads from config.

    These are not trusted merely because they are listed here. They are only
    candidates and must pass the same two-probe playlist/media-segment health
    test as every other source before they can appear in Stremio.
    """
    items = (config.get("discovery") or {}).get("seed_candidates") or []
    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []

    wanted = {canonical_tvg_id(item) for item in target_ids} if target_ids is not None else None
    for item in items:
        url = (item.get("url") or "").strip()
        tvg_id = (item.get("tvg_id") or "").strip()
        if wanted is not None and canonical_tvg_id(tvg_id) not in wanted:
            continue
        if not url or not tvg_id or not re.match(r"^https?://", url, re.I):
            continue
        if ".m3u8" not in url.lower():
            continue

        entry = {
            "name_raw": item.get("name") or tvg_id,
            "name": item.get("name") or tvg_id,
            "quality": item.get("quality"),
            "tvg_id": tvg_id,
            "logo": item.get("logo", ""),
            "group": item.get("group", ""),
            "language": "English",
            "country": "US",
            "source": item.get("source") or "Research lead",
            "family": "research-seed",
            "priority": int(item.get("priority", 8)),
            "philly": bool(item.get("philly", False)),
            "url": url,
            "headers": dict(item.get("headers") or {}),
            "source_type": item.get("source_type", "public-third-party"),
            "content_kind": item.get("content_kind", "unknown"),
            "content_note": item.get("content_note", ""),
            "research_url": item.get("research_url"),
        }
        discovered.append(entry)
        rows.append({
            "kind": "seed-candidate",
            "scope": scope,
            "status": "queued-for-health-check",
            "name": entry["name"],
            "tvg_id": tvg_id,
            "source": entry["source"],
            "source_type": entry["source_type"],
            "content_kind": entry["content_kind"],
            "url": url,
            "research_url": entry.get("research_url"),
        })

    return discovered, rows


def discover_dynamic_streams(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    if not (config.get("discovery") or {}).get("enabled", True):
        return [], []

    targeted_entries, targeted_rows, national_target_ids = discover_targeted_source_families(
        config,
        existing_entries,
    )
    local_entries, local_rows, philly_target_ids = discover_philly_local_sources(
        config,
        existing_entries,
    )
    national_issue_entries, national_issue_rows = discover_iptv_org_passed_issues(
        config, national_target_ids, scope="national",
    )
    local_issue_entries, local_issue_rows = discover_iptv_org_passed_issues(
        config, philly_target_ids, scope="philly-local",
    )
    national_seed_entries, national_seed_rows = discover_seed_candidates(
        config, national_target_ids, scope="national",
    )
    local_seed_entries, local_seed_rows = discover_seed_candidates(
        config, philly_target_ids, scope="philly-local",
    )

    combined: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    all_entries = (
        targeted_entries + local_entries + national_issue_entries + local_issue_entries
        + national_seed_entries + local_seed_entries
    )
    for entry in all_entries:
        marker = (canonical_tvg_id(entry.get("tvg_id", "")), entry.get("url", ""))
        if marker in seen:
            continue
        seen.add(marker)
        combined.append(entry)

    log(
        f"Dynamic discovery: {len(targeted_entries)} targeted-family + "
        f"{len(local_entries)} Philadelphia-local + "
        f"{len(national_issue_entries) + len(local_issue_entries)} IPTV-org issue + "
        f"{len(national_seed_entries) + len(local_seed_entries)} seed candidates"
    )
    return combined, (
        targeted_rows + local_rows + national_issue_rows + local_issue_rows
        + national_seed_rows + local_seed_rows
    )


def _parse_hls_variants(text: str, base_url: str) -> list[dict[str, Any]]:
    """Parse master-playlist variants without downloading every rendition."""
    variants: list[dict[str, Any]] = []
    lines = [line.strip() for line in text.splitlines()]
    for index, line in enumerate(lines):
        if not line.upper().startswith("#EXT-X-STREAM-INF:"):
            continue
        uri = ""
        for following in lines[index + 1:]:
            if not following:
                continue
            if following.startswith("#"):
                continue
            uri = urllib.parse.urljoin(base_url, following)
            break
        if not uri:
            continue
        resolution = re.search(r"(?:^|,)\s*RESOLUTION=(\d+)[xX](\d+)", line, re.I)
        bandwidth = re.search(r"(?:^|,)\s*(?:AVERAGE-)?BANDWIDTH=(\d+)", line, re.I)
        width = int(resolution.group(1)) if resolution else 0
        height = int(resolution.group(2)) if resolution else 0
        variants.append({
            "url": uri,
            "width": width,
            "height": height,
            "bandwidth": int(bandwidth.group(1)) if bandwidth else 0,
        })
    return variants


def _probe_stream(stream: dict[str, Any], timeout: int, verify_segment: bool) -> dict[str, Any]:
    url = stream.get("url") or ""
    if not re.match(r"^https?://", url, re.I):
        return {"status": "not-tested", "detail": "Non-HTTP stream"}

    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "*/*",
        "Accept-Encoding": "identity",
    }
    headers.update(stream.get("headers") or {})
    started = time.monotonic()

    def fetch(target: str, byte_limit: int = 262144) -> tuple[int, str, str, bytes]:
        req = urllib.request.Request(target, headers=headers)
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            status = int(getattr(resp, "status", 200) or 200)
            final_url = resp.geturl()
            content_type = (resp.headers.get("Content-Type") or "").lower()
            body = resp.read(byte_limit)
            return status, final_url, content_type, body

    try:
        status, final_url, content_type, body = fetch(url)
        elapsed_ms = round((time.monotonic() - started) * 1000)
        text = body.decode("utf-8", errors="ignore")
        is_hls = (
            "#EXTM3U" in text
            or ".m3u8" in urllib.parse.urlsplit(final_url).path.lower()
            or "mpegurl" in content_type
        )

        if is_hls:
            if "#EXTM3U" not in text:
                return {
                    "status": "failed",
                    "detail": "HLS URL did not return a valid playlist",
                    "http_status": status,
                    "latency_ms": elapsed_ms,
                }

            media_text = text
            media_url = final_url
            measurement: dict[str, Any] = {}
            variants = _parse_hls_variants(text, final_url)
            if variants:
                measured = max(variants, key=lambda item: (item["height"], item["width"], item["bandwidth"]))
                known_resolutions = sorted(
                    {
                        (item["width"], item["height"])
                        for item in variants
                        if item["width"] > 0 and item["height"] > 0
                    },
                    key=lambda item: (item[1], item[0]),
                    reverse=True,
                )
                if measured["height"] > 0:
                    measurement = {
                        "measured_width": measured["width"],
                        "measured_height": measured["height"],
                        "measured_quality": quality_label_for_height(measured["height"]),
                        "quality_source": "hls-master",
                        "available_resolutions": [f"{width}x{height}" for width, height in known_resolutions],
                    }
                candidate = measured["url"]
                child_status, child_url, _child_type, child_body = fetch(candidate)
                media_text = child_body.decode("utf-8", errors="ignore")
                media_url = child_url
                if "#EXTM3U" not in media_text:
                    return {
                        "status": "failed",
                        "detail": "Master playlist child was invalid",
                        "http_status": child_status,
                        "latency_ms": round((time.monotonic() - started) * 1000),
                    }

            if verify_segment:
                segments = [
                    line.strip()
                    for line in media_text.splitlines()
                    if line.strip() and not line.startswith("#")
                ]
                if segments:
                    segment_url = urllib.parse.urljoin(media_url, segments[0])
                    seg_headers = dict(headers)
                    seg_headers["Range"] = "bytes=0-1023"
                    req = urllib.request.Request(segment_url, headers=seg_headers)
                    with urllib.request.urlopen(req, timeout=timeout) as resp:
                        resp.read(1)

            return {
                "status": "ok",
                "detail": "Valid HLS playlist" + (" + segment" if verify_segment else ""),
                "http_status": status,
                "latency_ms": round((time.monotonic() - started) * 1000),
                **measurement,
            }

        if "text/html" in content_type:
            return {
                "status": "failed",
                "detail": "Returned HTML instead of media",
                "http_status": status,
                "latency_ms": elapsed_ms,
            }

        return {
            "status": "ok",
            "detail": "Reachable direct media URL",
            "http_status": status,
            "latency_ms": elapsed_ms,
        }

    except urllib.error.HTTPError as exc:
        return {
            "status": "failed",
            "detail": f"HTTP {exc.code}",
            "http_status": exc.code,
            "latency_ms": round((time.monotonic() - started) * 1000),
        }
    except Exception as exc:
        return {
            "status": "failed",
            "detail": str(exc)[:180],
            "latency_ms": round((time.monotonic() - started) * 1000),
        }


def _stream_stability_key(entry: dict[str, Any]) -> str:
    """Stable identity for one playback URL + request-header combination."""
    payload = {
        "url": entry.get("url") or "",
        "headers": sorted((entry.get("headers") or {}).items()),
    }
    encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
    return hashlib.sha1(encoded.encode("utf-8")).hexdigest()


def _classify_stream_stability(history: list[bool], history_size: int) -> dict[str, Any]:
    history = [bool(x) for x in history][-history_size:]
    passes = sum(1 for x in history if x)
    samples = len(history)
    if samples >= history_size and passes == history_size:
        classification = "Stable"
        rank = 0
    elif passes >= 3:
        classification = "Backup"
        rank = 1
    elif passes >= 1:
        classification = "Quarantine"
        rank = 2
    else:
        classification = "Dead"
        rank = 3
    return {
        "classification": classification,
        "rank": rank,
        "passes": passes,
        "samples": samples,
        "history_size": history_size,
        "history": history,
    }


def _load_previous_stream_stability(config: dict[str, Any]) -> dict[str, Any]:
    settings = config.get("stream_stability") or {}
    if not settings.get("enabled", True):
        return {}

    explicit = (settings.get("previous_state_url") or "").strip()
    public_base = derive_public_base()
    if explicit:
        state_url = explicit
    elif public_base:
        state_url = f"{public_base}/diagnostics/stream-stability.json"
    else:
        return {}

    cache_bust = os.getenv("GITHUB_RUN_ID") or str(int(time.time()))
    separator = "&" if "?" in state_url else "?"
    request_url = f"{state_url}{separator}build={urllib.parse.quote(cache_bust)}"
    req = urllib.request.Request(
        request_url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "application/json",
            "Accept-Encoding": "identity",
            "Cache-Control": "no-cache",
        },
    )
    try:
        with urllib.request.urlopen(
            req,
            timeout=int(settings.get("load_timeout_seconds", 8)),
        ) as resp:
            payload = json.loads(resp.read(8_000_000).decode("utf-8"))
        if isinstance(payload, dict) and isinstance(payload.get("streams"), dict):
            log(
                "Stream stability: loaded "
                f"{len(payload['streams']):,} prior URL histories"
            )
            return payload
    except urllib.error.HTTPError as exc:
        if exc.code != 404:
            log(f"WARNING: could not load prior stream stability state: HTTP {exc.code}")
    except Exception as exc:
        log(f"WARNING: could not load prior stream stability state: {str(exc)[:160]}")
    return {}


def apply_stream_stability(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
    previous: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """Persist and score URL health across builds.

    Current playback health remains mandatory. Stability only changes ordering
    among candidates that passed this build, so an old successful URL is never
    exposed while it is currently failing.
    """
    settings = config.get("stream_stability") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    history_size = max(1, int(settings.get("history_size", 5)))
    retention_builds = max(history_size, int(settings.get("retention_builds", 20)))
    cooldown_hours = [
        max(1, int(value))
        for value in settings.get("dead_source_cooldown_hours", [1, 6, 24])
    ] or [1, 6, 24]
    built_at = iso_z(dt.datetime.now(UTC))

    if not enabled:
        for entry in entries:
            entry["stability"] = {
                "classification": "Unknown",
                "rank": 2,
                "passes": 0,
                "samples": 0,
                "history_size": history_size,
                "history": [],
            }
        return {
            "version": 1,
            "enabled": False,
            "history_size": history_size,
            "updated_at": built_at,
            "streams": {},
            "current_counts": {},
        }

    previous = previous if previous is not None else _load_previous_stream_stability(config)
    previous_streams = previous.get("streams") or {}
    state_streams: dict[str, dict[str, Any]] = {}

    # Keep recently unseen candidates so a temporary source-playlist outage does
    # not erase four builds of useful history.
    for key, old in previous_streams.items():
        if not isinstance(old, dict):
            continue
        missed = int(old.get("missed_builds", 0)) + 1
        if missed > retention_builds:
            continue
        copied = dict(old)
        copied["missed_builds"] = missed
        state_streams[key] = copied

    current_by_key: dict[str, dict[str, Any]] = {}
    linked_by_key: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for entry in entries:
        key = _stream_stability_key(entry)
        current_by_key.setdefault(key, entry)
        linked_by_key[key].append(entry)

    current_counts: dict[str, int] = defaultdict(int)
    for key, entry in current_by_key.items():
        old = previous_streams.get(key) if isinstance(previous_streams.get(key), dict) else {}
        history = [bool(x) for x in (old.get("history") or [])][-history_size:]
        health = entry.get("health") or {}
        health_status = health.get("status")
        skipped_cooldown = health_status == "cooldown"
        passed = health_status == "ok"
        if not skipped_cooldown:
            history = (history + [passed])[-history_size:]
        score = _classify_stream_stability(history, history_size)
        score["last_result"] = "cooldown" if skipped_cooldown else "pass" if passed else "fail"
        score["last_seen"] = built_at

        old_failures = int(old.get("consecutive_failures", 0) or 0)
        if not old_failures and old.get("last_result") == "fail":
            for result in reversed(old.get("history") or []):
                if result:
                    break
                old_failures += 1
        if skipped_cooldown:
            consecutive_failures = old_failures
            next_probe_at = health.get("cooldown_until") or old.get("next_probe_at")
            last_probed_at = old.get("last_probed_at") or old.get("last_seen")
            last_failure_detail = old.get("last_failure_detail")
            last_failure_http_status = old.get("last_failure_http_status")
        elif passed:
            consecutive_failures = 0
            next_probe_at = None
            last_probed_at = built_at
            last_failure_detail = None
            last_failure_http_status = None
        else:
            consecutive_failures = old_failures + 1
            delay = cooldown_hours[min(consecutive_failures - 1, len(cooldown_hours) - 1)]
            next_probe_at = iso_z(dt.datetime.now(UTC) + dt.timedelta(hours=delay))
            last_probed_at = built_at
            last_failure_detail = health.get("detail")
            last_failure_http_status = health.get("http_status")

        record = {
            "url": entry.get("url"),
            "headers": entry.get("headers") or {},
            "tvg_id": entry.get("tvg_id"),
            "name": entry.get("name"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "quality": quality_label(entry),
            "declared_quality": entry.get("quality") or "Unknown",
            "measured_quality": entry.get("measured_quality"),
            **score,
            "consecutive_failures": consecutive_failures,
            "next_probe_at": next_probe_at,
            "last_probed_at": last_probed_at,
            "last_failure_detail": last_failure_detail,
            "last_failure_http_status": last_failure_http_status,
            "missed_builds": 0,
        }
        state_streams[key] = record
        current_counts[score["classification"]] += 1

        entry_score = {
            k: score[k]
            for k in (
                "classification",
                "rank",
                "passes",
                "samples",
                "history_size",
                "history",
                "last_result",
                "last_seen",
            )
        }
        entry_score["consecutive_failures"] = consecutive_failures
        entry_score["next_probe_at"] = next_probe_at
        for linked in linked_by_key[key]:
            linked["stability"] = dict(entry_score)

    log(
        "Cross-build stream stability: "
        + ", ".join(
            f"{name}={current_counts.get(name, 0)}"
            for name in ("Stable", "Backup", "Quarantine", "Dead")
        )
    )

    return {
        "version": 1,
        "enabled": True,
        "history_size": history_size,
        "dead_source_cooldown_hours": cooldown_hours,
        "updated_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "streams": state_streams,
        "current_counts": dict(current_counts),
    }


def _channel_state_key(row: dict[str, Any]) -> str:
    group = str(row.get("group") or "Unknown").strip()
    pattern = str(row.get("pattern") or "").strip()
    return f"{group}:{canonical_tvg_id(pattern)}"


_PREVIOUS_CHANNEL_STATE_CACHE: dict[str, Any] | None = None


def _load_previous_channel_state(config: dict[str, Any]) -> dict[str, Any]:
    """Load the last deployed exact-channel state for build-to-build comparison."""
    global _PREVIOUS_CHANNEL_STATE_CACHE
    if _PREVIOUS_CHANNEL_STATE_CACHE is not None:
        return _PREVIOUS_CHANNEL_STATE_CACHE

    settings = config.get("channel_diagnostics") or {}
    if not settings.get("enabled", True):
        _PREVIOUS_CHANNEL_STATE_CACHE = {}
        return _PREVIOUS_CHANNEL_STATE_CACHE

    explicit = (settings.get("previous_state_url") or "").strip()
    public_base = derive_public_base()
    if explicit:
        state_url = explicit
    elif public_base:
        state_url = f"{public_base}/diagnostics/channel-state.json"
    else:
        _PREVIOUS_CHANNEL_STATE_CACHE = {}
        return _PREVIOUS_CHANNEL_STATE_CACHE

    cache_bust = os.getenv("GITHUB_RUN_ID") or str(int(time.time()))
    separator = "&" if "?" in state_url else "?"
    request_url = f"{state_url}{separator}build={urllib.parse.quote(cache_bust)}"
    req = urllib.request.Request(
        request_url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "application/json",
            "Accept-Encoding": "identity",
            "Cache-Control": "no-cache",
        },
    )
    try:
        with urllib.request.urlopen(
            req,
            timeout=int(settings.get("load_timeout_seconds", 8)),
        ) as resp:
            payload = json.loads(resp.read(4_000_000).decode("utf-8"))
        if isinstance(payload, dict) and isinstance(payload.get("channels"), dict):
            log(
                "Channel diagnostics: loaded "
                f"{len(payload['channels']):,} prior exact-channel states"
            )
            _PREVIOUS_CHANNEL_STATE_CACHE = payload
            return _PREVIOUS_CHANNEL_STATE_CACHE
    except urllib.error.HTTPError as exc:
        if exc.code != 404:
            log(f"WARNING: could not load prior channel state: HTTP {exc.code}")
    except Exception as exc:
        log(f"WARNING: could not load prior channel state: {str(exc)[:160]}")
    _PREVIOUS_CHANNEL_STATE_CACHE = {}
    return _PREVIOUS_CHANNEL_STATE_CACHE


def build_channel_diagnostics(
    passing_allowlist_coverage: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> tuple[dict[str, Any], dict[str, Any]]:
    """Track exact allow-listed channels across builds.

    A channel is considered present only when at least one candidate for that exact
    allow-list ID passed the current build's health gate. This intentionally tracks
    zero-candidate IDs too, instead of looking only at channels that happened to
    make it into the merged catalog.
    """
    settings = config.get("channel_diagnostics") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    built_at = iso_z(dt.datetime.now(UTC))
    previous = _load_previous_channel_state(config) if enabled else {}
    previous_channels = previous.get("channels") if isinstance(previous.get("channels"), dict) else {}
    has_prior_state = bool(previous_channels)

    channels: dict[str, dict[str, Any]] = {}
    recovered: list[dict[str, Any]] = []
    lost: list[dict[str, Any]] = []
    still_missing: list[dict[str, Any]] = []
    stable: list[dict[str, Any]] = []

    for row in passing_allowlist_coverage:
        key = _channel_state_key(row)
        old = previous_channels.get(key) if isinstance(previous_channels.get(key), dict) else {}
        was_present = bool(old.get("present")) if old else False
        present = bool(row.get("found"))

        if present:
            stable_builds = int(old.get("stable_builds", 0)) + 1 if was_present else 1
            missing_builds = 0
        else:
            stable_builds = 0
            missing_builds = int(old.get("missing_builds", 0)) + 1 if old and not was_present else 1

        record = {
            "group": row.get("group"),
            "pattern": row.get("pattern"),
            "present": present,
            "candidate_count": int(row.get("candidate_count", 0)),
            "sources": row.get("sources") or [],
            "qualities": row.get("qualities") or [],
            "stable_builds": stable_builds,
            "missing_builds": missing_builds,
            "last_changed_at": (
                old.get("last_changed_at")
                if old and was_present == present
                else built_at
            ),
            "last_seen_at": built_at if present else old.get("last_seen_at"),
            "last_missing_at": built_at if not present else old.get("last_missing_at"),
        }
        channels[key] = record

        summary_row = {
            "group": record["group"],
            "channel_id": record["pattern"],
            "candidate_count": record["candidate_count"],
            "sources": record["sources"],
            "qualities": record["qualities"],
        }

        if present:
            stable.append({**summary_row, "stable_for_builds": stable_builds})
            if has_prior_state and old and not was_present:
                recovered.append({
                    **summary_row,
                    "stable_for_builds": stable_builds,
                    "previously_missing_for_builds": int(old.get("missing_builds", 0)),
                })
        else:
            if has_prior_state and old and was_present:
                lost.append({
                    **summary_row,
                    "missing_for_builds": missing_builds,
                    "previously_stable_for_builds": int(old.get("stable_builds", 0)),
                })
            else:
                still_missing.append({**summary_row, "missing_for_builds": missing_builds})

    sort_key = lambda item: (str(item.get("group") or ""), str(item.get("channel_id") or ""))
    recovered.sort(key=sort_key)
    lost.sort(key=sort_key)
    still_missing.sort(key=sort_key)
    stable.sort(key=sort_key)

    state = {
        "version": 1,
        "enabled": enabled,
        "updated_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "has_prior_state": has_prior_state,
        "channels": channels,
    }
    report = {
        "version": 1,
        "built_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "baseline_initialized": not has_prior_state,
        "expected_channels": len(channels),
        "present_channels": sum(1 for item in channels.values() if item.get("present")),
        "missing_channels": sum(1 for item in channels.values() if not item.get("present")),
        "recovered_this_build": recovered,
        "lost_this_build": lost,
        "still_missing": still_missing,
        "stable": stable,
    }

    log(
        "Channel changes: "
        f"recovered={len(recovered)}, lost={len(lost)}, "
        f"still_missing={len(still_missing)}, stable={len(stable)}"
        + (" (baseline initialized)" if not has_prior_state else "")
    )
    return state, report


def _dead_source_cooldown(
    entry: dict[str, Any],
    previous: dict[str, Any],
    config: dict[str, Any],
    now: dt.datetime,
) -> dict[str, Any] | None:
    """Return a skip result while a previously dead URL is backing off."""
    settings = config.get("stream_stability") or {}
    if not settings.get("enabled", True):
        return None
    old = (previous.get("streams") or {}).get(_stream_stability_key(entry))
    if not isinstance(old, dict) or old.get("classification") != "Dead":
        return None

    failures = int(old.get("consecutive_failures", 0) or 0)
    if failures <= 0:
        for result in reversed(old.get("history") or []):
            if result:
                break
            failures += 1
    failures = max(1, failures)
    hours = [
        max(1, int(value))
        for value in settings.get("dead_source_cooldown_hours", [1, 6, 24])
    ] or [1, 6, 24]

    next_probe_at = old.get("next_probe_at")
    try:
        if next_probe_at:
            retry_at = dt.datetime.fromisoformat(str(next_probe_at).replace("Z", "+00:00"))
        else:
            anchor_text = old.get("last_probed_at") or old.get("last_seen")
            if not anchor_text:
                return None
            anchor = dt.datetime.fromisoformat(str(anchor_text).replace("Z", "+00:00"))
            retry_at = anchor + dt.timedelta(hours=hours[min(failures - 1, len(hours) - 1)])
    except (TypeError, ValueError):
        return None
    if retry_at.tzinfo is None:
        retry_at = retry_at.replace(tzinfo=UTC)
    if retry_at <= now:
        return None
    prior_detail = str(old.get("last_failure_detail") or "").strip()
    detail = "Dead source probe deferred by backoff"
    if prior_detail:
        detail += f"; last failure: {prior_detail}"
    return {
        "status": "cooldown",
        "detail": detail,
        "cooldown_until": iso_z(retry_at),
        "consecutive_failures": failures,
        "last_http_status": old.get("last_failure_http_status"),
    }


def probe_candidate_entries(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> tuple[list[dict[str, Any]], list[dict[str, Any]], dict[str, Any]]:
    """Probe the full curated candidate pool before selecting channel streams.

    Production can require two successful probes separated by one global wait.
    This catches transient/cached URLs without multiplying the wait by the
    number of streams.
    """
    settings = config.get("stream_health") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    require_passed_only = bool(settings.get("require_passed_only", True))
    timeout = int(settings.get("timeout_seconds", 6))
    max_workers = int(settings.get("max_workers", 16))
    verify_segment = bool(settings.get("verify_segment_for_all_candidates", True))
    double_probe = bool(settings.get("double_probe", True))
    second_probe_delay = int(settings.get("second_probe_delay_seconds", 60))

    rows: list[dict[str, Any]] = []
    previous_stability: dict[str, Any] = {}

    if not enabled:
        for entry in entries:
            entry["health"] = {"status": "not-tested", "detail": "Health checks disabled"}
            rows.append({
                "name": entry.get("name"),
                "tvg_id": entry.get("tvg_id"),
                "source": entry.get("source"),
                "family": entry.get("family"),
                "quality": entry.get("quality") or "Unknown",
                "url": entry.get("url"),
                "health": entry.get("health"),
            })
        stability_state = apply_stream_stability(entries, config, enabled=False)
        for row, entry in zip(rows, entries):
            row["stability"] = entry.get("stability")
        return entries, rows, stability_state

    previous_stability = _load_previous_stream_stability(config)
    jobs: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    refs: dict[tuple[str, tuple[tuple[str, str], ...]], list[dict[str, Any]]] = defaultdict(list)
    results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    now = dt.datetime.now(UTC)

    for entry in entries:
        headers_key = tuple(sorted((entry.get("headers") or {}).items()))
        key = (entry.get("url") or "", headers_key)
        refs[key].append(entry)
        cooldown = _dead_source_cooldown(entry, previous_stability, config, now)
        if cooldown:
            results.setdefault(key, cooldown)
        else:
            jobs.setdefault(key, entry)

    def run_probe_batch(batch: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]) -> dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]:
        batch_results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
        with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
            future_map = {
                executor.submit(_probe_stream, entry, timeout, verify_segment): key
                for key, entry in batch.items()
            }
            for future in concurrent.futures.as_completed(future_map):
                key = future_map[future]
                try:
                    batch_results[key] = future.result()
                except Exception as exc:
                    batch_results[key] = {"status": "failed", "detail": str(exc)[:180]}
        return batch_results

    results.update(run_probe_batch(jobs))

    first_pass = {
        key: jobs[key]
        for key, health in results.items()
        if health.get("status") == "ok"
    }

    if double_probe and first_pass:
        log(
            f"Health stability check: {len(first_pass):,} candidates passed probe 1; "
            f"waiting {second_probe_delay}s before probe 2"
        )
        time.sleep(max(0, second_probe_delay))
        second_results = run_probe_batch(first_pass)

        for key, first_health in list(results.items()):
            if first_health.get("status") != "ok":
                continue
            second_health = second_results.get(key, {"status": "failed", "detail": "No second probe result"})
            if second_health.get("status") == "ok":
                results[key] = {
                    **first_health,
                    "status": "ok",
                    "detail": "Passed two HLS/media probes",
                    "probe_1": first_health,
                    "probe_2": second_health,
                    "stable": True,
                }
            else:
                results[key] = {
                    **second_health,
                    "status": "failed",
                    "detail": f"Probe 2 failed after probe 1 passed: {second_health.get('detail', 'unknown')}",
                    "probe_1": first_health,
                    "probe_2": second_health,
                    "stable": False,
                }

    for key, linked_entries in refs.items():
        health = results.get(key, {"status": "failed", "detail": "No probe result"})
        for entry in linked_entries:
            entry["health"] = dict(health)
            for field in (
                "measured_width",
                "measured_height",
                "measured_quality",
                "quality_source",
                "available_resolutions",
            ):
                if health.get(field) is not None:
                    entry[field] = health[field]

    stability_state = apply_stream_stability(
        entries,
        config,
        enabled=True,
        previous=previous_stability,
    )

    for entry in entries:
        rows.append({
            "name": entry.get("name"),
            "tvg_id": entry.get("tvg_id"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "source_type": entry.get("source_type"),
            "quality": quality_label(entry),
            "declared_quality": entry.get("quality") or "Unknown",
            "measured_quality": entry.get("measured_quality"),
            "measured_width": entry.get("measured_width"),
            "measured_height": entry.get("measured_height"),
            "quality_source": entry.get("quality_source"),
            "available_resolutions": entry.get("available_resolutions") or [],
            "url": entry.get("url"),
            "health": entry.get("health"),
            "stability": entry.get("stability"),
            "discovery_issue": entry.get("discovery_issue"),
            "content_note": entry.get("content_note"),
        })

    counts = defaultdict(int)
    for row in rows:
        counts[(row.get("health") or {}).get("status", "not-tested")] += 1
    log(
        "Candidate stream health: "
        + ", ".join(f"{status}={count}" for status, count in sorted(counts.items()))
    )

    if not require_passed_only:
        return entries, rows, stability_state

    healthy = [
        entry
        for entry in entries
        if (entry.get("health") or {}).get("status") == "ok"
    ]
    log(f"Passed-only policy: keeping {len(healthy):,}/{len(entries):,} curated candidates")
    return healthy, rows, stability_state


def label_channel_streams(channels: list[dict[str, Any]]) -> None:
    for channel in channels:
        for index, stream in enumerate(channel.get("streams", [])):
            stream["display_role"] = "Primary" if index == 0 else f"Backup {index}"


def build_source_ordering_diagnostics(
    channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> list[dict[str, Any]]:
    configured = (config.get("stream_selection") or {}).get("channel_preferences") or {}
    configured_ids = {canonical_tvg_id(item) for item in configured}
    rows: list[dict[str, Any]] = []
    for channel in channels:
        tvg_id = str(channel.get("tvg_id") or "")
        rules = channel_preference_rules(config, tvg_id)
        stream_rows = []
        for stream in channel.get("streams", []):
            stability = stream.get("stability") or {}
            stream_rows.append({
                "role": stream.get("display_role"),
                "source": stream.get("source"),
                "family": stream.get("family"),
                "quality": quality_label(stream),
                "declared_quality": stream.get("quality"),
                "measured_quality": stream.get("measured_quality"),
                "measured_resolution": (
                    f"{stream.get('measured_width')}x{stream.get('measured_height')}"
                    if stream.get("measured_width") and stream.get("measured_height")
                    else None
                ),
                "content_kind": stream.get("content_kind", "unknown"),
                "stability": stability.get("classification", "Unknown"),
                "stability_passes": stability.get("passes", 0),
                "stability_samples": stability.get("samples", 0),
                "preference_rank": stream.get("preference_rank"),
                "preference_label": stream.get("preference_label"),
                "url": stream.get("url"),
            })
        rows.append({
            "name": channel.get("name"),
            "tvg_id": tvg_id,
            "preference_mode": "channel-specific" if canonical_tvg_id(tvg_id) in configured_ids else "default",
            "preference_rules": rules,
            "primary_matches_first_preference": bool(
                stream_rows and stream_rows[0].get("preference_rank") == 1
            ),
            "streams": stream_rows,
        })
    return rows


def build_missing_source_diagnostics(
    candidate_coverage: list[dict[str, Any]],
    passing_coverage: list[dict[str, Any]],
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> dict[str, Any]:
    """Turn broad missing-source counts into mutually exclusive work queues."""
    passing_by_id = {canonical_tvg_id(row.get("pattern", "")): row for row in passing_coverage}
    health_by_id: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for row in health_rows:
        health_by_id[canonical_tvg_id(row.get("tvg_id", ""))].append(row)
    channels_by_id = {canonical_tvg_id(channel.get("tvg_id", "")): channel for channel in channels}
    target = config.get("stream_selection") or {}
    target_hd = int(target.get("target_hd", 2))
    target_sd = int(target.get("target_sd", 1))

    rows: list[dict[str, Any]] = []
    bucket_counts: dict[str, int] = defaultdict(int)
    without_passing_counts: dict[str, int] = defaultdict(int)
    for candidate in candidate_coverage:
        channel_id = candidate.get("pattern", "")
        key = canonical_tvg_id(channel_id)
        passing = passing_by_id.get(key) or {}
        channel = channels_by_id.get(key)
        probes = health_by_id.get(key, [])
        statuses = [(row.get("health") or {}).get("status", "not-tested") for row in probes]
        details = [str((row.get("health") or {}).get("detail") or "") for row in probes]
        candidate_count = int(candidate.get("candidate_count", 0) or 0)
        passing_count = int(passing.get("candidate_count", 0) or 0)
        failure_kind: str | None = None

        if candidate_count == 0:
            bucket = "no_candidate"
            action = "Find an exact-ID public source."
        elif passing_count == 0:
            blocked = any(
                re.search(r"(?:HTTP\s+(?:401|403|451|429)|geo|forbidden|access denied)", detail, re.I)
                for detail in details
            )
            if statuses and all(status == "cooldown" for status in statuses):
                bucket = "failed_probes"
                action = "Wait for the dead-source cooldown, then review the next scheduled retry."
                failure_kind = "dead_source_cooldown"
            elif blocked:
                bucket = "runner_or_geo_blocked"
                action = "Compare the URL in Stremio with the GitHub runner result before replacing it."
                failure_kind = "access_or_geo_signal"
            else:
                bucket = "failed_probes"
                action = "Replace or repair candidates that failed playlist/media validation."
                failure_kind = "playback_validation"
        else:
            streams = (channel or {}).get("streams", [])
            measured = [stream for stream in streams if stream.get("measured_height")]
            hd_count = sum(1 for stream in streams if is_hd_stream(stream))
            sd_count = sum(1 for stream in streams if is_sd_stream(stream))
            if streams and not measured:
                bucket = "missing_quality_metadata"
                action = "Add or expose an HLS master playlist with RESOLUTION metadata."
            elif hd_count < target_hd or sd_count < target_sd:
                bucket = "insufficient_quality_mix"
                action = "Find the missing measured HD/SD rendition or backup family."
            else:
                bucket = "complete"
                action = "No source work required."
            failure_kind = None

        bucket_counts[bucket] += 1
        if passing_count == 0:
            without_passing_counts[bucket] += 1
        row = {
            "group": candidate.get("group"),
            "tvg_id": channel_id,
            "bucket": bucket,
            "candidate_count": candidate_count,
            "passing_candidate_count": passing_count,
            "probe_statuses": dict(Counter(statuses)),
            "sources": candidate.get("sources") or [],
            "action": action,
        }
        if failure_kind:
            row["failure_kind"] = failure_kind
        rows.append(row)

    rows.sort(key=lambda row: (row["bucket"], row.get("group", ""), row["tvg_id"]))
    return {
        "summary": {
            "expected_channels": len(rows),
            "channels_without_passing_stream": sum(1 for row in rows if row["passing_candidate_count"] == 0),
            "bucket_counts": dict(sorted(bucket_counts.items())),
            "without_passing_bucket_counts": dict(sorted(without_passing_counts.items())),
        },
        "channels": rows,
    }


def build_philly_recovery_diagnostics(
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> dict[str, Any]:
    settings = ((config.get("discovery") or {}).get("philly_local_sources") or {})
    target_ids = settings.get("target_ids") or []
    client_results = config.get("stremio_client_results") or {}
    health_by_id: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for row in health_rows:
        health_by_id[canonical_tvg_id(row.get("tvg_id", ""))].append(row)
    channels_by_id = {canonical_tvg_id(channel.get("tvg_id", "")): channel for channel in channels}

    rows: list[dict[str, Any]] = []
    for tvg_id in target_ids:
        key = canonical_tvg_id(tvg_id)
        probes = health_by_id.get(key, [])
        statuses = [(row.get("health") or {}).get("status", "not-tested") for row in probes]
        channel = channels_by_id.get(key)
        selected = (channel or {}).get("streams", [])
        client_ready = [
            stream for stream in selected
            if str(stream.get("url") or "").lower().startswith("https://") and not stream.get("headers")
        ]
        runner_status = (
            "passing" if "ok" in statuses
            else "cooldown" if statuses and all(status == "cooldown" for status in statuses)
            else "failing" if probes
            else "no-candidate"
        )
        configured_client = client_results.get(tvg_id) or {}
        if isinstance(configured_client, str):
            configured_client = {"status": configured_client}
        client_status = configured_client.get("status", "not-tested")
        if runner_status == "passing" and client_status == "not-tested":
            next_action = "Test the Primary URL in Stremio and record pass/fail in stremio_client_results."
        elif runner_status == "passing" and client_status == "failed":
            next_action = "Investigate a Stremio-versus-runner playback mismatch."
        elif runner_status == "no-candidate":
            next_action = "Continue exact-station source discovery; do not substitute a news/FAST lookalike."
        elif runner_status == "cooldown":
            next_action = "Wait for the scheduled retry or add a new exact-station candidate."
        else:
            next_action = "Review failed probe details and compare any plausible URL in Stremio."
        rows.append({
            "tvg_id": tvg_id,
            "name": (channel or {}).get("name") or tvg_id,
            "candidate_count": len(probes),
            "probe_statuses": dict(Counter(statuses)),
            "runner_status": runner_status,
            "selected_streams": len(selected),
            "stremio_ready_selected_streams": len(client_ready),
            "stremio_client_status": client_status,
            "stremio_client_tested_at": configured_client.get("tested_at"),
            "stremio_client_note": configured_client.get("note"),
            "next_action": next_action,
        })
    return {
        "summary": {
            "targets": len(rows),
            "runner_passing": sum(1 for row in rows if row["runner_status"] == "passing"),
            "client_tested": sum(1 for row in rows if row["stremio_client_status"] != "not-tested"),
            "client_runner_mismatches": sum(
                1 for row in rows
                if row["runner_status"] == "passing" and row["stremio_client_status"] == "failed"
            ),
        },
        "channels": rows,
    }


def build_quality_measurements(
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
) -> dict[str, Any]:
    selected_streams = [
        stream
        for channel in channels
        for stream in channel.get("streams", [])
    ]
    selected_urls = {
        stream.get("url")
        for stream in selected_streams
        if stream.get("url")
    }
    rows: list[dict[str, Any]] = []
    for stream in health_rows:
        health = stream.get("health") or {}
        if health.get("status") != "ok":
            continue
        rows.append({
            "name": stream.get("name"),
            "tvg_id": stream.get("tvg_id"),
            "source": stream.get("source"),
            "family": stream.get("family"),
            "selected": stream.get("url") in selected_urls,
            "declared_quality": stream.get("declared_quality") or "Unknown",
            "measured_quality": stream.get("measured_quality"),
            "measured_width": stream.get("measured_width"),
            "measured_height": stream.get("measured_height"),
            "available_resolutions": stream.get("available_resolutions") or [],
            "quality_source": stream.get("quality_source"),
            "url": stream.get("url"),
        })
    return {
        "summary": {
            "passing_candidates": len(rows),
            "measured_passing_candidates": sum(1 for row in rows if row["measured_height"]),
            "selected_streams": len(selected_streams),
            "measured_selected_streams": sum(1 for stream in selected_streams if stream.get("measured_height")),
            "selected_hd": sum(1 for stream in selected_streams if is_hd_stream(stream)),
            "selected_sd": sum(1 for stream in selected_streams if is_sd_stream(stream)),
            "selected_quality_unknown": sum(
                1 for stream in selected_streams
                if not stream.get("measured_height") and not stream.get("quality")
            ),
        },
        "streams": rows,
    }


def channel_meta(channel: dict[str, Any], programmes: list[dict[str, Any]] | None = None) -> dict[str, Any]:
    programmes = sorted(programmes or [], key=lambda p: p.get("startTime", ""))
    now = dt.datetime.now(UTC)

    current: dict[str, Any] | None = None
    upcoming: dict[str, Any] | None = None
    for programme in programmes:
        try:
            p_start = dt.datetime.fromisoformat(programme["startTime"].replace("Z", "+00:00"))
            p_end = dt.datetime.fromisoformat(programme["endTime"].replace("Z", "+00:00"))
        except (KeyError, ValueError):
            continue
        if p_start <= now < p_end:
            current = programme
            continue
        if p_start >= now and upcoming is None:
            upcoming = programme

    description_lines: list[str] = []
    if current:
        description_lines.append(
            f"NOW: {current.get('title', 'Untitled')} • "
            f"{_display_time(current['startTime'])}–{_display_time(current['endTime'])}"
        )
    if upcoming:
        label = "NEXT" if current else "UP NEXT"
        description_lines.append(
            f"{label}: {upcoming.get('title', 'Untitled')} • "
            f"{_display_time(upcoming['startTime'])}–{_display_time(upcoming['endTime'])}"
        )
    if not current and not upcoming and channel.get("epg_matched"):
        description_lines.append("LIVE • No current guide data")
    elif channel.get("epg_status") == "unavailable":
        description_lines.append("LIVE • No exact listing in the guide source")
    elif not channel.get("epg_matched"):
        description_lines.append("LIVE • Guide match pending")

    info_bits = []
    if channel.get("philly"):
        info_bits.append("Philadelphia")
    info_bits.append(channel.get("category", "Live TV"))
    info_bits.append(
        f"{len(channel.get('streams', []))} source"
        + ("s" if len(channel.get("streams", [])) != 1 else "")
    )
    description_lines.append(" • ".join(info_bits))

    version = channel.get("build_version")
    built_at = channel.get("built_at")
    build_id = channel.get("build_id")
    if version or built_at or build_id:
        build_bits = []
        if version:
            build_bits.append(f"v{version}")
        if build_id:
            build_bits.append(f"build {build_id}")
        if built_at:
            build_bits.append(built_at)
        description_lines.append(" • ".join(build_bits))

    meta: dict[str, Any] = {
        "id": channel["id"],
        "type": "tv",
        "name": channel["name"],
        "description": "\n".join(description_lines),
        "genres": [channel.get("category", "Live TV")],
        "posterShape": "square",
        "behaviorHints": {"isLive": True},
    }
    if channel.get("logo"):
        meta["poster"] = channel["logo"]
        meta["background"] = channel["logo"]
    if build_id:
        meta["stremioTvBuild"] = {
            "build_id": build_id,
            "version": version,
            "built_at": built_at,
            "source_revision": channel.get("build_revision"),
        }
    return meta


def stream_object(
    stream: dict[str, Any],
    index: int,
    build_info: dict[str, Any] | None = None,
) -> dict[str, Any]:
    quality = quality_label(stream, "Live")
    source = _short_source_name(stream)
    role = stream.get("display_role") or ("Primary" if index == 1 else f"Backup {index - 1}")
    url = stream["url"]
    health = stream.get("health") or {}
    health_status = health.get("status", "not-tested")

    description_parts = [source]
    if stream.get("measured_quality"):
        description_parts.append(f"Measured {quality}")
    elif stream.get("quality"):
        description_parts.append(f"Declared {quality}")
    else:
        description_parts.append("Quality unknown")
    if health_status == "ok":
        description_parts.append("Builder check: OK")
    elif health_status == "failed":
        description_parts.append("Builder check failed — may still work locally")
    else:
        description_parts.append("Not health-tested")
    stability = (stream.get("stability") or {}).get("classification")
    if stability and stability != "Unknown":
        description_parts.append(stability)
    if build_info:
        description_parts.append(f"Build {build_info.get('build_id')}")

    obj: dict[str, Any] = {
        "name": f"{role} • {quality}",
        "description": " • ".join(description_parts),
        "url": url,
    }
    if build_info:
        obj["stremioTvBuild"] = {
            "build_id": build_info.get("build_id"),
            "version": build_info.get("version"),
            "built_at": build_info.get("built_at"),
            "source_revision": build_info.get("source_revision"),
        }
    headers = stream.get("headers") or {}
    not_web_ready = (not url.lower().startswith("https://")) or bool(headers)
    if not_web_ready:
        obj["behaviorHints"] = {"notWebReady": True}
        if headers:
            obj["behaviorHints"]["proxyHeaders"] = {"request": headers}
    return obj


def write_paginated_catalog(
    base_dir: Path,
    catalog_id: str,
    channels: list[dict[str, Any]],
    page_size: int,
    discover_only: bool = False,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / catalog_id
    catalog_dir.mkdir(parents=True, exist_ok=True)
    previews = [channel_preview(c) for c in channels]

    first = {"metas": previews[:page_size]}
    # Keep the plain endpoint for debugging/compatibility. For discover-only
    # catalogs the manifest requires genre=All, so Stremio's Board/Home will
    # not request this endpoint.
    write_json(base_dir / "catalog" / "tv" / f"{catalog_id}.json", first)
    write_json(catalog_dir / "skip=0.json", first)

    if discover_only:
        write_json(catalog_dir / "genre=All.json", first)
        write_json(catalog_dir / "genre=All&skip=0.json", first)
        write_json(catalog_dir / "skip=0&genre=All.json", first)

    skip = page_size
    while skip < len(previews):
        page = {"metas": previews[skip : skip + page_size]}
        write_json(catalog_dir / f"skip={skip}.json", page)
        if discover_only:
            write_json(catalog_dir / f"genre=All&skip={skip}.json", page)
            write_json(catalog_dir / f"skip={skip}&genre=All.json", page)
        skip += page_size

    # Stremio continues until it receives an empty page.
    final_skip = ((len(previews) + page_size - 1) // page_size) * page_size
    if final_skip == 0:
        final_skip = page_size
    empty = {"metas": []}
    write_json(catalog_dir / f"skip={final_skip}.json", empty)
    if discover_only:
        write_json(catalog_dir / f"genre=All&skip={final_skip}.json", empty)
        write_json(catalog_dir / f"skip={final_skip}&genre=All.json", empty)

def write_epg_pages(
    base_dir: Path,
    channels: list[dict[str, Any]],
    schedules: dict[str, dict[str, list[dict[str, Any]]]],
    dates: Iterable[str],
    page_size: int,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / "channels"
    catalog_dir.mkdir(parents=True, exist_ok=True)

    for day in sorted(dates):
        detailed = [channel_meta(c, schedules.get(c["id"], {}).get(day, [])) for c in channels]
        first = detailed[:page_size]
        write_json(catalog_dir / f"date={day}.json", {"metasDetailed": first})
        write_json(catalog_dir / f"date={day}&skip=0.json", {"metasDetailed": first})
        write_json(catalog_dir / f"skip=0&date={day}.json", {"metasDetailed": first})

        skip = page_size
        while skip < len(detailed):
            page = detailed[skip : skip + page_size]
            write_json(catalog_dir / f"date={day}&skip={skip}.json", {"metasDetailed": page})
            write_json(catalog_dir / f"skip={skip}&date={day}.json", {"metasDetailed": page})
