== 0.0):
                return epg_id, "fuzzy-name", best_score

    return None, "unmatched", 0.0


def parse_xmltv_time(value: str) -> dt.datetime | None:
    value = (value or "").strip()
    if not value:
        return None
    # XMLTV often uses: YYYYMMDDHHMMSS ±HHMM
    formats = ["%Y%m%d%H%M%S %z", "%Y%m%d%H%M %z", "%Y%m%d%H%M%S", "%Y%m%d%H%M"]
    for fmt in formats:
        try:
            parsed = dt.datetime.strptime(value, fmt)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=UTC)
            return parsed.astimezone(UTC)
        except ValueError:
            continue
    return None


def iso_z(value: dt.datetime) -> str:
    return value.astimezone(UTC).isoformat(timespec="seconds").replace("+00:00", "Z")


def date_range_for_program(start: dt.datetime, end: dt.datetime) -> list[str]:
    if end <= start:
        return []
    cursor = start.date()
    last = (end - dt.timedelta(microseconds=1)).date()
    result: list[str] = []
    while cursor <= last:
        result.append(cursor.isoformat())
        cursor += dt.timedelta(days=1)
    return result


def build_program_index(
    epg_path: Path,
    epg_to_channels: dict[str, list[str]],
) -> tuple[dict[str, dict[str, list[dict[str, Any]]]], set[str]]:
    schedules: dict[str, dict[str, list[dict[str, Any]]]] = defaultdict(lambda: defaultdict(list))
    all_dates: set[str] = set()
    matched_programmes = 0

    with gzip.open(epg_path, "rb") as f:
        for event, elem in ET.iterparse(f, events=("end",)):
            if elem.tag != "programme":
                if elem.tag == "channel":
                    elem.clear()
                continue

            epg_id = elem.attrib.get("channel", "")
            channel_ids = epg_to_channels.get(epg_id)
            if not channel_ids:
                elem.clear()
                continue

            start = parse_xmltv_time(elem.attrib.get("start", ""))
            end = parse_xmltv_time(elem.attrib.get("stop", ""))
            if not start or not end or end <= start:
                elem.clear()
                continue

            title = extract_text(elem, "title") or "Untitled"
            desc = extract_text(elem, "desc")
            categories = [
                (x.text or "").strip()
                for x in elem.findall("category")
                if (x.text or "").strip()
            ]
            icon_elem = elem.find("icon")
            thumbnail = icon_elem.attrib.get("src", "") if icon_elem is not None else ""
            date_text = extract_text(elem, "date")
            runtime_min = max(1, round((end - start).total_seconds() / 60))
            start_iso = iso_z(start)
            end_iso = iso_z(end)

            for channel_id in channel_ids:
                programme: dict[str, Any] = {
                    "id": f"{channel_id}:epg:{start_iso}",
                    "title": title,
                    "released": start_iso,
                    "startTime": start_iso,
                    "endTime": end_iso,
                    "runtime": f"{runtime_min} min",
                }
                if desc:
                    programme["overview"] = desc
                if thumbnail:
                    programme["thumbnail"] = thumbnail
                if categories:
                    programme["genres"] = categories[:8]
                if date_text[:4].isdigit():
                    programme["releaseInfo"] = date_text[:4]

                for day in date_range_for_program(start, end):
                    schedules[channel_id][day].append(programme)
                    all_dates.add(day)
            matched_programmes += 1
            elem.clear()

    for channel_days in schedules.values():
        for day, programmes in channel_days.items():
            programmes.sort(key=lambda p: p["startTime"])

    log(f"Matched {matched_programmes:,} XMLTV programmes to playlist channels")
    return schedules, all_dates

def _fetch_public_text(url: str, timeout: int = 12, byte_limit: int = 2_000_000) -> tuple[str, str]:
    """Fetch a public page/API response without credentials or session cookies."""
    req = urllib.request.Request(
        url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "text/html,application/json,text/plain,*/*",
            "Accept-Encoding": "identity",
        },
    )
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        final_url = resp.geturl()
        body = resp.read(byte_limit)
    return body.decode("utf-8", errors="ignore"), final_url


def _extract_public_hls_urls(text: str, base_url: str = "") -> list[str]:
    """Extract plainly published HLS URLs from public text/JSON.

    This intentionally does not copy cookies, bearer tokens, DRM data, or attempt
    to manufacture signed URLs. Query strings that are already present in public
    page source are preserved and still have to pass the health checker.
    """
    normalized = html.unescape(text or "")
    normalized = normalized.replace("\\u002F", "/").replace("\\/", "/")
    normalized = normalized.replace("\\u0026", "&")

    found: list[str] = []
    absolute = re.findall(
        r'https?://[^\\s\\"\'<>]+?\\.m3u8(?:\\?[^\\s\\"\'<>]*)?',
        normalized,
        flags=re.I,
    )
    for url in absolute:
        found.append(url.rstrip("),.;"))

    if base_url:
        relative = re.findall(
            r'(?:(?:src|url|file|stream|hls)[\"\'=: ]+)(/[^\\s\\"\'<>]+?\\.m3u8(?:\\?[^\\s\\"\'<>]*)?)',
            normalized,
            flags=re.I,
        )
        for path in relative:
            found.append(urllib.parse.urljoin(base_url, path.rstrip("),.;")))

    return list(dict.fromkeys(found))


def discover_official_page_streams(config: dict[str, Any]) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    settings = (config.get("discovery") or {}).get("official_pages") or []

    for item in settings:
        page_url = item.get("page_url")
        tvg_id = item.get("tvg_id")
        if not page_url or not tvg_id:
            continue

        row: dict[str, Any] = {
            "kind": "official-page",
            "name": item.get("name") or tvg_id,
            "tvg_id": tvg_id,
            "page_url": page_url,
            "status": "pending",
            "candidates": [],
        }
        try:
            page_text, final_url = _fetch_public_text(
                page_url,
                timeout=int(item.get("timeout_seconds", 12)),
            )
            urls = _extract_public_hls_urls(page_text, final_url)

            # Guard against unrelated clip/ad manifests on a station page. If the
            # page publishes several HLS URLs, keep only URLs whose nearby source
            # context looks like a live/linear stream.
            accepted: list[str] = []
            normalized = html.unescape(page_text).replace("\\u002F", "/").replace("\\/", "/")
            for url in urls:
                idx = normalized.find(url)
                context = normalized[max(0, idx - 500): idx + len(url) + 500].lower() if idx >= 0 else ""
                if len(urls) == 1 or any(word in context for word in ("live", "livestream", "linear", "streaming")):
                    accepted.append(url)

            for url in accepted[: int(item.get("max_candidates", 4))]:
                discovered.append({
                    "name_raw": item.get("name") or tvg_id,
                    "name": item.get("name") or tvg_id,
                    "quality": item.get("quality"),
                    "tvg_id": tvg_id,
                    "logo": item.get("logo", ""),
                    "group": item.get("group", "Local"),
                    "language": "English",
                    "country": "US",
                    "source": item.get("source") or f"Official page: {urllib.parse.urlsplit(page_url).netloc}",
                    "family": "official-page",
                    "priority": int(item.get("priority", -10)),
                    "philly": bool(item.get("philly", True)),
                    "url": url,
                    "headers": {},
                    "source_type": "official-public-page",
                    "content_kind": item.get("content_kind", "unknown"),
                    "content_note": item.get("content_note", ""),
                })
            row["candidates"] = accepted[: int(item.get("max_candidates", 4))]
            row["status"] = "found" if row["candidates"] else "no-public-hls-in-page-source"
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    return discovered, rows


def discover_iptv_org_passed_issues(config: dict[str, Any]) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Discover recent public IPTV-org additions already labeled check:passed.

    IPTV-org's label is only a discovery pre-filter; every URL still goes through
    our own double HLS/media-segment probe before it can reach Stremio.
    """
    settings = (config.get("discovery") or {}).get("iptv_org_issues") or {}
    if not settings.get("enabled", True):
        return [], []

    targets = settings.get("targets") or []
    if not targets:
        return [], []

    query = 'repo:iptv-org/iptv is:issue is:open label:"check:passed"'
    params = urllib.parse.urlencode({
        "q": query,
        "sort": "updated",
        "order": "desc",
        "per_page": int(settings.get("per_page", 100)),
    })
    api_url = "https://api.github.com/search/issues?" + params
    rows: list[dict[str, Any]] = []
    discovered: list[dict[str, Any]] = []

    try:
        payload_text, _ = _fetch_public_text(api_url, timeout=int(settings.get("timeout_seconds", 20)))
        payload = json.loads(payload_text)
        issues = payload.get("items") or []
    except Exception as exc:
        return [], [{
            "kind": "iptv-org-issues",
            "status": "failed",
            "error": str(exc)[:240],
            "api_url": api_url,
        }]

    for issue in issues:
        title = issue.get("title") or ""
        body = issue.get("body") or ""
        combined = f"{title}\n{body}"
        combined_norm = norm(combined)

        matched_target: dict[str, Any] | None = None
        for target in targets:
            terms = [target.get("tvg_id", "")] + list(target.get("aliases") or [])
            if any(term and (norm_id(term) in norm_id(combined) or norm(term) in combined_norm) for term in terms):
                matched_target = target
                break
        if not matched_target:
            continue

        urls = _extract_public_hls_urls(combined)
        urls = [u for u in urls if ".m3u8" in u.lower()]
        if not urls:
            continue

        issue_number = issue.get("number")
        issue_url = issue.get("html_url")
        for url in urls[: int(settings.get("max_urls_per_issue", 3))]:
            discovered.append({
                "name_raw": matched_target.get("name") or matched_target["tvg_id"],
                "name": matched_target.get("name") or matched_target["tvg_id"],
                "quality": None,
                "tvg_id": matched_target["tvg_id"],
                "logo": "",
                "group": matched_target.get("group", ""),
                "language": "English",
                "country": "US",
                "source": f"IPTV-org check:passed issue #{issue_number}",
                "family": "iptv-org-issue",
                "priority": int(settings.get("priority", 6)),
                "philly": bool(matched_target.get("philly", False)),
                "url": url,
                "headers": {},
                "source_type": "public-third-party",
                "content_kind": matched_target.get("content_kind", "unknown"),
                "discovery_issue": issue_url,
            })
        rows.append({
            "kind": "iptv-org-issue",
            "status": "candidate-found",
            "issue": issue_number,
            "issue_url": issue_url,
            "tvg_id": matched_target["tvg_id"],
            "name": matched_target.get("name"),
            "urls": urls[: int(settings.get("max_urls_per_issue", 3))],
        })

    return discovered, rows

def discover_dynamic_streams(config: dict[str, Any]) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    if not (config.get("discovery") or {}).get("enabled", True):
        return [], []

    official_entries, official_rows = discover_official_page_streams(config)
    issue_entries, issue_rows = discover_iptv_org_passed_issues(config)
    seed_entries, seed_rows = discover_seed_candidates(config)

    combined: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    for entry in official_entries + issue_entries + seed_entries:
        marker = (canonical_tvg_id(entry.get("tvg_id", "")), entry.get("url", ""))
        if marker in seen:
            continue
        seen.add(marker)
        combined.append(entry)

    log(
        f"Dynamic discovery: {len(official_entries)} official-page + "
        f"{len(issue_entries)} IPTV-org issue + {len(seed_entries)} seed candidates"
    )
    return combined, official_rows + issue_rows + seed_rows


def _probe_stream(stream: dict[str, Any], timeout: int, verify_segment: bool) -> dict[str, Any]:
    url = stream.get("url") or ""
    if not re.match(r"^https?://", url, re.I):
        return {"status": "not-tested", "detail": "Non-HTTP stream"}

    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "*/*",
        "Accept-Encoding": "identity",
    }
    headers.update(stream.get("headers") or {})
    started = time.monotonic()

    def fetch(target: str, byte_limit: int = 262144) -> tuple[int, str, str, bytes]:
        req = urllib.request.Request(target, headers=headers)
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            status = int(getattr(resp, "status", 200) or 200)
            final_url = resp.geturl()
            content_type = (resp.headers.get("Content-Type") or "").lower()
            body = resp.read(byte_limit)
            return status, final_url, content_type, body

    try:
        status, final_url, content_type, body = fetch(url)
        elapsed_ms = round((time.monotonic() - started) * 1000)
        text = body.decode("utf-8", errors="ignore")
        is_hls = (
            "#EXTM3U" in text
            or ".m3u8" in urllib.parse.urlsplit(final_url).path.lower()
            or "mpegurl" in content_type
        )

        if is_hls:
            if "#EXTM3U" not in text:
                return {
                    "status": "failed",
                    "detail": "HLS URL did not return a valid playlist",
                    "http_status": status,
                    "latency_ms": elapsed_ms,
                }

            media_text = text
            media_url = final_url
            lines = [line.strip() for line in text.splitlines() if line.strip() and not line.startswith("#")]
            if "#EXT-X-STREAM-INF" in text and lines:
                candidate = urllib.parse.urljoin(final_url, lines[0])
                child_status, child_url, _child_type, child_body = fetch(candidate)
                media_text = child_body.decode("utf-8", errors="ignore")
                media_url = child_url
                if "#EXTM3U" not in media_text:
                    return {
                        "status": "failed",
                        "detail": "Master playlist child was invalid",
                        "http_status": child_status,
                        "latency_ms": round((time.monotonic() - started) * 1000),
                    }

            if verify_segment:
                segments = [
                    line.strip()
                    for line in media_text.splitlines()
                    if line.strip() and not line.startswith("#")
                ]
                if segments:
                    segment_url = urllib.parse.urljoin(media_url, segments[0])
                    seg_headers = dict(headers)
                    seg_headers["Range"] = "bytes=0-1023"
                    req = urllib.request.Request(segment_url, headers=seg_headers)
                    with urllib.request.urlopen(req, timeout=timeout) as resp:
                        resp.read(1)

            return {
                "status": "ok",
                "detail": "Valid HLS playlist" + (" + segment" if verify_segment else ""),
                "http_status": status,
                "latency_ms": round((time.monotonic() - started) * 1000),
            }

        if "text/html" in content_type:
            return {
                "status": "failed",
                "detail": "Returned HTML instead of media",
                "http_status": status,
                "latency_ms": elapsed_ms,
            }

        return {
            "status": "ok",
            "detail": "Reachable direct media URL",
            "http_status": status,
            "latency_ms": elapsed_ms,
        }

    except urllib.error.HTTPError as exc:
        return {
            "status": "failed",
            "detail": f"HTTP {exc.code}",
            "http_status": exc.code,
            "latency_ms": round((time.monotonic() - started) * 1000),
        }
    except Exception as exc:
        return {
            "status": "failed",
            "detail": str(exc)[:180],
            "latency_ms": round((time.monotonic() - started) * 1000),
        }


def probe_candidate_entries(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Probe the full curated candidate pool before selecting channel streams.

    Production can require two successful probes separated by one global wait.
    This catches transient/cached URLs without multiplying the wait by the
    number of streams.
    """
    settings = config.get("stream_health") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    require_passed_only = bool(settings.get("require_passed_only", True))
    timeout = int(settings.get("timeout_seconds", 6))
    max_workers = int(settings.get("max_workers", 16))
    verify_segment = bool(settings.get("verify_segment_for_all_candidates", True))
    double_probe = bool(settings.get("double_probe", True))
    second_probe_delay = int(settings.get("second_probe_delay_seconds", 60))

    rows: list[dict[str, Any]] = []

    if not enabled:
        for entry in entries:
            entry["health"] = {"status": "not-tested", "detail": "Health checks disabled"}
            rows.append({
                "name": entry.get("name"),
                "tvg_id": entry.get("tvg_id"),
                "source": entry.get("source"),
                "family": entry.get("family"),
                "quality": entry.get("quality") or "Unknown",
                "url": entry.get("url"),
                "health": entry.get("health"),
            })
        return entries, rows

    jobs: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    refs: dict[tuple[str, tuple[tuple[str, str], ...]], list[dict[str, Any]]] = defaultdict(list)

    for entry in entries:
        headers_key = tuple(sorted((entry.get("headers") or {}).items()))
        key = (entry.get("url") or "", headers_key)
        jobs.setdefault(key, entry)
        refs[key].append(entry)

    def run_probe_batch(batch: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]) -> dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]:
        batch_results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
        with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
            future_map = {
                executor.submit(_probe_stream, entry, timeout, verify_segment): key
                for key, entry in batch.items()
            }
            for future in concurrent.futures.as_completed(future_map):
                key = future_map[future]
                try:
                    batch_results[key] = future.result()
                except Exception as exc:
                    batch_results[key] = {"status": "failed", "detail": str(exc)[:180]}
        return batch_results

    results = run_probe_batch(jobs)

    first_pass = {
        key: jobs[key]
        for key, health in results.items()
        if health.get("status") == "ok"
    }

    if double_probe and first_pass:
        log(
            f"Health stability check: {len(first_pass):,} candidates passed probe 1; "
            f"waiting {second_probe_delay}s before probe 2"
        )
        time.sleep(max(0, second_probe_delay))
        second_results = run_probe_batch(first_pass)

        for key, first_health in list(results.items()):
            if first_health.get("status") != "ok":
                continue
            second_health = second_results.get(key, {"status": "failed", "detail": "No second probe result"})
            if second_health.get("status") == "ok":
                results[key] = {
                    **first_health,
                    "status": "ok",
                    "detail": "Passed two HLS/media probes",
                    "probe_1": first_health,
                    "probe_2": second_health,
                    "stable": True,
                }
            else:
                results[key] = {
                    **second_health,
                    "status": "failed",
                    "detail": f"Probe 2 failed after probe 1 passed: {second_health.get('detail', 'unknown')}",
                    "probe_1": first_health,
                    "probe_2": second_health,
                    "stable": False,
                }

    for key, linked_entries in refs.items():
        health = results.get(key, {"status": "failed", "detail": "No probe result"})
        for entry in linked_entries:
            entry["health"] = dict(health)

    for entry in entries:
        rows.append({
            "name": entry.get("name"),
            "tvg_id": entry.get("tvg_id"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "source_type": entry.get("source_type"),
            "quality": entry.get("quality") or "Unknown",
            "url": entry.get("url"),
            "health": entry.get("health"),
            "discovery_issue": entry.get("discovery_issue"),
            "content_note": entry.get("content_note"),
        })

    counts = defaultdict(int)
    for row in rows:
        counts[(row.get("health") or {}).get("status", "not-tested")] += 1
    log(
        "Candidate stream health: "
        + ", ".join(f"{status}={count}" for status, count in sorted(counts.items()))
    )

    if not require_passed_only:
        return entries, rows

    healthy = [
        entry
        for entry in entries
        if (entry.get("health") or {}).get("status") == "ok"
    ]
    log(f"Passed-only policy: keeping {len(healthy):,}/{len(entries):,} curated candidates")
    return healthy, rows


def label_channel_streams(channels: list[dict[str, Any]]) -> None:
    for channel in channels:
        for index, stream in enumerate(channel.get("streams", [])):
            stream["display_role"] = "Primary" if index == 0 else f"Backup {index}"


def channel_meta(channel: dict[str, Any], programmes: list[dict[str, Any]] | None = None) -> dict[str, Any]:
    programmes = sorted(programmes or [], key=lambda p: p.get("startTime", ""))
    now = dt.datetime.now(UTC)

    current: dict[str, Any] | None = None
    upcoming: dict[str, Any] | None = None
    for programme in programmes:
        try:
            p_start = dt.datetime.fromisoformat(programme["startTime"].replace("Z", "+00:00"))
            p_end = dt.datetime.fromisoformat(programme["endTime"].replace("Z", "+00:00"))
        except (KeyError, ValueError):
            continue
        if p_start <= now < p_end:
            current = programme
            continue
        if p_start >= now and upcoming is None:
            upcoming = programme

    description_lines: list[str] = []
    if current:
        description_lines.append(
            f"NOW: {current.get('title', 'Untitled')} • "
            f"{_display_time(current['startTime'])}–{_display_time(current['endTime'])}"
        )
    if upcoming:
        label = "NEXT" if current else "UP NEXT"
        description_lines.append(
            f"{label}: {upcoming.get('title', 'Untitled')} • "
            f"{_display_time(upcoming['startTime'])}–{_display_time(upcoming['endTime'])}"
        )
    if not current and not upcoming and channel.get("epg_matched"):
        description_lines.append("LIVE TV • No current programme data")
    elif not channel.get("epg_matched"):
        description_lines.append("LIVE TV • EPG unavailable")

    info_bits = []
    if channel.get("philly"):
        info_bits.append("Philadelphia local")
    info_bits.append(channel.get("category", "Live TV"))
    info_bits.append(
        f"{len(channel.get('streams', []))} stream source"
        + ("s" if len(channel.get("streams", [])) != 1 else "")
    )
    description_lines.append("")
    description_lines.append(" • ".join(info_bits))

    version = channel.get("build_version")
    built_at = channel.get("built_at")
    if version or built_at:
        build_bits = []
        if version:
            build_bits.append(f"Stremio TV v{version}")
        if built_at:
            build_bits.append(f"built {_display_time(built_at)} ET")
        description_lines.append(" • ".join(build_bits))

    meta: dict[str, Any] = {
        "id": channel["id"],
        "type": "tv",
        "name": channel["name"],
        "description": "\n".join(description_lines),
        "genres": [channel.get("category", "Live TV")],
        "posterShape": "square",
        "behaviorHints": {"isLive": True},
    }
    if channel.get("logo"):
        meta["poster"] = channel["logo"]
        meta["background"] = channel["logo"]
    return meta


def stream_object(stream: dict[str, Any], index: int) -> dict[str, Any]:
    quality = stream.get("quality") or "Live"
    source = _short_source_name(stream)
    role = stream.get("display_role") or ("Primary" if index == 1 else f"Backup {index - 1}")
    url = stream["url"]
    health = stream.get("health") or {}
    health_status = health.get("status", "not-tested")

    description_parts = [source, quality]
    if health_status == "ok":
        description_parts.append("Builder check: OK")
    elif health_status == "failed":
        description_parts.append("Builder check failed — may still work locally")
    else:
        description_parts.append("Not health-tested")

    obj: dict[str, Any] = {
        "name": f"{role} • {quality} • {source}",
        "description": " • ".join(description_parts),
        "url": url,
    }
    headers = stream.get("headers") or {}
    not_web_ready = (not url.lower().startswith("https://")) or bool(headers)
    if not_web_ready:
        obj["behaviorHints"] = {"notWebReady": True}
        if headers:
            obj["behaviorHints"]["proxyHeaders"] = {"request": headers}
    return obj


def write_paginated_catalog(
    base_dir: Path,
    catalog_id: str,
    channels: list[dict[str, Any]],
    page_size: int,
    discover_only: bool = False,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / catalog_id
    catalog_dir.mkdir(parents=True, exist_ok=True)
    previews = [channel_preview(c) for c in channels]

    first = {"metas": previews[:page_size]}
    # Keep the plain endpoint for debugging/compatibility. For discover-only
    # catalogs the manifest requires genre=All, so Stremio's Board/Home will
    # not request this endpoint.
    write_json(base_dir / "catalog" / "tv" / f"{catalog_id}.json", first)
    write_json(catalog_dir / "skip=0.json", first)

    if discover_only:
        write_json(catalog_dir / "genre=All.json", first)
        write_json(catalog_dir / "genre=All&skip=0.json", first)
        write_json(catalog_dir / "skip=0&genre=All.json", first)

    skip = page_size
    while skip < len(previews):
        page = {"metas": previews[skip : skip + page_size]}
        write_json(catalog_dir / f"skip={skip}.json", page)
        if discover_only:
            write_json(catalog_dir / f"genre=All&skip={skip}.json", page)
            write_json(catalog_dir / f"skip={skip}&genre=All.json", page)
        skip += page_size

    # Stremio continues until it receives an empty page.
    final_skip = ((len(previews) + page_size - 1) // page_size) * page_size
    if final_skip == 0:
        final_skip = page_size
    empty = {"metas": []}
    write_json(catalog_dir / f"skip={final_skip}.json", empty)
    if discover_only:
        write_json(catalog_dir / f"genre=All&skip={final_skip}.json", empty)
        write_json(catalog_dir / f"skip={final_skip}&genre=All.json", empty)
