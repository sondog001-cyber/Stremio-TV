                return epg_id, "fuzzy-name", best_score

    return None, "unmatched", 0.0


def parse_xmltv_time(value: str) -> dt.datetime | None:
    value = (value or "").strip()
    if not value:
        return None
    # XMLTV often uses: YYYYMMDDHHMMSS ±HHMM
    formats = ["%Y%m%d%H%M%S %z", "%Y%m%d%H%M %z", "%Y%m%d%H%M%S", "%Y%m%d%H%M"]
    for fmt in formats:
        try:
            parsed = dt.datetime.strptime(value, fmt)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=UTC)
            return parsed.astimezone(UTC)
        except ValueError:
            continue
    return None


def iso_z(value: dt.datetime) -> str:
    return value.astimezone(UTC).isoformat(timespec="seconds").replace("+00:00", "Z")


def date_range_for_program(start: dt.datetime, end: dt.datetime) -> list[str]:
    if end <= start:
        return []
    cursor = start.date()
    last = (end - dt.timedelta(microseconds=1)).date()
    result: list[str] = []
    while cursor <= last:
        result.append(cursor.isoformat())
        cursor += dt.timedelta(days=1)
    return result


def build_program_index(
    epg_path: Path,
    epg_to_channels: dict[str, list[str]],
) -> tuple[dict[str, dict[str, list[dict[str, Any]]]], set[str]]:
    schedules: dict[str, dict[str, list[dict[str, Any]]]] = defaultdict(lambda: defaultdict(list))
    all_dates: set[str] = set()
    matched_programmes = 0

    with gzip.open(epg_path, "rb") as f:
        for event, elem in ET.iterparse(f, events=("end",)):
            if elem.tag != "programme":
                if elem.tag == "channel":
                    elem.clear()
                continue

            epg_id = elem.attrib.get("channel", "")
            channel_ids = epg_to_channels.get(epg_id)
            if not channel_ids:
                elem.clear()
                continue

            start = parse_xmltv_time(elem.attrib.get("start", ""))
            end = parse_xmltv_time(elem.attrib.get("stop", ""))
            if not start or not end or end <= start:
                elem.clear()
                continue

            title = extract_text(elem, "title") or "Untitled"
            desc = extract_text(elem, "desc")
            categories = [
                (x.text or "").strip()
                for x in elem.findall("category")
                if (x.text or "").strip()
            ]
            icon_elem = elem.find("icon")
            thumbnail = icon_elem.attrib.get("src", "") if icon_elem is not None else ""
            date_text = extract_text(elem, "date")
            runtime_min = max(1, round((end - start).total_seconds() / 60))
            start_iso = iso_z(start)
            end_iso = iso_z(end)

            for channel_id in channel_ids:
                programme: dict[str, Any] = {
                    "id": f"{channel_id}:epg:{start_iso}",
                    "title": title,
                    "released": start_iso,
                    "startTime": start_iso,
                    "endTime": end_iso,
                    "runtime": f"{runtime_min} min",
                }
                if desc:
                    programme["overview"] = desc
                if thumbnail:
                    programme["thumbnail"] = thumbnail
                if categories:
                    programme["genres"] = categories[:8]
                if date_text[:4].isdigit():
                    programme["releaseInfo"] = date_text[:4]

                for day in date_range_for_program(start, end):
                    schedules[channel_id][day].append(programme)
                    all_dates.add(day)
            matched_programmes += 1
            elem.clear()

    for channel_days in schedules.values():
        for day, programmes in channel_days.items():
            programmes.sort(key=lambda p: p["startTime"])

    log(f"Matched {matched_programmes:,} XMLTV programmes to playlist channels")
    return schedules, all_dates

_PUBLIC_TEXT_CACHE: dict[tuple[str, int], tuple[str, str]] = {}


def _fetch_public_text(url: str, timeout: int = 12, byte_limit: int = 2_000_000) -> tuple[str, str]:
    """Fetch public text, using the Actions token only for GitHub's public API."""
    cache_key = (url, byte_limit)
    if cache_key in _PUBLIC_TEXT_CACHE:
        return _PUBLIC_TEXT_CACHE[cache_key]
    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "text/html,application/json,text/plain,*/*",
        "Accept-Encoding": "identity",
    }
    if urllib.parse.urlsplit(url).netloc.lower() == "api.github.com":
        headers["Accept"] = "application/vnd.github+json"
        headers["X-GitHub-Api-Version"] = "2022-11-28"
        github_token = os.getenv("GITHUB_TOKEN", "").strip()
        if github_token:
            headers["Authorization"] = f"Bearer {github_token}"
    req = urllib.request.Request(url, headers=headers)
    with urllib.request.urlopen(req, timeout=timeout) as resp:
        final_url = resp.geturl()
        body = resp.read(byte_limit)
    result = (body.decode("utf-8", errors="ignore"), final_url)
    _PUBLIC_TEXT_CACHE[cache_key] = result
    return result


def _extract_public_hls_urls(text: str, base_url: str = "") -> list[str]:
    """Extract plainly published HLS URLs from public text/JSON.

    This intentionally does not copy cookies, bearer tokens, DRM data, or attempt
    to manufacture signed URLs. Query strings that are already present in public
    page source are preserved and still have to pass the health checker.
    """
    normalized = html.unescape(text or "")
    normalized = normalized.replace("\\u002F", "/").replace("\\/", "/")
    normalized = normalized.replace("\\u0026", "&")

    found: list[str] = []
    absolute = re.findall(
        r'https?://[^\s\\"\'<>]+?\.m3u8(?:\?[^\s\\"\'<>]*)?',
        normalized,
        flags=re.I,
    )
    for url in absolute:
        found.append(url.rstrip("),.;"))

    if base_url:
        relative = re.findall(
            r'(?:(?:src|url|file|stream|hls)[\"\'=: ]+)(/[^\s\\"\'<>]+?\.m3u8(?:\?[^\s\\"\'<>]*)?)',
            normalized,
            flags=re.I,
        )
        for path in relative:
            found.append(urllib.parse.urljoin(base_url, path.rstrip("),.;")))

    return list(dict.fromkeys(found))


def _is_safe_public_document_url(url: str) -> bool:
    parsed = urllib.parse.urlsplit(url)
    if parsed.scheme not in {"http", "https"} or not parsed.hostname:
        return False
    hostname = parsed.hostname.lower().rstrip(".")
    if hostname == "localhost" or hostname.endswith(".localhost"):
        return False
    try:
        address = ipaddress.ip_address(hostname)
    except ValueError:
        return True
    return not (
        address.is_private
        or address.is_loopback
        or address.is_link_local
        or address.is_reserved
        or address.is_multicast
    )


def _extract_player_document_urls(text: str, base_url: str) -> list[tuple[str, str]]:
    normalized = html.unescape(text or "").replace("\\u002F", "/").replace("\\/", "/")
    found: list[tuple[str, str]] = []
    patterns = (
        ("iframe", r'<iframe[^>]+src=["\']([^"\']+)["\']'),
        ("script", r'<script[^>]+src=["\']([^"\']+)["\']'),
        ("player-config", r'["\'](https?://[^"\'<>\s]+(?:player|embed|manifest|livestream|live)[^"\'<>\s]*(?:\.json|\.js)?[^"\'<>\s]*)["\']'),
    )
    for kind, pattern in patterns:
        for match in re.findall(pattern, normalized, flags=re.I):
            candidate = urllib.parse.urljoin(base_url, match.rstrip("),.;"))
            if ".m3u8" in urllib.parse.urlsplit(candidate).path.lower():
                continue
            if _is_safe_public_document_url(candidate):
                found.append((candidate, kind))
    return list(dict.fromkeys(found))


def scan_official_player_documents(
    page_text: str,
    page_url: str,
    config: dict[str, Any],
    item: dict[str, Any],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Follow a bounded set of public player documents and find HLS manifests."""
    settings = (config.get("discovery") or {}).get("official_player_scan") or {}
    enabled = bool(settings.get("enabled", True))
    max_documents = max(1, int(item.get("max_player_documents", settings.get("max_documents", 8))))
    max_depth = max(0, min(3, int(item.get("player_scan_depth", settings.get("max_depth", 2)))))
    timeout = max(1, int(item.get("timeout_seconds", settings.get("timeout_seconds", 12))))
    byte_limit = max(100_000, int(settings.get("byte_limit", 2_000_000)))
    queue: list[tuple[str, str, int, str]] = [(page_url, page_text, 0, "page")]
    seen_documents: set[str] = set()
    seen_manifests: set[str] = set()
    manifests: list[dict[str, Any]] = []
    documents: list[dict[str, Any]] = []

    while queue and len(documents) < max_documents:
        document_url, document_text, depth, kind = queue.pop(0)
        if document_url in seen_documents:
            continue
        seen_documents.add(document_url)
        urls = _extract_public_hls_urls(document_text, document_url)
        normalized = html.unescape(document_text).replace("\\u002F", "/").replace("\\/", "/")
        accepted: list[str] = []
        for url in urls:
            if url in seen_manifests or not _is_safe_public_document_url(url):
                continue
            index = normalized.find(url)
            context = normalized[max(0, index - 500): index + len(url) + 500].lower() if index >= 0 else ""
            if len(urls) == 1 or any(word in context for word in ("live", "livestream", "linear", "streaming")):
                seen_manifests.add(url)
                accepted.append(url)
                manifests.append({
                    "url": url,
                    "discovered_in": document_url,
                    "document_kind": kind,
                    "depth": depth,
                })
        document_row = {
            "url": document_url,
            "kind": kind,
            "depth": depth,
            "status": "scanned",
            "hls_found": len(accepted),
        }
        documents.append(document_row)

        if not enabled or depth >= max_depth:
            continue
        for linked_url, linked_kind in _extract_player_document_urls(document_text, document_url):
            if linked_url in seen_documents or any(linked_url == queued[0] for queued in queue):
                continue
            if len(documents) + len(queue) >= max_documents:
                break
            try:
                linked_text, final_url = _fetch_public_text(
                    linked_url,
                    timeout=timeout,
                    byte_limit=byte_limit,
                )
                queue.append((final_url, linked_text, depth + 1, linked_kind))
            except Exception as exc:
                documents.append({
                    "url": linked_url,
                    "kind": linked_kind,
                    "depth": depth + 1,
                    "status": "failed",
                    "error": str(exc)[:180],
                    "hls_found": 0,
                })
                if len(documents) >= max_documents:
                    break
    return manifests, documents


def _exact_allowlist_ids(config: dict[str, Any], scope: str = "all") -> list[str]:
    """Return deduplicated exact IDs for all, Philly, or national scope."""
    curation = config.get("curation") or {}
    local_settings = (config.get("discovery") or {}).get("philly_local_sources") or {}
    ids: list[str] = []
    seen: set[str] = set()
    keys = {
        "all": ("philly_allow", "national_allow"),
        "philly": ("philly_allow",),
        "national": ("national_allow",),
    }.get(scope)
    if keys is None:
        raise ValueError(f"Unknown allowlist scope: {scope}")
    for key in keys:
        patterns = curation.get(key, [])
        if scope == "philly" and key == "philly_allow" and local_settings.get("target_ids"):
            patterns = local_settings["target_ids"]
        for pattern in patterns:
            if "." not in (pattern or ""):
                continue
            target = _desired_id(pattern, curation)
            marker = canonical_tvg_id(target)
            if marker in seen:
                continue
            seen.add(marker)
            ids.append(target)
    return ids


def _private_connector_only_ids(config: dict[str, Any]) -> set[str]:
    policy = (config.get("discovery") or {}).get("public_search_policy") or {}
    return {
        canonical_tvg_id(str(item))
        for item in policy.get("private_connector_only_ids", [])
        if str(item).strip()
    }


def _target_ids_for_source_hunt(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
    scope: str = "all",
) -> list[str]:
    """Target only current candidate gaps plus channels missing last build.

    A source can disappear before health probing, so the previous deployed channel
    state is included alongside exact IDs that have no candidate in this build.
    The union means a recovered URL is actively hunted without feeding every
    unrelated upstream channel into the health checker.
    """
    in_scope = {
        canonical_tvg_id(item): item
        for item in _exact_allowlist_ids(config, scope=scope)
    }
    private_only = _private_connector_only_ids(config)
    in_scope = {key: value for key, value in in_scope.items() if key not in private_only}
    targets: dict[str, str] = {}
    coverage_entries = canonicalize_approved_sources(existing_entries, config)
    for row in build_allowlist_coverage(coverage_entries, config):
        if not row.get("found"):
            pattern = str(row.get("pattern") or "")
            if pattern and canonical_tvg_id(pattern) in in_scope:
                targets[canonical_tvg_id(pattern)] = pattern

    previous = _load_previous_channel_state(config)
    for record in (previous.get("channels") or {}).values():
        if not isinstance(record, dict) or record.get("present"):
            continue
        pattern = str(record.get("pattern") or "")
        if pattern and canonical_tvg_id(pattern) in in_scope:
            targets[canonical_tvg_id(pattern)] = pattern

    # Keep hunting a channel after its first recovery until at least one of its
    # current candidate URLs has established Stable/Backup history. A one-build
    # recovery is not enough evidence to stop looking for an independent or
    # more durable replacement.
    hunt_settings = (config.get("discovery") or {}).get("targeted_source_families") or {}
    if bool(hunt_settings.get("hunt_unstable_channels", False)):
        previous_stability = _load_previous_stream_stability(config)
        previous_streams = previous_stability.get("streams") or {}
        if previous_streams:
            established_classes = {"Stable", "Backup"}
            by_channel: dict[str, list[dict[str, Any]]] = defaultdict(list)
            for entry in coverage_entries:
                marker = canonical_tvg_id(str(entry.get("tvg_id") or ""))
                if marker in in_scope:
                    by_channel[marker].append(entry)

            for marker, pattern in in_scope.items():
                entries_for_channel = by_channel.get(marker) or []
                if not entries_for_channel:
                    continue
                has_established = False
                for entry in entries_for_channel:
                    old = previous_streams.get(_stream_stability_key(entry))
                    if isinstance(old, dict) and old.get("classification") in established_classes:
                        has_established = True
                        break
                if not has_established:
                    targets[marker] = pattern

    allow_order = _exact_allowlist_ids(config, scope=scope)
    priority_ids = list(hunt_settings.get("priority_target_ids") or [])
    secondary_ids = list(hunt_settings.get("secondary_target_ids") or [])
    preferred_order = list(dict.fromkeys(priority_ids + secondary_ids + allow_order))
    return [
        item
        for item in preferred_order
        if canonical_tvg_id(item) in targets
        and canonical_tvg_id(item) in in_scope
        and canonical_tvg_id(item) not in private_only
    ]


def _filter_entries_to_exact_targets(
    entries: list[dict[str, Any]],
    target_ids: list[str],
    config: dict[str, Any],
) -> list[dict[str, Any]]:
    """Canonicalize a source and retain only the requested exact logical IDs."""
    wanted = {canonical_tvg_id(item) for item in target_ids}
    canonicalized = canonicalize_approved_sources(entries, config)
    return [
        entry
        for entry in canonicalized
        if canonical_tvg_id(entry.get("tvg_id", "")) in wanted
    ]


def _parse_aria_markdown(text: str, source: dict[str, Any]) -> tuple[list[dict[str, Any]], int]:
    """Parse aria-tv's small US channel table without loading its global M3U."""
    entries: list[dict[str, Any]] = []
    scanned = 0
    row_re = re.compile(
        r"^\|\s*\d+\s*\|\s*(?P<name>[^|]+?)\s*\|\s*\[>\]\((?P<url>[^)]+)\)\s*\|(?P<rest>.*)$",
        re.I,
    )
    for raw in text.splitlines():
        match = row_re.match(raw.strip())
        if not match:
            continue
        scanned += 1
        name = html.unescape(match.group("name").strip())
        url = html.unescape(match.group("url").strip())
        rest = match.group("rest")
        fields = [field.strip() for field in rest.split("|") if field.strip()]
        status = fields[-1].lower() if fields else ""
        if status == "not-working" or not re.match(r"^https?://", url, re.I):
            continue
        logo_match = re.search(r'<img[^>]+src="([^"]+)"', rest, re.I)
        entries.append({
            "name_raw": name,
            "name": clean_display_name(name),
            "quality": extract_quality(name),
            "tvg_id": "",
            "logo": html.unescape(logo_match.group(1)) if logo_match else "",
            "group": "",
            "language": "English",
            "country": "US",
            "source": source.get("name", "aria-tv US"),
            "family": source.get("family", "aria-tv"),
            "priority": int(source.get("priority", 8)),
            "philly": bool(source.get("philly", False)),
            "url": url,
            "headers": {},
            "source_status": status or "unknown",
        })
    return entries, scanned


def discover_targeted_source_families(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]], list[str]]:
    """Scan high-value public source families only for remaining exact IDs."""
    settings = (config.get("discovery") or {}).get("targeted_source_families") or {}
    if not settings.get("enabled", True):
        return [], [], []

    target_ids = _target_ids_for_source_hunt(config, existing_entries, scope="national")
    sources = settings.get("sources") or []
    if not target_ids or not sources:
        return [], [{
            "kind": "targeted-source-summary",
            "scope": "national",
            "status": "nothing-to-scan",
            "target_ids": target_ids,
            "source_count": len(sources),
        }], target_ids

    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    timeout = int(settings.get("timeout_seconds", 20))
    byte_limit = int(settings.get("byte_limit", 8_000_000))
    default_cap = max(1, int(settings.get("max_candidates_per_target", 4)))

    for source in sources:
        url = str(source.get("url") or "").strip()
        fmt = str(source.get("format") or "m3u").lower()
        row: dict[str, Any] = {
            "kind": "targeted-source-family",
            "scope": "national",
            "name": source.get("name"),
            "family": source.get("family"),
            "url": url,
            "format": fmt,
            "target_count": len(target_ids),
            "target_ids": target_ids,
            "status": "pending",
            "records_scanned": 0,
            "candidates_found": 0,
            "matched_ids": [],
        }
        try:
            text, final_url = _fetch_public_text(
                url,
                timeout=int(source.get("timeout_seconds", timeout)),
                byte_limit=int(source.get("byte_limit", byte_limit)),
            )
            source_info = dict(source)
            if fmt == "aria-markdown":
                parsed, scanned = _parse_aria_markdown(text, source_info)
            else:
                parsed = parse_m3u(text, source_info)
                scanned = len(parsed)
            matched = _filter_entries_to_exact_targets(parsed, target_ids, config)
            accepted: list[dict[str, Any]] = []
            rejected_credential_urls = 0
            per_target: dict[str, int] = defaultdict(int)
            source_cap = max(1, int(source.get("max_candidates_per_target", default_cap)))
            allow_signed = bool(
                ((config.get("discovery") or {}).get("public_search_policy") or {}).get(
                    "allow_public_signed_urls", False
                )
            )
            for entry in matched:
                if not _is_clean_github_media_url(
                    str(entry.get("url") or ""),
                    allow_public_signed_urls=allow_signed,
                ):
                    rejected_credential_urls += 1
                    continue
                target_marker = canonical_tvg_id(entry.get("tvg_id", ""))
                if per_target[target_marker] >= source_cap:
                    continue
                marker = (target_marker, entry.get("url", ""))
                if marker in seen:
                    continue
                seen.add(marker)
                per_target[target_marker] += 1
                entry["targeted_scan"] = True
                discovered.append(entry)
                accepted.append(entry)

            row["status"] = "ok"
            row["final_url"] = final_url
            row["records_scanned"] = scanned
            row["matches_before_cap"] = len(matched)
            row["credential_urls_rejected"] = rejected_credential_urls
            row["candidates_found"] = len(accepted)
            row["matched_ids"] = sorted({entry.get("tvg_id") for entry in accepted if entry.get("tvg_id")})
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    rows.insert(0, {
        "kind": "targeted-source-summary",
        "scope": "national",
        "status": "complete",
        "target_ids": target_ids,
        "source_count": len(sources),
        "candidates_found": len(discovered),
    })
    log(
        f"Targeted source-family scan: {len(target_ids)} exact IDs across "
        f"{len(sources)} sources; {len(discovered)} candidates retained"
    )
    return discovered, rows, target_ids

            "url": url,
            "target_count": len(target_ids),
            "target_ids": target_ids,
            "status": "pending",
            "records_scanned": 0,
            "candidates_found": 0,
            "matched_ids": [],
        }
        if not target_ids:
            row["status"] = "nothing-to-scan"
            rows.append(row)
            continue
        try:
            text, final_url = _fetch_public_text(
                url,
                timeout=int(source.get("timeout_seconds", timeout)),
                byte_limit=int(source.get("byte_limit", byte_limit)),
            )
            parsed = parse_m3u(text, dict(source))
            matched = _filter_entries_to_exact_targets(parsed, target_ids, config)
            accepted: list[dict[str, Any]] = []
            per_target: dict[str, int] = defaultdict(int)
            source_cap = max(1, int(source.get("max_candidates_per_target", default_cap)))
            for entry in matched:
                target_marker = canonical_tvg_id(entry.get("tvg_id", ""))
                if per_target[target_marker] >= source_cap:
                    continue
                marker = (target_marker, entry.get("url", ""))
                if marker in seen:
                    continue
                seen.add(marker)
                per_target[target_marker] += 1
                entry["philly"] = True
                entry["philly_local_scan"] = True
                discovered.append(entry)
                accepted.append(entry)
            row["status"] = "ok"
            row["final_url"] = final_url
            row["records_scanned"] = len(parsed)
            row["matches_before_cap"] = len(matched)
            row["candidates_found"] = len(accepted)
            row["matched_ids"] = sorted({entry.get("tvg_id") for entry in accepted if entry.get("tvg_id")})
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    official_entries, official_rows = discover_official_page_streams(
        config,
        items=settings.get("official_pages") or [],
        target_ids=target_ids,
        row_kind="philly-official-page",
    )
    discovered.extend(official_entries)
    rows.extend(official_rows)
    rows.insert(0, {
        "kind": "philly-local-summary",
        "scope": "philly-local",
        "status": "complete" if target_ids else "nothing-to-scan",
        "target_ids": target_ids,
        "source_count": len(sources),
        "official_page_count": len(settings.get("official_pages") or []),
        "candidates_found": len(discovered),
    })
    log(
        f"Philadelphia local scan: {len(target_ids)} exact IDs across "
        f"{len(sources)} playlists and {len(settings.get('official_pages') or [])} official pages; "
        f"{len(discovered)} candidates retained"
    )
    return discovered, rows, target_ids


def discover_official_page_streams(
    config: dict[str, Any],
    items: list[dict[str, Any]] | None = None,
    target_ids: list[str] | None = None,
    row_kind: str = "official-page",
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    settings = items if items is not None else (config.get("discovery") or {}).get("official_pages") or []
    wanted = {canonical_tvg_id(item) for item in target_ids} if target_ids is not None else None

    for item in settings:
        page_url = item.get("page_url")
        tvg_id = item.get("tvg_id")
        if not page_url or not tvg_id:
            continue
        if wanted is not None and canonical_tvg_id(tvg_id) not in wanted:
            continue

        row: dict[str, Any] = {
            "kind": row_kind,
            "scope": "philly-local" if row_kind.startswith("philly-") else "shared",
            "name": item.get("name") or tvg_id,
            "tvg_id": tvg_id,
            "page_url": page_url,
            "status": "pending",
            "candidates": [],
        }
        try:
            page_text, final_url = _fetch_public_text(
                page_url,
                timeout=int(item.get("timeout_seconds", 12)),
            )
            manifest_rows, document_rows = scan_official_player_documents(
                page_text,
                final_url,
                config,
                item,
            )
            accepted = [candidate["url"] for candidate in manifest_rows]
            evidence_by_url = {candidate["url"]: candidate for candidate in manifest_rows}

            candidate_mode = str(item.get("candidate_mode") or "emit").lower()
            emitted = accepted[: int(item.get("max_candidates", 4))]
            if candidate_mode == "research-only":
                emitted = []

            for url in emitted:
                discovered.append({
                    "name_raw": item.get("name") or tvg_id,
                    "name": item.get("name") or tvg_id,
                    "quality": item.get("quality"),
                    "tvg_id": tvg_id,
                    "logo": item.get("logo", ""),
                    "group": item.get("group", "Local"),
                    "language": "English",
                    "country": "US",
                    "source": item.get("source") or f"Official page: {urllib.parse.urlsplit(page_url).netloc}",
                    "family": "official-page",
                    "priority": int(item.get("priority", -10)),
                    "philly": bool(item.get("philly", True)),
                    "url": url,
                    "headers": {},
                    "source_type": "official-public-page",
                    "content_kind": item.get("content_kind", "unknown"),
                    "content_note": item.get("content_note", ""),
                    "manifest_provenance": evidence_by_url.get(url),
                })
            row["candidates"] = accepted[: int(item.get("max_candidates", 4))]
            row["candidate_evidence"] = [
                evidence_by_url[url]
                for url in row["candidates"]
                if url in evidence_by_url
            ]
            row["documents_scanned"] = document_rows
            row["candidates_emitted"] = len(emitted)
            if candidate_mode == "research-only":
                row["status"] = (
                    "research-only-public-hls-found"
                    if row["candidates"]
                    else "research-only-no-public-hls"
                )
            else:
                row["status"] = "found" if row["candidates"] else "no-public-hls-in-page-source"
        except Exception as exc:
            row["status"] = "failed"
            row["error"] = str(exc)[:240]
        rows.append(row)

    return discovered, rows


def _extract_iptv_org_issue_stream_id(body: str) -> str:
    """Return the explicit IPTV-org template Stream ID, when present."""
    match = re.search(
        r"(?im)^###\s*Stream ID(?:\s*\([^\n)]*\))?\s*$\s*\n+\s*([^\n]+)",
        body or "",
    )
    if not match:
        return ""
    value = match.group(1).strip()
    if value.lower() in {"_no response_", "none", "n/a"}:
        return ""
    return value


def _iptv_org_issue_target(
    issue_title: str,
    issue_body: str,
    targets: list[dict[str, Any]],
    config: dict[str, Any],
) -> dict[str, Any] | None:
    """Resolve one issue to one exact target without substring alias leakage."""
    stream_id = _extract_iptv_org_issue_stream_id(issue_body)
    if stream_id:
        id_aliases = {
            canonical_tvg_id(source): target
            for source, target in ((config.get("curation") or {}).get("canonical_id_aliases") or {}).items()
        }
        marker = canonical_tvg_id(stream_id)
        marker = canonical_tvg_id(id_aliases.get(marker, stream_id))
        exact = [
            target
            for target in targets
            if canonical_tvg_id(str(target.get("tvg_id") or "")) == marker
        ]
        return exact[0] if len(exact) == 1 else None

    # Legacy issues without the template field may still be useful, but require
    # a whole-token/phrase match. This deliberately rejects tiny aliases such as
    # "ID" from matching words like "required".
    combined_norm = norm(f"{issue_title}\n{issue_body}")
    matches: list[dict[str, Any]] = []
    for target in targets:
        terms = [target.get("tvg_id", "")] + list(target.get("aliases") or [])
        hit = False
        for term in terms:
            term_norm = norm(str(term or "")).strip()
            if len(term_norm) < 3:
                continue
            if re.search(
                rf"(?<![a-z0-9]){re.escape(term_norm)}(?![a-z0-9])",
                combined_norm,
                flags=re.I,
            ):
                hit = True
                break
        if hit:
            matches.append(target)
    return matches[0] if len(matches) == 1 else None


def discover_iptv_org_passed_issues(
    config: dict[str, Any],
    target_ids: list[str] | None = None,
    scope: str = "shared",
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Inspect fresh IPTV-org add/remove issues for remaining exact IDs.

    Additions that passed IPTV-org's check can become candidates. Removal issues
    are diagnostics only and never reintroduce their broken URL. Every addition
    still goes through our own double HLS/media-segment probe.
    """
    settings = (config.get("discovery") or {}).get("iptv_org_issues") or {}
    if not settings.get("enabled", True):
        return [], []

    targets = settings.get("targets") or []
    if target_ids is not None:
        wanted = {canonical_tvg_id(item) for item in target_ids}
        targets = [
            target
            for target in targets
            if canonical_tvg_id(target.get("tvg_id", "")) in wanted
        ]
    if not targets:
        return [], []

    params = urllib.parse.urlencode({
        "state": "all",
        "sort": "updated",
        "direction": "desc",
        "per_page": int(settings.get("per_page", 100)),
    })
    api_url = "https://api.github.com/repos/iptv-org/iptv/issues?" + params
    rows: list[dict[str, Any]] = []
    discovered: list[dict[str, Any]] = []

    try:
        payload_text, _ = _fetch_public_text(api_url, timeout=int(settings.get("timeout_seconds", 20)))
        payload = json.loads(payload_text)
        issues = payload if isinstance(payload, list) else []
    except Exception as exc:
        return [], [{
            "kind": "iptv-org-issues",
            "scope": scope,
            "status": "failed",
            "error": str(exc)[:240],
            "api_url": api_url,
        }]

    for issue in issues:
        labels = {
            str(label.get("name") or "")
            for label in (issue.get("labels") or [])
            if isinstance(label, dict)
        }
        change_kind = "add" if "streams:add" in labels else "remove" if "streams:remove" in labels else ""
        if not change_kind or "check:passed" not in labels:
            continue
        if any(label.startswith("rejected:") for label in labels):
            continue

        title = issue.get("title") or ""
        body = issue.get("body") or ""
        combined = f"{title}\n{body}"

        matched_target = _iptv_org_issue_target(title, body, targets, config)
        if not matched_target:
            continue

        issue_number = issue.get("number")
        issue_url = issue.get("html_url")
        if change_kind == "remove":
            rows.append({
                "kind": "iptv-org-issue",
                "scope": scope,
                "status": "removal-signal",
                "change": "remove",
                "issue": issue_number,
                "issue_url": issue_url,
                "tvg_id": matched_target["tvg_id"],
                "name": matched_target.get("name"),
            })
            continue

        urls = _extract_public_hls_urls(combined)
        urls = [u for u in urls if ".m3u8" in u.lower()]
        if not urls:
            continue

        for url in urls[: int(settings.get("max_urls_per_issue", 3))]:
            discovered.append({
                "name_raw": matched_target.get("name") or matched_target["tvg_id"],
                "name": matched_target.get("name") or matched_target["tvg_id"],
                "quality": None,
                "tvg_id": matched_target["tvg_id"],
                "logo": "",
                "group": matched_target.get("group", ""),
                "language": "English",
                "country": "US",
                "source": f"IPTV-org check:passed issue #{issue_number}",
                "family": "iptv-org-issue",
                "priority": int(settings.get("priority", 6)),
                "philly": bool(matched_target.get("philly", False)),
                "url": url,
                "headers": {},
                "source_type": "public-third-party",
                "content_kind": matched_target.get("content_kind", "unknown"),
                "discovery_issue": issue_url,
            })
        rows.append({
            "kind": "iptv-org-issue",
            "scope": scope,
            "status": "candidate-found",
            "change": "add",
            "issue": issue_number,
            "issue_url": issue_url,
            "tvg_id": matched_target["tvg_id"],
            "name": matched_target.get("name"),
            "urls": urls[: int(settings.get("max_urls_per_issue", 3))],
        })

    return discovered, rows

def score_github_candidate_freshness(
    updated_at: Any,
    match_kind: str,
    now: dt.datetime | None = None,
) -> dict[str, Any]:
    """Score a GitHub lead without treating repository activity as stream health."""
    now = now or dt.datetime.now(UTC)
    updated = _parse_state_datetime(updated_at)
    age_days = None if updated is None else max(0.0, (now - updated).total_seconds() / 86400)
    identity_points = 50 if match_kind == "exact-id" else 35
    recency_points = 10 if age_days is None else max(0, round(50 - age_days / 7))
    score = min(100, identity_points + recency_points)
    if age_days is None:
        label = "unknown-age"
    elif age_days <= 30:
        label = "fresh"
    elif age_days <= 180:
        label = "recent"
    elif age_days <= 365:
        label = "aging"
    else:
        label = "stale"
    return {
        "score": score,
        "label": label,
        "match_kind": match_kind,
        "updated_at": iso_z(updated) if updated else None,
        "age_days": round(age_days, 1) if age_days is not None else None,
        "basis": "repository_pushed_at",
    }


def _github_raw_file_url(item: dict[str, Any]) -> str:
    html_url = str(item.get("html_url") or "")
    match = re.match(r"https://github\.com/([^/]+)/([^/]+)/blob/([^/]+)/(.*)", html_url)
    if not match:
        return ""
    owner, repository, revision, path = match.groups()
    return f"https://raw.githubusercontent.com/{owner}/{repository}/{revision}/{path}"


def _github_repository_updated_at(
    repository: dict[str, Any],
    timeout: int,
) -> Any:
    """Resolve repository activity from the core API, not code-search metadata.

    GitHub's code-search response normally omits ``pushed_at`` and ``updated_at``.
    Treating that omission as an unknown-age exact match made the freshness score
    largely cosmetic. Repository metadata uses GitHub's separate core-rate bucket
    and is cached by ``_fetch_public_text`` for the remainder of the build.
    """
    direct = repository.get("pushed_at") or repository.get("updated_at")
    if direct:
        return direct
    api_url = str(repository.get("url") or "").strip()
    if not api_url.startswith("https://api.github.com/repos/"):
        return None
    try:
        payload_text, _ = _fetch_public_text(api_url, timeout=timeout)
        payload = json.loads(payload_text)
        if isinstance(payload, dict):
            return payload.get("pushed_at") or payload.get("updated_at")
    except Exception:
        return None
    return None


def _github_rate_limit_wait(exc: urllib.error.HTTPError, maximum: float) -> float:
    """Return a bounded GitHub-request delay from Retry-After/reset headers."""
    retry_after = str(exc.headers.get("Retry-After") or "").strip()
    wait = 0.0
    try:
        wait = float(retry_after)
    except (TypeError, ValueError):
        reset = str(exc.headers.get("X-RateLimit-Reset") or "").strip()
        try:
            wait = max(0.0, float(reset) - time.time())
        except (TypeError, ValueError):
            wait = 0.0
    if wait <= 0:
        wait = 7.0
    return min(maximum, wait + 0.25)


def _is_clean_github_media_url(
    url: str,
    *,
    allow_public_signed_urls: bool = False,
) -> bool:
    """Reject account-shaped provider links copied into public playlists.

    A public GitHub file is not evidence that embedded provider credentials are
    intended for public reuse. Keep ordinary public HLS/CDN URLs, but discard
    URL userinfo, email-shaped path segments, explicit user/password query
    parameters, and the common Xtream ``/live/user/pass/id`` layout.
    """
    parsed = urllib.parse.urlsplit(url)
    if parsed.scheme not in {"http", "https"} or not parsed.hostname:
        return False
    if parsed.username is not None or parsed.password is not None:
        return False

    segments = [urllib.parse.unquote(part) for part in parsed.path.split("/") if part]
    if any(re.search(r"^[^/@\s]+@[^/@\s]+\.[^/@\s]+$", part) for part in segments):
        return False
    if len(segments) >= 4 and segments[0].lower() in {"live", "movie", "series"}:
        return False
    if len(segments) >= 4 and [part.lower() for part in segments[:2]] == ["api", "stream"]:
        return False

    # Static GitHub playlists frequently copy short-lived signed player URLs.
    # They may pass once, then expire and create exactly the churn this scanner
    # is meant to reduce, so do not promote them as durable discoveries.
    credential_keys = {
        "user", "username", "pass", "password", "pwd",
    }
    signed_keys = {
        "token", "access_token", "auth", "authorization", "jwt",
        "signature", "sig", "expires", "expiry", "exp", "policy",
        "key-pair-id", "hdnts", "hdnea",
    }
    query_keys = {key.lower() for key, _ in urllib.parse.parse_qsl(parsed.query)}
    if query_keys & credential_keys:
        return False
    if query_keys & signed_keys and not allow_public_signed_urls:
        return False
    return True


def discover_github_candidates(
    config: dict[str, Any],
    target_ids: list[str],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    """Search fresh public GitHub playlists for exact IDs and approved aliases."""
    settings = (config.get("discovery") or {}).get("github_candidate_search") or {}
    if not settings.get("enabled", True) or not target_ids:
        return [], []

    target_defs = {
        canonical_tvg_id(row.get("tvg_id", "")): row
        for row in ((config.get("discovery") or {}).get("iptv_org_issues") or {}).get("targets", [])
    }
    alias_map = (config.get("curation") or {}).get("approved_source_aliases") or {}
    max_targets = max(1, int(settings.get("max_targets_per_build", 6)))
    max_queries = max(1, int(settings.get("max_queries_per_build", 12)))
    per_page = max(1, min(20, int(settings.get("results_per_query", 5))))
    min_score = max(0, int(settings.get("min_freshness_score", 55)))
    byte_limit = max(100_000, int(settings.get("byte_limit", 6_000_000)))
    timeout = max(1, int(settings.get("timeout_seconds", 20)))
    min_interval = max(0.0, float(settings.get("min_query_interval_seconds", 0)))
    max_rate_wait = max(0.0, float(settings.get("max_rate_limit_wait_seconds", 15)))
    rate_retries = max(0, min(2, int(settings.get("rate_limit_retries", 1))))
    run_number = int(os.getenv("GITHUB_RUN_NUMBER", "0") or 0)
    ordered_targets = list(target_ids)
    if ordered_targets:
        offset = run_number % len(ordered_targets)
        ordered_targets = (ordered_targets[offset:] + ordered_targets[:offset])[:max_targets]

    discovered: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    parsed_file_cache: dict[str, list[dict[str, Any]]] = {}
    seen_candidates: set[tuple[str, str]] = set()
    query_count = 0
    paced_wait_seconds = 0.0
    rate_limit_retries = 0
    last_query_started: float | None = None
    target_found: dict[str, bool] = {target_id: False for target_id in ordered_targets}
    terms_by_target: dict[str, list[tuple[str, str]]] = {}
    for target_id in ordered_targets:
        target = target_defs.get(canonical_tvg_id(target_id)) or {
            "tvg_id": target_id,
            "name": target_id,
        }
        aliases = list(dict.fromkeys(
            list(target.get("aliases") or []) + list(alias_map.get(target_id, []))
        ))
        terms_by_target[target_id] = (
            [(target_id, "exact-id")]
            + [(alias, "approved-alias") for alias in aliases[:2]]
        )

    # Exact IDs get one fair pass before any channel consumes alias searches.
    # Alias pass 1 then pass 2 are also round-robin across the same targets.
    query_queue: list[tuple[str, str, str]] = []
    for term_index in range(3):
        for target_id in ordered_targets:
            terms = terms_by_target.get(target_id) or []
            if term_index < len(terms):
                term, match_kind = terms[term_index]
                query_queue.append((target_id, term, match_kind))

    for target_id, term, match_kind in query_queue:
        if query_count >= max_queries:
            break
        if match_kind != "exact-id" and target_found.get(target_id):
            continue
        query_count += 1
        query = f'"{term}" extension:m3u'
        api_url = "https://api.github.com/search/code?" + urllib.parse.urlencode({
            "q": query,
            "sort": "indexed",
            "order": "desc",
            "per_page": per_page,
        })
        search_row: dict[str, Any] = {
            "kind": "github-candidate-search",
            "scope": "shared",
            "status": "pending",
            "tvg_id": target_id,
            "term": term,
            "match_kind": match_kind,
            "api_url": api_url,
            "files_found": 0,
            "candidates_found": 0,
            "files": [],
        }
        try:
            if last_query_started is not None and min_interval > 0:
                delay = max(0.0, min_interval - (time.monotonic() - last_query_started))
                if delay > 0:
                    time.sleep(delay)
                    paced_wait_seconds += delay
            payload_text = ""
            for attempt in range(rate_retries + 1):
                last_query_started = time.monotonic()
                try:
                    payload_text, _ = _fetch_public_text(api_url, timeout=timeout)
                    break
                except urllib.error.HTTPError as exc:
                    if exc.code not in {403, 429} or attempt >= rate_retries:
                        raise
                    wait = _github_rate_limit_wait(exc, max_rate_wait)
                    if wait <= 0:
                        raise
                    search_row.setdefault("rate_limit_waits", []).append(round(wait, 2))
                    rate_limit_retries += 1
                    time.sleep(wait)
                    paced_wait_seconds += wait
            payload = json.loads(payload_text)
            items = payload.get("items") if isinstance(payload, dict) else []
            items = items if isinstance(items, list) else []
            search_row["files_found"] = len(items)
            for item in items:
                raw_url = _github_raw_file_url(item)
                if not raw_url:
                    continue
                repository = item.get("repository") if isinstance(item.get("repository"), dict) else {}
                repository_updated_at = _github_repository_updated_at(repository, timeout)
                freshness = score_github_candidate_freshness(
                    repository_updated_at,
                    match_kind,
                )
                file_row = {
                    "repository": repository.get("full_name"),
                    "path": item.get("path"),
                    "html_url": item.get("html_url"),
                    "raw_url": raw_url,
                    "freshness": freshness,
                    "records_scanned": 0,
                    "candidates_found": 0,
                    "credential_urls_rejected": 0,
                }
                if freshness["score"] < min_score:
                    file_row["status"] = "below-freshness-threshold"
                    search_row["files"].append(file_row)
                    continue
                if raw_url in parsed_file_cache:
                    parsed = parsed_file_cache[raw_url]
                else:
                    playlist_text, _ = _fetch_public_text(raw_url, timeout=timeout, byte_limit=byte_limit)
                    source = {
                        "name": f"GitHub candidate: {repository.get('full_name') or 'public repository'}",
                        "family": "github-candidate",
                        "priority": int(settings.get("priority", 7)),
                    }
                    parsed = parse_m3u(playlist_text, source)
                    parsed_file_cache[raw_url] = parsed
                matched = _filter_entries_to_exact_targets(parsed, [target_id], config)
                file_row["records_scanned"] = len(parsed)
                accepted = 0
                for entry in matched[: int(settings.get("max_candidates_per_file", 3))]:
                    allow_signed = bool(
                        ((config.get("discovery") or {}).get("public_search_policy") or {}).get(
                            "allow_public_signed_urls", False
                        )
                    )
                    if not _is_clean_github_media_url(
                        str(entry.get("url") or ""),
                        allow_public_signed_urls=allow_signed,
                    ):
                        file_row["credential_urls_rejected"] += 1
                        continue
                    marker = (canonical_tvg_id(target_id), str(entry.get("url") or ""))
                    if marker in seen_candidates:
                        continue
                    seen_candidates.add(marker)
                    entry.update({
                        "github_candidate": True,
                        "github_repository": repository.get("full_name"),
                        "github_path": item.get("path"),
                        "github_updated_at": freshness["updated_at"],
                        "freshness_score": freshness["score"],
                        "freshness_label": freshness["label"],
                        "research_url": item.get("html_url"),
                        "source_type": "public-github-playlist",
                    })
                    discovered.append(entry)
                    accepted += 1
                file_row["candidates_found"] = accepted
                file_row["status"] = "candidate-found" if accepted else "no-exact-match"
                search_row["files"].append(file_row)
                if accepted:
                    target_found[target_id] = True
                    search_row["candidates_found"] += accepted
            search_row["status"] = "ok"
        except Exception as exc:
            search_row["status"] = "failed"
            search_row["error"] = str(exc)[:240]
        rows.append(search_row)

    rows.insert(0, {
        "kind": "github-candidate-summary",
        "scope": "shared",
        "status": "complete",
        "target_ids": ordered_targets,
        "queries_run": query_count,
        "paced_wait_seconds": round(paced_wait_seconds, 2),
        "rate_limit_retries": rate_limit_retries,
        "min_query_interval_seconds": min_interval,
        "candidates_found": len(discovered),
        "min_freshness_score": min_score,
    })
    log(
        f"GitHub candidate search: {len(ordered_targets)} rotating exact IDs, "
        f"{query_count} queries, {len(discovered)} candidates retained"
    )
    return discovered, rows


def discover_dynamic_streams(
    config: dict[str, Any],
    existing_entries: list[dict[str, Any]],
) -> tuple[list[dict[str, Any]], list[dict[str, Any]]]:
    if not (config.get("discovery") or {}).get("enabled", True):
        return [], []

    targeted_entries, targeted_rows, national_target_ids = discover_targeted_source_families(
        config,
        existing_entries,
    )
    local_entries, local_rows, philly_target_ids = discover_philly_local_sources(
        config,
        existing_entries,
    )
    national_issue_entries, national_issue_rows = discover_iptv_org_passed_issues(
        config, national_target_ids, scope="national",
    )
    local_issue_entries, local_issue_rows = discover_iptv_org_passed_issues(
        config, philly_target_ids, scope="philly-local",
    )
    national_seed_entries, national_seed_rows = discover_seed_candidates(
        config, national_target_ids, scope="national",
    )
    local_seed_entries, local_seed_rows = discover_seed_candidates(
        config, philly_target_ids, scope="philly-local",
    )
    github_entries, github_rows = discover_github_candidates(
        config,
        list(dict.fromkeys(philly_target_ids + national_target_ids)),
    )

    combined: list[dict[str, Any]] = []
    seen: set[tuple[str, str]] = set()
    all_entries = (
        targeted_entries + local_entries + national_issue_entries + local_issue_entries
        + national_seed_entries + local_seed_entries + github_entries
    )
    for entry in all_entries:
        marker = (canonical_tvg_id(entry.get("tvg_id", "")), entry.get("url", ""))
        if marker in seen:
            continue
        seen.add(marker)
        combined.append(entry)

    log(
        f"Dynamic discovery: {len(targeted_entries)} targeted-family + "
        f"{len(local_entries)} Philadelphia-local + "
        f"{len(national_issue_entries) + len(local_issue_entries)} IPTV-org issue + "
        f"{len(national_seed_entries) + len(local_seed_entries)} seed + "
        f"{len(github_entries)} GitHub candidates"
    )
    return combined, (
        targeted_rows + local_rows + national_issue_rows + local_issue_rows
        + national_seed_rows + local_seed_rows + github_rows
    )


def _parse_hls_variants(text: str, base_url: str) -> list[dict[str, Any]]:
    """Parse master-playlist variants without downloading every rendition."""
    variants: list[dict[str, Any]] = []
    lines = [line.strip() for line in text.splitlines()]
    for index, line in enumerate(lines):
        if not line.upper().startswith("#EXT-X-STREAM-INF:"):
            continue
        uri = ""
        for following in lines[index + 1:]:
            if not following:
                continue
            if following.startswith("#"):
                continue
            uri = urllib.parse.urljoin(base_url, following)
            break
        if not uri:
            continue
        resolution = re.search(r"(?:^|,)\s*RESOLUTION=(\d+)[xX](\d+)", line, re.I)
        bandwidth = re.search(r"(?:^|,)\s*(?:AVERAGE-)?BANDWIDTH=(\d+)", line, re.I)
        width = int(resolution.group(1)) if resolution else 0
        height = int(resolution.group(2)) if resolution else 0
        variants.append({
            "url": uri,
            "width": width,
            "height": height,
            "bandwidth": int(bandwidth.group(1)) if bandwidth else 0,
        })
    return variants


def _repeated_hls_segment_sequence(segments: list[str], max_width: int = 3) -> dict[str, Any] | None:
    """Detect immediately repeated URI sequences in an HLS media playlist."""
    clean = [segment.strip() for segment in segments if segment.strip()]
    if len(clean) < 2:
        return None
    max_width = min(max_width, len(clean) // 2)
    for width in range(1, max_width + 1):
        for start in range(0, len(clean) - (2 * width) + 1):
            first = clean[start:start + width]
            second = clean[start + width:start + (2 * width)]
            if first == second:
                return {
                    "start_index": start,
                    "sequence_width": width,
                    "sequence": first,
                }
    return None


def _probe_stream(stream: dict[str, Any], timeout: int, verify_segment: bool) -> dict[str, Any]:
    url = stream.get("url") or ""
    if not re.match(r"^https?://", url, re.I):
        return {"status": "not-tested", "detail": "Non-HTTP stream"}

    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "*/*",
        "Accept-Encoding": "identity",
    }
    headers.update(stream.get("headers") or {})
    started = time.monotonic()

    def fetch(target: str, byte_limit: int = 262144) -> tuple[int, str, str, bytes]:
        req = urllib.request.Request(target, headers=headers)
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            status = int(getattr(resp, "status", 200) or 200)
            final_url = resp.geturl()
            content_type = (resp.headers.get("Content-Type") or "").lower()
            body = resp.read(byte_limit)
            return status, final_url, content_type, body

    try:
        status, final_url, content_type, body = fetch(url)
        elapsed_ms = round((time.monotonic() - started) * 1000)
        text = body.decode("utf-8", errors="ignore")
        is_hls = (
            "#EXTM3U" in text
            or ".m3u8" in urllib.parse.urlsplit(final_url).path.lower()
            or "mpegurl" in content_type
        )

        if is_hls:
            if "#EXTM3U" not in text:
                return {
                    "status": "failed",
                    "detail": "HLS URL did not return a valid playlist",
                    "http_status": status,
                    "final_url": final_url,
                    "latency_ms": elapsed_ms,
                }

            media_text = text
            media_url = final_url
            measurement: dict[str, Any] = {}
            variants = _parse_hls_variants(text, final_url)
            if variants:
                measured = max(variants, key=lambda item: (item["height"], item["width"], item["bandwidth"]))
                known_resolutions = sorted(
                    {
                        (item["width"], item["height"])
                        for item in variants
                        if item["width"] > 0 and item["height"] > 0
                    },
                    key=lambda item: (item[1], item[0]),
                    reverse=True,
                )
                if measured["height"] > 0:
                    measurement = {
                        "measured_width": measured["width"],
                        "measured_height": measured["height"],
                        "measured_quality": quality_label_for_height(measured["height"]),
                        "quality_source": "hls-master",
                        "available_resolutions": [f"{width}x{height}" for width, height in known_resolutions],
                    }
                candidate = measured["url"]
                child_status, child_url, _child_type, child_body = fetch(candidate)
                media_text = child_body.decode("utf-8", errors="ignore")
                media_url = child_url
                if "#EXTM3U" not in media_text:
                    return {
                        "status": "failed",
                        "detail": "Master playlist child was invalid",
                        "http_status": child_status,
                        "final_url": child_url,
                        "latency_ms": round((time.monotonic() - started) * 1000),
                    }

            segments = [
                line.strip()
                for line in media_text.splitlines()
                if line.strip() and not line.startswith("#")
            ]
            media_sequence_match = re.search(
                r"^#EXT-X-MEDIA-SEQUENCE:\s*(\d+)\s*$",
                media_text,
                re.I | re.M,
            )
            media_sequence = int(media_sequence_match.group(1)) if media_sequence_match else None
            hls_is_live = "#EXT-X-ENDLIST" not in media_text.upper()
            tail_material = "\n".join(segments[-3:]).encode("utf-8") if segments else b""
            hls_tail_sha256 = hashlib.sha256(tail_material).hexdigest() if tail_material else None

            if verify_segment:
                # Reusing the same URI with EXT-X-BYTERANGE is valid HLS, so
                # only apply URI-repeat detection to ordinary segment lists.
                repeated = None
                if "#EXT-X-BYTERANGE" not in media_text.upper():
                    repeated = _repeated_hls_segment_sequence(segments)
                if repeated is not None:
                    return {
                        "status": "failed",
                        "detail": (
                            "Repeated HLS segment sequence detected "
                            f"(width={repeated['sequence_width']})"
                        ),
                        "http_status": status,
                        "final_url": media_url,
                        "latency_ms": round((time.monotonic() - started) * 1000),
                        "continuity_failure": "repeated-hls-segments",
                    }
                if segments:
                    segment_url = urllib.parse.urljoin(media_url, segments[0])
                    seg_headers = dict(headers)
                    seg_headers["Range"] = "bytes=0-1023"
                    req = urllib.request.Request(segment_url, headers=seg_headers)
                    with urllib.request.urlopen(req, timeout=timeout) as resp:
                        resp.read(1)

            return {
                "status": "ok",
                "detail": "Valid HLS playlist" + (" + segment" if verify_segment else ""),
                "http_status": status,
                "final_url": media_url,
                "latency_ms": round((time.monotonic() - started) * 1000),
                "hls_is_live": hls_is_live,
                "hls_media_sequence": media_sequence,
                "hls_tail_sha256": hls_tail_sha256,
                "hls_segment_count": len(segments),
                **measurement,
            }

        if "text/html" in content_type:
            return {
                "status": "failed",
                "detail": "Returned HTML instead of media",
                "http_status": status,
                "final_url": final_url,
                "latency_ms": elapsed_ms,
            }

        sample = body[:65536]
        sample_hash = hashlib.sha256(sample).hexdigest() if len(sample) >= 32768 else None
        return {
            "status": "ok",
            "detail": "Reachable direct media URL",
            "http_status": status,
            "final_url": final_url,
            "latency_ms": elapsed_ms,
            "media_sample_sha256": sample_hash,
            "media_sample_bytes": len(sample),
        }

    except urllib.error.HTTPError as exc:
        return {
            "status": "failed",
            "detail": f"HTTP {exc.code}",
            "http_status": exc.code,
            "latency_ms": round((time.monotonic() - started) * 1000),
        }
    except Exception as exc:
        return {
            "status": "failed",
            "detail": str(exc)[:180],
            "latency_ms": round((time.monotonic() - started) * 1000),
        }


def _framemd5_hashes(text: str) -> list[str]:
    hashes: list[str] = []
    for raw in (text or "").splitlines():
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        parts = [part.strip() for part in line.split(",")]
        if len(parts) >= 6 and re.fullmatch(r"[0-9a-fA-F]{32}", parts[-1]):
            hashes.append(parts[-1].lower())
    return hashes


def _find_repeated_frame_clip(hashes: list[str], fps: int, min_seconds: int = 2, max_seconds: int = 4) -> dict[str, Any] | None:
    if fps <= 0:
        return None
    min_width = max(1, int(min_seconds * fps))
    max_width = min(int(max_seconds * fps), len(hashes) // 2)
    for width in range(min_width, max_width + 1):
        for start in range(0, len(hashes) - (2 * width) + 1):
            first = hashes[start:start + width]
            second = hashes[start + width:start + (2 * width)]
            if len(set(first)) < 3:
                continue
            if first == second:
                return {"start_frame": start, "width_frames": width, "clip_seconds": round(width / fps, 2), "unique_frames": len(set(first))}
    return None


def _validate_direct_media_survival(stream: dict[str, Any], settings: dict[str, Any]) -> dict[str, Any]:
    """Require repeated successful decoded connections for direct media."""
    if not bool(settings.get("enabled", False)):
        return {"status": "not-tested", "detail": "Direct-media survival disabled"}

    ffmpeg = shutil.which("ffmpeg")
    if not ffmpeg:
        return {
            "status": "failed",
            "detail": "FFmpeg unavailable for direct-media decoder validation",
            "continuity_failure": "direct-media-decoder-unavailable",
        }

    url = str(stream.get("url") or "")
    duration = max(1, int(settings.get("duration_seconds", 12)))
    min_decoded_seconds = max(
        1,
        min(duration, int(settings.get("min_decoded_seconds", 10))),
    )
    fps = max(1, int(settings.get("fps", 2)))
    timeout = max(duration + 3, int(settings.get("timeout_seconds", duration + 8)))
    attempts_required = max(1, int(settings.get("attempts", 3)))
    required_passes = max(
        1,
        min(attempts_required, int(settings.get("required_passes", 2))),
    )
    min_frames = max(1, min_decoded_seconds * fps)
    detect_repeats = bool(settings.get("detect_repeated_clips", True))
    repeat_min_seconds = max(1, int(settings.get("repeat_min_seconds", 2)))
    repeat_max_seconds = max(
        repeat_min_seconds,
        int(settings.get("repeat_max_seconds", 4)),
    )

    headers = stream.get("headers") or {}
    base_command = [
        ffmpeg,
        "-hide_banner",
        "-loglevel",
        "error",
        "-nostdin",
        "-user_agent",
        USER_AGENT,
        "-rw_timeout",
        "10000000",
    ]
    if headers:
        base_command.extend([
            "-headers",
            "".join(f"{key}: {value}\\r\\n" for key, value in headers.items()),
        ])
    base_command.extend([
        "-i",
        url,
        "-t",
        str(duration),
        "-map",
        "0:v:0",
        "-vf",
        f"fps={fps},scale=160:-2",
        "-an",
        "-f",
        "framemd5",
        "-",
    ])

    attempts: list[dict[str, Any]] = []
    passed_attempts = 0

    for attempt_number in range(1, attempts_required + 1):
        started = time.monotonic()
        try:
            completed = subprocess.run(
                base_command,
                stdout=subprocess.PIPE,
                stderr=subprocess.PIPE,
                text=True,
                timeout=timeout,
                check=False,
            )
            hashes = _framemd5_hashes(completed.stdout)
            decoded_frames = len(hashes)
            decoded_seconds = round(decoded_frames / fps, 2)
            repeat = (
                _find_repeated_frame_clip(
                    hashes,
                    fps,
                    repeat_min_seconds,
                    repeat_max_seconds,
                )
                if detect_repeats
                else None
            )
            attempt = {
                "attempt": attempt_number,
                "decoded_frames": decoded_frames,
                "decoded_seconds": decoded_seconds,
                "elapsed_seconds": round(time.monotonic() - started, 2),
                "ffmpeg_exit_code": completed.returncode,
                "repeat": repeat,
                "passed": False,
            }

            if repeat is not None:
                attempts.append(attempt)
                survival = {
                    "duration_seconds": duration,
                    "min_decoded_seconds": min_decoded_seconds,
                    "fps": fps,
                    "attempts_required": attempts_required,
                    "required_passes": required_passes,
                    "attempts_completed": len(attempts),
                    "passed_attempts": passed_attempts,
                    "attempts": attempts,
                    "passed": False,
                }
                return {
                    "status": "failed",
                    "detail": (
                        "Repeated direct-media video clip detected "
                        f"(~{repeat['clip_seconds']}s immediately replayed)"
                    ),
                    "continuity_failure": "direct-media-repeated-clip",
                    "direct_media_survival": survival,
                }

            if decoded_frames >= min_frames:
                attempt["passed"] = True
                passed_attempts += 1

            attempts.append(attempt)
        except subprocess.TimeoutExpired:
            attempts.append({
                "attempt": attempt_number,
                "decoded_frames": 0,
                "decoded_seconds": 0.0,
                "elapsed_seconds": round(time.monotonic() - started, 2),
                "ffmpeg_exit_code": None,
                "repeat": None,
                "passed": False,
                "failure": "timeout",
            })
        except Exception as exc:
            attempts.append({
                "attempt": attempt_number,
                "decoded_frames": 0,
                "decoded_seconds": 0.0,
                "elapsed_seconds": round(time.monotonic() - started, 2),
                "ffmpeg_exit_code": None,
                "repeat": None,
                "passed": False,
                "failure": str(exc)[:160],
            })

        if passed_attempts >= required_passes:
            break

        attempts_left = attempts_required - attempt_number
        if passed_attempts + attempts_left < required_passes:
            break

    survival = {
        "duration_seconds": duration,
        "min_decoded_seconds": min_decoded_seconds,
        "fps": fps,
        "attempts_required": attempts_required,
        "required_passes": required_passes,
        "attempts_completed": len(attempts),
        "passed_attempts": passed_attempts,
        "attempts": attempts,
        "passed": passed_attempts >= required_passes,
    }

    if survival["passed"]:
        return {
            "status": "ok",
            "detail": (
                f"Passed {passed_attempts}/{len(attempts)} direct-media "
                f"decoder attempts (requires {required_passes})"
            ),
            "direct_media_survival": survival,
        }

    best_seconds = max(
        (float(attempt.get("decoded_seconds", 0.0)) for attempt in attempts),
        default=0.0,
    )
    return {
        "status": "failed",
        "detail": (
            f"Direct media passed {passed_attempts}/{len(attempts)} decoder attempts; "
            f"requires {required_passes} (best decode {best_seconds:.1f}s)"
        ),
        "continuity_failure": "direct-media-insufficient-survival",
        "direct_media_survival": survival,
    }


def _validate_primary_playback_stream(
    stream: dict[str, Any],
    settings: dict[str, Any],
) -> dict[str, Any]:
    """Decode one selected stream end-to-end, regardless of HLS/direct transport."""
    direct_survival = ((stream.get("health") or {}).get("direct_media_survival") or {})
    if direct_survival.get("passed") is True:
        return {
            "status": "ok",
            "detail": "Already passed stronger 2-of-3 direct-media decoder gate",
            "primary_playback_qa": {
                "reused_direct_media_survival": True,
                **direct_survival,
            },
        }

    one_attempt = {
        **settings,
        "enabled": True,
        "attempts": 1,
        "required_passes": 1,
    }
    result = _validate_direct_media_survival(stream, one_attempt)
    survival = result.get("direct_media_survival") or {}
    if result.get("status") == "ok":
        return {
            "status": "ok",
            "detail": result.get("detail") or "Primary playback decode passed",
            "primary_playback_qa": survival,
        }

    failure = result.get("continuity_failure")
    mapped_failure = (
        "primary-playback-repeated-clip"
        if failure == "direct-media-repeated-clip"
        else "primary-playback-decode-failed"
    )
    return {
        "status": "failed",
        "detail": result.get("detail") or "Primary playback decode failed",
        "continuity_failure": mapped_failure,
        "primary_playback_qa": survival,
    }


def apply_primary_playback_qa(
    channels: list[dict[str, Any]],
    config: dict[str, Any],
    *,
    enabled: bool = True,
) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    """Decode selected Primaries and automatically promote a passing backup."""
    settings = (config.get("stream_health") or {}).get("primary_playback_qa") or {}
    enabled = enabled and bool(settings.get("enabled", False))
    built_at = iso_z(dt.datetime.now(UTC))
    if not enabled:
        return channels, {
            "version": 1,
            "enabled": False,
            "built_at": built_at,
            "summary": {
                "channels_tested": 0,
                "primary_passed": 0,
                "backup_promoted": 0,
                "channels_dropped": 0,
            },
            "channels": [],
        }

    max_workers = max(1, int(settings.get("max_workers", 10)))

    def inspect_channel(index: int, channel: dict[str, Any]) -> tuple[int, dict[str, Any]]:
        attempts: list[dict[str, Any]] = []
        streams = list(channel.get("streams") or [])
        passing_index: int | None = None
        passing_qa: dict[str, Any] | None = None

        for stream_index, stream in enumerate(streams):
            result = _validate_primary_playback_stream(stream, settings)
            attempts.append({
                "stream_index": stream_index,
                "role_before": "Primary" if stream_index == 0 else f"Backup {stream_index}",
                "source": stream.get("source"),
                "family": stream.get("family"),
                "provider_host": stream_provider_host(stream),
                "url": stream.get("url"),
                "status": result.get("status"),
                "detail": result.get("detail"),
                "continuity_failure": result.get("continuity_failure"),
                "qa": result.get("primary_playback_qa"),
            })
            if result.get("status") == "ok":
                passing_index = stream_index
                passing_qa = result.get("primary_playback_qa") or {}
                break

        return index, {
            "channel_id": channel.get("id"),
            "tvg_id": channel.get("tvg_id"),
            "name": channel.get("name"),
            "passing_index": passing_index,
            "passing_qa": passing_qa,
            "attempts": attempts,
        }

    results_by_index: dict[int, dict[str, Any]] = {}
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        futures = {
            executor.submit(inspect_channel, index, channel): index
            for index, channel in enumerate(channels)
        }
        for future in concurrent.futures.as_completed(futures):
            index = futures[future]
            try:
                result_index, result = future.result()
                results_by_index[result_index] = result
            except Exception as exc:
                channel = channels[index]
                results_by_index[index] = {
                    "channel_id": channel.get("id"),
                    "tvg_id": channel.get("tvg_id"),
                    "name": channel.get("name"),
                    "passing_index": None,
                    "passing_qa": None,
                    "attempts": [],
                    "error": str(exc)[:200],
                }

    kept_channels: list[dict[str, Any]] = []
    rows: list[dict[str, Any]] = []
    primary_passed = 0
    backup_promoted = 0
    dropped = 0

    for index, channel in enumerate(channels):
        row = results_by_index[index]
        passing_index = row.get("passing_index")
        if passing_index is None:
            row["action"] = "dropped-channel"
            dropped += 1
            rows.append(row)
            continue

        passing_index = int(passing_index)
        updated = dict(channel)
        streams = [dict(stream) for stream in (channel.get("streams") or [])]
        chosen = streams[passing_index]
        chosen["primary_playback_qa"] = row.get("passing_qa") or {}
        if passing_index == 0:
            primary_passed += 1
            row["action"] = "kept-primary"
            updated["streams"] = streams
        else:
            backup_promoted += 1
            row["action"] = "promoted-backup"
            row["failed_streams_removed"] = passing_index
            # Failed streams before the first passing backup are not published.
            updated["streams"] = [chosen] + streams[passing_index + 1:]
        kept_channels.append(updated)
        rows.append(row)

    report = {
        "version": 1,
        "enabled": True,
        "built_at": built_at,
        "settings": {
            "duration_seconds": int(settings.get("duration_seconds", 10)),
            "min_decoded_seconds": int(settings.get("min_decoded_seconds", 8)),
            "fps": int(settings.get("fps", 2)),
            "max_workers": max_workers,
        },
        "summary": {
            "channels_tested": len(channels),
            "primary_passed": primary_passed,
            "backup_promoted": backup_promoted,
            "channels_dropped": dropped,
            "channels_published": len(kept_channels),
        },
        "channels": rows,
    }
    log(
        "Primary playback QA: "
        f"tested={len(channels):,}, primary_passed={primary_passed:,}, "
        f"backup_promoted={backup_promoted:,}, dropped={dropped:,}"
    )
    return kept_channels, report


def _parse_resolver_vlc_headers(command: str) -> dict[str, str]:
    headers: dict[str, str] = {}
    ua_match = re.search(r":http-user-agent='([^']+)'", command or "")
    ref_match = re.search(r":http-referrer='([^']+)'", command or "")
    if ua_match:
        headers["User-Agent"] = ua_match.group(1)
    if ref_match:
        headers["Referer"] = ref_match.group(1)
    return headers


def discover_daddylive_resolver_streams(
    config: dict[str, Any],
) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    """Resolve fresh signed DaddyLive HLS URLs from the pinned local resolver."""
    settings = ((config.get("discovery") or {}).get("daddylive_resolver") or {})
    built_at = iso_z(dt.datetime.now(UTC))
    if not bool(settings.get("enabled", False)):
        return [], {
            "kind": "daddylive-resolver-summary",
            "enabled": False,
            "status": "disabled",
            "built_at": built_at,
            "targets": 0,
            "resolved": 0,
            "candidates_found": 0,
            "results": [],
        }

    base_url = (
        os.getenv(str(settings.get("base_url_env") or "DADDYLIVE_RESOLVER_URL"))
        or str(settings.get("base_url") or "")
    ).strip().rstrip("/")
    targets = list(settings.get("targets") or [])
    servers = [
        str(server).strip()
        for server in settings.get("servers", ["stream", "cast"])
        if str(server).strip()
    ] or ["stream"]
    timeout = max(3, int(settings.get("timeout_seconds", 18)))

    if not base_url:
        return [], {
            "kind": "daddylive-resolver-summary",
            "enabled": True,
            "status": "unavailable",
            "detail": "Resolver base URL is not configured",
            "built_at": built_at,
            "targets": len(targets),
            "resolved": 0,
            "candidates_found": 0,
            "results": [],
        }

    entries: list[dict[str, Any]] = []
    results: list[dict[str, Any]] = []

    for target in targets:
        tvg_id = str(target.get("tvg_id") or "").strip()
        channel_id = target.get("channel_id")
        if not tvg_id or channel_id is None:
            continue

        row: dict[str, Any] = {
            "tvg_id": tvg_id,
            "name": target.get("name") or tvg_id,
            "channel_id": int(channel_id),
            "attempts": [],
            "status": "failed",
        }

        for server in servers:
            query = urllib.parse.urlencode({
                "channel": int(channel_id),
                "server": server,
            })
            resolve_url = f"{base_url}/api/resolve?{query}"
            started = time.monotonic()
            try:
                req = urllib.request.Request(
                    resolve_url,
                    headers={
                        "User-Agent": USER_AGENT,
                        "Accept": "application/json",
                    },
                )
                with urllib.request.urlopen(req, timeout=timeout) as response:
                    payload = json.loads(response.read(1_000_000).decode("utf-8"))

                direct = str(payload.get("direct") or "").strip()
                headers = _parse_resolver_vlc_headers(str(payload.get("vlc") or ""))
                attempt = {
                    "server": server,
                    "status": "ok" if direct.startswith(("http://", "https://")) else "failed",
                    "elapsed_seconds": round(time.monotonic() - started, 2),
                    "expires_at": payload.get("expiresAt"),
                    "is_hls": payload.get("isHls"),
                    "has_user_agent": bool(headers.get("User-Agent")),
                    "has_referer": bool(headers.get("Referer")),
                }
                row["attempts"].append(attempt)
                if attempt["status"] != "ok":
                    continue

                entry = {
                    "name": target.get("name") or tvg_id,
                    "name_raw": target.get("name") or tvg_id,
                    "tvg_id": tvg_id,
                    "url": direct,
                    "headers": headers,
                    "source": settings.get("source_name") or "DaddyLive maintained resolver",
                    "family": settings.get("family") or "daddylive-resolver",
                    "priority": int(target.get("priority", settings.get("priority", 6))),
                    "philly": bool(target.get("philly", False)),
                    "group": target.get("group") or "Live TV",
                    "quality": target.get("quality") or "Unknown",
                    "resolver_channel_id": int(channel_id),
                    "resolver_server": server,
                    "expires_at": payload.get("expiresAt"),
                    "ephemeral_signed": True,
                    "stability_identity": f"daddylive-resolver:{int(channel_id)}:{server}",
                }
                entries.append(entry)
                row["status"] = "ok"
                row["server"] = server
                row["expires_at"] = payload.get("expiresAt")
                row["has_user_agent"] = bool(headers.get("User-Agent"))
                row["has_referer"] = bool(headers.get("Referer"))
                break
            except urllib.error.HTTPError as exc:
                row["attempts"].append({
                    "server": server,
                    "status": "failed",
                    "http_status": exc.code,
                    "elapsed_seconds": round(time.monotonic() - started, 2),
                    "error": f"HTTP {exc.code}",
                })
            except Exception as exc:
                row["attempts"].append({
                    "server": server,
                    "status": "failed",
                    "elapsed_seconds": round(time.monotonic() - started, 2),
                    "error": str(exc)[:200],
                })

        results.append(row)

    report = {
        "kind": "daddylive-resolver-summary",
        "enabled": True,
        "status": "ok",
        "built_at": built_at,
        "base_url": base_url,
        "targets": len(targets),
        "resolved": sum(1 for row in results if row.get("status") == "ok"),
        "candidates_found": len(entries),
        "results": results,
    }
    log(
        "DaddyLive resolver: "
        f"resolved={report['resolved']:,}/{report['targets']:,}, "
        f"candidates={len(entries):,}"
    )
    return entries, report


def _stream_stability_key(entry: dict[str, Any]) -> str:
    """Stable identity for a logical source or one playback URL + headers."""
    stable_identity = str(entry.get("stability_identity") or "").strip()
    if stable_identity:
        payload = {
            "stability_identity": stable_identity,
            "tvg_id": canonical_tvg_id(str(entry.get("tvg_id") or "")),
        }
    else:
        payload = {
            "url": entry.get("url") or "",
            "headers": sorted((entry.get("headers") or {}).items()),
        }
    encoded = json.dumps(payload, sort_keys=True, separators=(",", ":"))
    return hashlib.sha1(encoded.encode("utf-8")).hexdigest()


def _classify_stream_stability(history: list[bool], history_size: int) -> dict[str, Any]:
    history = [bool(x) for x in history][-history_size:]
    passes = sum(1 for x in history if x)
    samples = len(history)
    if samples >= history_size and passes == history_size:
        classification = "Stable"
        rank = 0
    elif passes >= 3:
        classification = "Backup"
        rank = 1
    elif passes >= 1:
        classification = "Quarantine"
        rank = 2
    else:
        classification = "Dead"
        rank = 3
    return {
        "classification": classification,
        "rank": rank,
        "passes": passes,
        "samples": samples,
        "history_size": history_size,
        "history": history,
    }


def _load_previous_stream_stability(config: dict[str, Any]) -> dict[str, Any]:
    settings = config.get("stream_stability") or {}
    if not settings.get("enabled", True):
        return {}

    explicit = (settings.get("previous_state_url") or "").strip()
    public_base = derive_public_base()
    if explicit:
        state_url = explicit
    elif public_base:
        state_url = f"{public_base}/diagnostics/stream-stability.json"
    else:
        return {}

    cache_bust = os.getenv("GITHUB_RUN_ID") or str(int(time.time()))
    separator = "&" if "?" in state_url else "?"
    request_url = f"{state_url}{separator}build={urllib.parse.quote(cache_bust)}"
    req = urllib.request.Request(
        request_url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "application/json",
            "Accept-Encoding": "identity",
            "Cache-Control": "no-cache",
        },
    )
    try:
        with urllib.request.urlopen(
            req,
            timeout=int(settings.get("load_timeout_seconds", 8)),
        ) as resp:
            payload = json.loads(resp.read(8_000_000).decode("utf-8"))
        if isinstance(payload, dict) and isinstance(payload.get("streams"), dict):
            log(
                "Stream stability: loaded "
                f"{len(payload['streams']):,} prior URL histories"
            )
            return payload
    except urllib.error.HTTPError as exc:
        if exc.code != 404:
            log(f"WARNING: could not load prior stream stability state: HTTP {exc.code}")
    except Exception as exc:
        log(f"WARNING: could not load prior stream stability state: {str(exc)[:160]}")
    return {}


def apply_stream_stability(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
    previous: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """Persist and score URL health across builds.

    Current playback health remains mandatory. Stability only changes ordering
    among candidates that passed this build, so an old successful URL is never
    exposed while it is currently failing.
    """
    settings = config.get("stream_stability") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    history_size = max(1, int(settings.get("history_size", 5)))
    retention_builds = max(history_size, int(settings.get("retention_builds", 20)))
    cooldown_hours = [
        max(1, int(value))
        for value in settings.get("dead_source_cooldown_hours", [1, 6, 24])
    ] or [1, 6, 24]
    built_at = iso_z(dt.datetime.now(UTC))

    if not enabled:
        for entry in entries:
            entry["stability"] = {
                "classification": "Unknown",
                "rank": 2,
                "passes": 0,
                "samples": 0,
                "history_size": history_size,
                "history": [],
            }
        return {
            "version": 1,
            "enabled": False,
            "history_size": history_size,
            "updated_at": built_at,
            "streams": {},
            "current_counts": {},
        }

    previous = previous if previous is not None else _load_previous_stream_stability(config)
    previous_streams = previous.get("streams") or {}
    state_streams: dict[str, dict[str, Any]] = {}

    # Keep recently unseen candidates so a temporary source-playlist outage does
    # not erase four builds of useful history.
    for key, old in previous_streams.items():
        if not isinstance(old, dict):
            continue
        missed = int(old.get("missed_builds", 0)) + 1
        if missed > retention_builds:
            continue
        copied = dict(old)
        copied["missed_builds"] = missed
        state_streams[key] = copied

    current_by_key: dict[str, dict[str, Any]] = {}
    linked_by_key: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for entry in entries:
        key = _stream_stability_key(entry)
        current_by_key.setdefault(key, entry)
        linked_by_key[key].append(entry)

    current_counts: dict[str, int] = defaultdict(int)
    for key, entry in current_by_key.items():
        old = previous_streams.get(key) if isinstance(previous_streams.get(key), dict) else {}
        history = [bool(x) for x in (old.get("history") or [])][-history_size:]
        health = entry.get("health") or {}
        health_status = health.get("status")
        skipped_cooldown = health_status == "cooldown"
        passed = health_status == "ok"
        if not skipped_cooldown:
            history = (history + [passed])[-history_size:]
        score = _classify_stream_stability(history, history_size)
        score["last_result"] = "cooldown" if skipped_cooldown else "pass" if passed else "fail"
        score["last_seen"] = built_at

        old_failures = int(old.get("consecutive_failures", 0) or 0)
        if not old_failures and old.get("last_result") == "fail":
            for result in reversed(old.get("history") or []):
                if result:
                    break
                old_failures += 1
        if skipped_cooldown:
            consecutive_failures = old_failures
            next_probe_at = health.get("cooldown_until") or old.get("next_probe_at")
            last_probed_at = old.get("last_probed_at") or old.get("last_seen")
            last_failure_detail = old.get("last_failure_detail")
            last_failure_http_status = old.get("last_failure_http_status")
        elif passed:
            consecutive_failures = 0
            next_probe_at = None
            last_probed_at = built_at
            last_failure_detail = None
            last_failure_http_status = None
        else:
            consecutive_failures = old_failures + 1
            delay = cooldown_hours[min(consecutive_failures - 1, len(cooldown_hours) - 1)]
            next_probe_at = iso_z(dt.datetime.now(UTC) + dt.timedelta(hours=delay))
            last_probed_at = built_at
            last_failure_detail = health.get("detail")
            last_failure_http_status = health.get("http_status")

        record = {
            "url": entry.get("url"),
            "headers": entry.get("headers") or {},
            "tvg_id": entry.get("tvg_id"),
            "name": entry.get("name"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "quality": quality_label(entry),
            "declared_quality": entry.get("quality") or "Unknown",
            "measured_quality": entry.get("measured_quality"),
            **score,
            "consecutive_failures": consecutive_failures,
            "next_probe_at": next_probe_at,
            "last_probed_at": last_probed_at,
            "last_failure_detail": last_failure_detail,
            "last_failure_http_status": last_failure_http_status,
            "missed_builds": 0,
        }
        state_streams[key] = record
        current_counts[score["classification"]] += 1

        entry_score = {
            k: score[k]
            for k in (
                "classification",
                "rank",
                "passes",
                "samples",
                "history_size",
                "history",
                "last_result",
                "last_seen",
            )
        }
        entry_score["consecutive_failures"] = consecutive_failures
        entry_score["next_probe_at"] = next_probe_at
        for linked in linked_by_key[key]:
            linked["stability"] = dict(entry_score)

    log(
        "Cross-build stream stability: "
        + ", ".join(
            f"{name}={current_counts.get(name, 0)}"
            for name in ("Stable", "Backup", "Quarantine", "Dead")
        )
    )

    return {
        "version": 1,
        "enabled": True,
        "history_size": history_size,
        "dead_source_cooldown_hours": cooldown_hours,
        "updated_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "streams": state_streams,
        "current_counts": dict(current_counts),
    }


def _channel_state_key(row: dict[str, Any]) -> str:
    group = str(row.get("group") or "Unknown").strip()
    pattern = str(row.get("pattern") or "").strip()
    return f"{group}:{canonical_tvg_id(pattern)}"


_PREVIOUS_CHANNEL_STATE_CACHE: dict[str, Any] | None = None


def _load_previous_channel_state(config: dict[str, Any]) -> dict[str, Any]:
    """Load the last deployed exact-channel state for build-to-build comparison."""
    global _PREVIOUS_CHANNEL_STATE_CACHE
    if _PREVIOUS_CHANNEL_STATE_CACHE is not None:
        return _PREVIOUS_CHANNEL_STATE_CACHE

    settings = config.get("channel_diagnostics") or {}
    if not settings.get("enabled", True):
        _PREVIOUS_CHANNEL_STATE_CACHE = {}
        return _PREVIOUS_CHANNEL_STATE_CACHE

    explicit = (settings.get("previous_state_url") or "").strip()
    public_base = derive_public_base()
    if explicit:
        state_url = explicit
    elif public_base:
        state_url = f"{public_base}/diagnostics/channel-state.json"
    else:
        _PREVIOUS_CHANNEL_STATE_CACHE = {}
        return _PREVIOUS_CHANNEL_STATE_CACHE

    cache_bust = os.getenv("GITHUB_RUN_ID") or str(int(time.time()))
    separator = "&" if "?" in state_url else "?"
    request_url = f"{state_url}{separator}build={urllib.parse.quote(cache_bust)}"
    req = urllib.request.Request(
        request_url,
        headers={
            "User-Agent": USER_AGENT,
            "Accept": "application/json",
            "Accept-Encoding": "identity",
            "Cache-Control": "no-cache",
        },
    )
    try:
        with urllib.request.urlopen(
            req,
            timeout=int(settings.get("load_timeout_seconds", 8)),
        ) as resp:
            payload = json.loads(resp.read(4_000_000).decode("utf-8"))
        if isinstance(payload, dict) and isinstance(payload.get("channels"), dict):
            log(
                "Channel diagnostics: loaded "
                f"{len(payload['channels']):,} prior exact-channel states"
            )
            _PREVIOUS_CHANNEL_STATE_CACHE = payload
            return _PREVIOUS_CHANNEL_STATE_CACHE
    except urllib.error.HTTPError as exc:
        if exc.code != 404:
            log(f"WARNING: could not load prior channel state: HTTP {exc.code}")
    except Exception as exc:
        log(f"WARNING: could not load prior channel state: {str(exc)[:160]}")
    _PREVIOUS_CHANNEL_STATE_CACHE = {}
    return _PREVIOUS_CHANNEL_STATE_CACHE


def build_channel_diagnostics(
    passing_allowlist_coverage: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> tuple[dict[str, Any], dict[str, Any]]:
    """Track exact allow-listed channels across builds.

    A channel is considered present only when at least one candidate for that exact
    allow-list ID passed the current build's health gate. This intentionally tracks
    zero-candidate IDs too, instead of looking only at channels that happened to
    make it into the merged catalog.
    """
    settings = config.get("channel_diagnostics") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    built_at = iso_z(dt.datetime.now(UTC))
    previous = _load_previous_channel_state(config) if enabled else {}
    previous_channels = previous.get("channels") if isinstance(previous.get("channels"), dict) else {}
    has_prior_state = bool(previous_channels)

    channels: dict[str, dict[str, Any]] = {}
    recovered: list[dict[str, Any]] = []
    lost: list[dict[str, Any]] = []
    still_missing: list[dict[str, Any]] = []
    stable: list[dict[str, Any]] = []

    for row in passing_allowlist_coverage:
        key = _channel_state_key(row)
        old = previous_channels.get(key) if isinstance(previous_channels.get(key), dict) else {}
        was_present = bool(old.get("present")) if old else False
        present = bool(row.get("found"))

        if present:
            stable_builds = int(old.get("stable_builds", 0)) + 1 if was_present else 1
            missing_builds = 0
        else:
            stable_builds = 0
            missing_builds = int(old.get("missing_builds", 0)) + 1 if old and not was_present else 1

        record = {
            "group": row.get("group"),
            "pattern": row.get("pattern"),
            "present": present,
            "candidate_count": int(row.get("candidate_count", 0)),
            "sources": row.get("sources") or [],
            "qualities": row.get("qualities") or [],
            "stable_builds": stable_builds,
            "missing_builds": missing_builds,
            "last_changed_at": (
                old.get("last_changed_at")
                if old and was_present == present
                else built_at
            ),
            "last_seen_at": built_at if present else old.get("last_seen_at"),
            "last_missing_at": built_at if not present else old.get("last_missing_at"),
        }
        channels[key] = record

        summary_row = {
            "group": record["group"],
            "channel_id": record["pattern"],
            "candidate_count": record["candidate_count"],
            "sources": record["sources"],
            "qualities": record["qualities"],
        }

        if present:
            stable.append({**summary_row, "stable_for_builds": stable_builds})
            if has_prior_state and old and not was_present:
                recovered.append({
                    **summary_row,
                    "stable_for_builds": stable_builds,
                    "previously_missing_for_builds": int(old.get("missing_builds", 0)),
                })
        else:
            if has_prior_state and old and was_present:
                lost.append({
                    **summary_row,
                    "missing_for_builds": missing_builds,
                    "previously_stable_for_builds": int(old.get("stable_builds", 0)),
                })
            else:
                still_missing.append({**summary_row, "missing_for_builds": missing_builds})

    sort_key = lambda item: (str(item.get("group") or ""), str(item.get("channel_id") or ""))
    recovered.sort(key=sort_key)
    lost.sort(key=sort_key)
    still_missing.sort(key=sort_key)
    stable.sort(key=sort_key)

    state = {
        "version": 1,
        "enabled": enabled,
        "updated_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "has_prior_state": has_prior_state,
        "channels": channels,
    }
    report = {
        "version": 1,
        "built_at": built_at,
        "source_state_built_at": previous.get("updated_at"),
        "baseline_initialized": not has_prior_state,
        "expected_channels": len(channels),
        "present_channels": sum(1 for item in channels.values() if item.get("present")),
        "missing_channels": sum(1 for item in channels.values() if not item.get("present")),
        "recovered_this_build": recovered,
        "lost_this_build": lost,
        "still_missing": still_missing,
        "stable": stable,
    }

    log(
        "Channel changes: "
        f"recovered={len(recovered)}, lost={len(lost)}, "
        f"still_missing={len(still_missing)}, stable={len(stable)}"
        + (" (baseline initialized)" if not has_prior_state else "")
    )
    return state, report


def _dead_source_cooldown(
    entry: dict[str, Any],
    previous: dict[str, Any],
    config: dict[str, Any],
    now: dt.datetime,
) -> dict[str, Any] | None:
    """Return a skip result while a previously dead URL is backing off."""
    if entry.get("ephemeral_signed"):
        return None
    settings = config.get("stream_stability") or {}
    if not settings.get("enabled", True):
        return None
    old = (previous.get("streams") or {}).get(_stream_stability_key(entry))
    if not isinstance(old, dict) or old.get("classification") != "Dead":
        return None

    failures = int(old.get("consecutive_failures", 0) or 0)
    if failures <= 0:
        for result in reversed(old.get("history") or []):
            if result:
                break
            failures += 1
    failures = max(1, failures)
    hours = [
        max(1, int(value))
        for value in settings.get("dead_source_cooldown_hours", [1, 6, 24])
    ] or [1, 6, 24]

    next_probe_at = old.get("next_probe_at")
    try:
        if next_probe_at:
            retry_at = dt.datetime.fromisoformat(str(next_probe_at).replace("Z", "+00:00"))
        else:
            anchor_text = old.get("last_probed_at") or old.get("last_seen")
            if not anchor_text:
                return None
            anchor = dt.datetime.fromisoformat(str(anchor_text).replace("Z", "+00:00"))
            retry_at = anchor + dt.timedelta(hours=hours[min(failures - 1, len(hours) - 1)])
    except (TypeError, ValueError):
        return None
    if retry_at.tzinfo is None:
        retry_at = retry_at.replace(tzinfo=UTC)
    if retry_at <= now:
        return None
    prior_detail = str(old.get("last_failure_detail") or "").strip()
    detail = "Dead source probe deferred by backoff"
    if prior_detail:
        detail += f"; last failure: {prior_detail}"
    return {
        "status": "cooldown",
        "detail": detail,
        "cooldown_until": iso_z(retry_at),
        "consecutive_failures": failures,
        "last_http_status": old.get("last_failure_http_status"),
    }


def _parse_state_datetime(value: Any) -> dt.datetime | None:
    if not value:
        return None
    try:
        parsed = dt.datetime.fromisoformat(str(value).replace("Z", "+00:00"))
    except (TypeError, ValueError):
        return None
    return parsed.replace(tzinfo=UTC) if parsed.tzinfo is None else parsed.astimezone(UTC)


def plan_priority_recovery_probes(
    cooldown_entries: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]],
    previous_stability: dict[str, Any],
    previous_channel_state: dict[str, Any],
    config: dict[str, Any],
    now: dt.datetime,
) -> tuple[set[tuple[str, tuple[tuple[str, str], ...]]], dict[str, Any]]:
    """Select a bounded, rotating set of dead candidates for an early retry."""
    settings = (config.get("stream_health") or {}).get("priority_recovery") or {}
    enabled = bool(settings.get("enabled", True))
    max_urls = max(0, int(settings.get("max_forced_urls_per_build", 12)))
    min_retry_hours = max(0, int(settings.get("min_retry_hours", 6)))
    max_per_host = max(1, int(settings.get("max_per_host", 2)))
    favorite_patterns = config.get("favorites") or []
    prior_channels = previous_channel_state.get("channels") or {}
    missing_ids = {
        canonical_tvg_id(str(row.get("pattern") or ""))
        for row in prior_channels.values()
        if isinstance(row, dict) and not row.get("present") and row.get("pattern")
    }
    restrict_to_prior_missing = bool(missing_ids)
    previous_streams = previous_stability.get("streams") or {}
    candidates: list[dict[str, Any]] = []
    deferred_too_recent = 0
    deferred_not_missing = 0

    for key, entry in cooldown_entries.items():
        channel_id = canonical_tvg_id(str(entry.get("tvg_id") or ""))
        if restrict_to_prior_missing and channel_id not in missing_ids:
            deferred_not_missing += 1
            continue
        old = previous_streams.get(_stream_stability_key(entry))
        old = old if isinstance(old, dict) else {}
        last_probed = _parse_state_datetime(old.get("last_probed_at") or old.get("last_seen"))
        age_hours = None if last_probed is None else max(0.0, (now - last_probed).total_seconds() / 3600)
        if age_hours is not None and age_hours < min_retry_hours:
            deferred_too_recent += 1
            continue
        hostname = (urllib.parse.urlparse(str(entry.get("url") or "")).hostname or "unknown").lower()
        favorite = is_favorite(
            {"name": entry.get("name"), "tvg_id": entry.get("tvg_id")},
            favorite_patterns,
        )
        candidates.append({
            "key": key,
            "entry": entry,
            "channel_id": channel_id,
            "hostname": hostname,
            "philly": bool(entry.get("philly")),
            "favorite": favorite,
            "last_probed_at": iso_z(last_probed) if last_probed else None,
            "age_hours": age_hours,
        })

    candidates.sort(key=lambda row: (
        0 if row["philly"] else 1,
        0 if row["favorite"] else 1,
        row["last_probed_at"] or "",
        int(row["entry"].get("priority", 50)),
        row["channel_id"],
        str(row["entry"].get("url") or ""),
    ))

    selected: list[dict[str, Any]] = []
    selected_keys: set[tuple[str, tuple[tuple[str, str], ...]]] = set()
    selected_channels: set[str] = set()
    host_counts: dict[str, int] = defaultdict(int)
    for candidate in candidates:
        if len(selected) >= max_urls:
            break
        if candidate["channel_id"] in selected_channels:
            continue
        if host_counts[candidate["hostname"]] >= max_per_host:
            continue
        entry = candidate["entry"]
        selected_keys.add(candidate["key"])
        selected_channels.add(candidate["channel_id"])
        host_counts[candidate["hostname"]] += 1
        selected.append({
            "channel_id": entry.get("tvg_id"),
            "name": entry.get("name"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "url": entry.get("url"),
            "hostname": candidate["hostname"],
            "philly": candidate["philly"],
            "favorite": candidate["favorite"],
            "last_probed_at": candidate["last_probed_at"],
            "age_hours": round(candidate["age_hours"], 1) if candidate["age_hours"] is not None else None,
            "status": "scheduled",
        })

    report = {
        "version": 1,
        "built_at": iso_z(now),
        "enabled": enabled,
        "max_forced_urls_per_build": max_urls,
        "min_retry_hours": min_retry_hours,
        "max_per_host": max_per_host,
        "restricted_to_prior_missing": restrict_to_prior_missing,
        "prior_missing_channels": len(missing_ids),
        "cooldown_candidates": len(cooldown_entries),
        "eligible_candidates": len(candidates),
        "selected_count": len(selected) if enabled else 0,
        "deferred_too_recent": deferred_too_recent,
        "deferred_not_missing": deferred_not_missing,
        "selected": selected if enabled else [],
    }
    if not enabled or max_urls == 0:
        return set(), report
    return selected_keys, report


def probe_candidate_entries(
    entries: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> tuple[list[dict[str, Any]], list[dict[str, Any]], dict[str, Any]]:
    """Probe the full curated candidate pool before selecting channel streams.

    Production can require a multi-probe burn-in window. All candidates are
    probed in parallel at each interval, so the survival window is global
    rather than multiplied by the number of streams.
    """
    settings = config.get("stream_health") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    require_passed_only = bool(settings.get("require_passed_only", True))
    timeout = int(settings.get("timeout_seconds", 6))
    max_workers = int(settings.get("max_workers", 16))
    verify_segment = bool(settings.get("verify_segment_for_all_candidates", True))
    double_probe = bool(settings.get("double_probe", True))
    second_probe_delay = int(settings.get("second_probe_delay_seconds", 60))
    burn_in_settings = settings.get("burn_in") or {}
    burn_in_enabled = bool(burn_in_settings.get("enabled", False))
    burn_in_scope = str(burn_in_settings.get("scope", "promotion-only")).strip().lower()
    burn_in_duration = max(1, int(burn_in_settings.get("duration_seconds", 60)))
    burn_in_interval = max(1, int(burn_in_settings.get("interval_seconds", 15)))
    established_classifications = {
        str(value).strip()
        for value in burn_in_settings.get(
            "established_classifications",
            ["Stable", "Backup"],
        )
        if str(value).strip()
    }

    rows: list[dict[str, Any]] = []
    previous_stability: dict[str, Any] = {}

    if not enabled:
        for entry in entries:
            entry["health"] = {"status": "not-tested", "detail": "Health checks disabled"}
            rows.append({
                "name": entry.get("name"),
                "tvg_id": entry.get("tvg_id"),
                "source": entry.get("source"),
                "family": entry.get("family"),
                "quality": entry.get("quality") or "Unknown",
                "url": entry.get("url"),
                "health": entry.get("health"),
            })
        stability_state = apply_stream_stability(entries, config, enabled=False)
        for row, entry in zip(rows, entries):
            row["stability"] = entry.get("stability")
        return entries, rows, stability_state

    previous_stability = _load_previous_stream_stability(config)
    jobs: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    refs: dict[tuple[str, tuple[tuple[str, str], ...]], list[dict[str, Any]]] = defaultdict(list)
    results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    entries_by_key: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    cooldown_entries: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    cooldown_results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    now = dt.datetime.now(UTC)

    for entry in entries:
        headers_key = tuple(sorted((entry.get("headers") or {}).items()))
        key = (entry.get("url") or "", headers_key)
        refs[key].append(entry)
        entries_by_key.setdefault(key, entry)
        cooldown = _dead_source_cooldown(entry, previous_stability, config, now)
        if cooldown:
            cooldown_entries.setdefault(key, entry)
            cooldown_results.setdefault(key, cooldown)

    previous_channel_state = _load_previous_channel_state(config)
    forced_keys, priority_recovery = plan_priority_recovery_probes(
        cooldown_entries,
        previous_stability,
        previous_channel_state,
        config,
        now,
    )
    for key, entry in entries_by_key.items():
        if key in cooldown_results and key not in forced_keys:
            results[key] = cooldown_results[key]
        else:
            jobs[key] = entry
    if forced_keys:
        log(
            "Priority recovery: forcing "
            f"{len(forced_keys):,}/{len(cooldown_entries):,} cooldown candidates"
        )

    def run_probe_batch(batch: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]) -> dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]]:
        batch_results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
        with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
            future_map = {
                executor.submit(_probe_stream, entry, timeout, verify_segment): key
                for key, entry in batch.items()
            }
            for future in concurrent.futures.as_completed(future_map):
                key = future_map[future]
                try:
                    batch_results[key] = future.result()
                except Exception as exc:
                    batch_results[key] = {"status": "failed", "detail": str(exc)[:180]}
        return batch_results

    results.update(run_probe_batch(jobs))

    first_pass = {
        key: jobs[key]
        for key, health in results.items()
        if health.get("status") == "ok"
    }
    first_probe_health = {
        key: dict(results[key])
        for key in first_pass
    }

    def finalize_continuity(
        first_health: dict[str, Any],
        final_health: dict[str, Any],
        success_detail: str,
        *,
        failure_detail: str,
        burn_in: dict[str, Any] | None = None,
    ) -> dict[str, Any]:
        extra = {"burn_in": burn_in} if burn_in is not None else {}
        if final_health.get("status") != "ok":
            return {
                **final_health,
                "status": "failed",
                "detail": failure_detail,
                "probe_1": first_health,
                "probe_2": final_health,
                "stable": False,
                **extra,
            }

        first_tail = first_health.get("hls_tail_sha256")
        final_tail = final_health.get("hls_tail_sha256")
        first_sequence = first_health.get("hls_media_sequence")
        final_sequence = final_health.get("hls_media_sequence")
        stalled_live_hls = bool(
            first_health.get("hls_is_live")
            and final_health.get("hls_is_live")
            and first_tail
            and final_tail
            and first_tail == final_tail
            and (
                first_sequence is None
                or final_sequence is None
                or first_sequence == final_sequence
            )
        )
        if stalled_live_hls:
            return {
                **final_health,
                "status": "failed",
                "detail": "Live HLS window did not advance across validation window",
                "continuity_failure": "stalled-live-hls-window",
                "probe_1": first_health,
                "probe_2": final_health,
                "stable": False,
                **extra,
            }

        first_sample = first_health.get("media_sample_sha256")
        final_sample = final_health.get("media_sample_sha256")
        frozen_direct_media = bool(
            first_sample
            and final_sample
            and first_sample == final_sample
            and int(first_health.get("media_sample_bytes") or 0) >= 32768
            and int(final_health.get("media_sample_bytes") or 0) >= 32768
        )
        if frozen_direct_media:
            return {
                **final_health,
                "status": "failed",
                "detail": "Direct media sample did not advance across validation window",
                "continuity_failure": "identical-direct-media-sample",
                "probe_1": first_health,
                "probe_2": final_health,
                "stable": False,
                **extra,
            }

        return {
            **first_health,
            "status": "ok",
            "detail": success_detail,
            "probe_1": first_health,
            "probe_2": final_health,
            "stable": True,
            **extra,
        }

    if burn_in_enabled and first_pass:
        previous_streams = previous_stability.get("streams") or {}
        burn_in_pass: dict[
            tuple[str, tuple[tuple[str, str], ...]],
            dict[str, Any],
        ] = {}
        established_pass: dict[
            tuple[str, tuple[tuple[str, str], ...]],
            dict[str, Any],
        ] = {}
        prior_classification: dict[
            tuple[str, tuple[tuple[str, str], ...]],
            str | None,
        ] = {}

        for key, entry in first_pass.items():
            old = previous_streams.get(_stream_stability_key(entry))
            old = old if isinstance(old, dict) else {}
            old_class = str(old.get("classification") or "").strip() or None
            prior_classification[key] = old_class
            established = bool(
                old_class
                and old_class in established_classifications
                and burn_in_scope != "all"
            )
            if established:
                established_pass[key] = entry
            else:
                burn_in_pass[key] = entry

        checks_after_first = max(
            1,
            (burn_in_duration + burn_in_interval - 1) // burn_in_interval,
        )
        planned_elapsed = checks_after_first * burn_in_interval
        active = dict(burn_in_pass)
        samples_by_key: dict[
            tuple[str, tuple[tuple[str, str], ...]],
            list[dict[str, Any]],
        ] = {
            key: [results[key]]
            for key in active
        }
        log(
            "Health validation split: "
            f"{len(first_pass):,} passed probe 1; "
            f"burn-in={len(burn_in_pass):,}, "
            f"established-two-probe={len(established_pass):,}; "
            f"{checks_after_first} interval(s) every {burn_in_interval}s "
            f"across ~{planned_elapsed}s"
        )

        for probe_index in range(2, checks_after_first + 2):
            is_final_probe = probe_index == checks_after_first + 1
            batch = dict(active)
            if is_final_probe and double_probe:
                batch.update(established_pass)
            if not batch:
                if double_probe and established_pass and not is_final_probe:
                    time.sleep(burn_in_interval)
                    continue
                break

            time.sleep(burn_in_interval)
            next_results = run_probe_batch(batch)

            for key in list(active):
                health = next_results.get(
                    key,
                    {"status": "failed", "detail": f"No burn-in probe {probe_index} result"},
                )
                samples_by_key[key].append(health)
                if health.get("status") == "ok":
                    results[key] = health
                    continue

                elapsed = (probe_index - 1) * burn_in_interval
                results[key] = {
                    **health,
                    "status": "failed",
                    "detail": (
                        f"Burn-in probe {probe_index} failed after initial pass: "
                        f"{health.get('detail', 'unknown')}"
                    ),
                    "continuity_failure": "burn-in-probe-failed",
                    "validation_mode": "promotion-burn-in",
                    "prior_classification": prior_classification.get(key),
                    "probe_1": samples_by_key[key][0],
                    "probe_2": health,
                    "stable": False,
                    "burn_in": {
                        "enabled": True,
                        "scope": burn_in_scope,
                        "duration_seconds": burn_in_duration,
                        "interval_seconds": burn_in_interval,
                        "planned_probes": checks_after_first + 1,
                        "probes_attempted": len(samples_by_key[key]),
                        "elapsed_seconds": elapsed,
                        "failed_probe": probe_index,
                        "passed": False,
                    },
                }
                del active[key]

            if is_final_probe and double_probe:
                for key, entry in established_pass.items():
                    first_health = first_probe_health[key]
                    final_health = next_results.get(
                        key,
                        {"status": "failed", "detail": "No established final probe result"},
                    )
                    result = finalize_continuity(
                        first_health,
                        final_health,
                        "Passed established-stream two-probe validation",
                        failure_detail=(
                            "Established final probe failed after probe 1 passed: "
                            f"{final_health.get('detail', 'unknown')}"
                        ),
                    )
                    result["validation_mode"] = "established-two-probe"
                    result["prior_classification"] = prior_classification.get(key)
                    results[key] = result

        for key in active:
            samples = samples_by_key[key]
            burn_in = {
                "enabled": True,
                "scope": burn_in_scope,
                "duration_seconds": burn_in_duration,
                "interval_seconds": burn_in_interval,
                "planned_probes": checks_after_first + 1,
                "probes_attempted": len(samples),
                "elapsed_seconds": planned_elapsed,
                "failed_probe": None,
                "passed": True,
            }
            result = finalize_continuity(
                samples[0],
                samples[-1],
                f"Passed {planned_elapsed}s burn-in ({len(samples)} probes) before promotion",
                failure_detail=(
                    f"Final burn-in probe failed after probe 1 passed: "
                    f"{samples[-1].get('detail', 'unknown')}"
                ),
                burn_in=burn_in,
            )
            result["validation_mode"] = "promotion-burn-in"
            result["prior_classification"] = prior_classification.get(key)
            if result.get("status") != "ok":
                result["burn_in"]["passed"] = False
            results[key] = result

    elif double_probe and first_pass:
        log(
            f"Health stability check: {len(first_pass):,} candidates passed probe 1; "
            f"waiting {second_probe_delay}s before probe 2"
        )
        time.sleep(max(0, second_probe_delay))
        second_results = run_probe_batch(first_pass)

        for key, first_health in list(results.items()):
            if first_health.get("status") != "ok":
                continue
            second_health = second_results.get(
                key,
                {"status": "failed", "detail": "No second probe result"},
            )
            results[key] = finalize_continuity(
                first_health,
                second_health,
                "Passed two HLS/media probes",
                failure_detail=(
                    "Probe 2 failed after probe 1 passed: "
                    f"{second_health.get('detail', 'unknown')}"
                ),
            )

    direct_survival_settings = settings.get("direct_media_survival") or {}
    direct_survival_enabled = bool(direct_survival_settings.get("enabled", False))
    direct_survival_workers = max(1, int(direct_survival_settings.get("max_workers", 8)))
    if direct_survival_enabled:
        direct_jobs = {
            key: entries_by_key[key]
            for key, health in results.items()
            if health.get("status") == "ok"
            and int(health.get("media_sample_bytes") or 0) > 0
            and not health.get("hls_segment_count")
        }
        if direct_jobs:
            log(f"Direct-media decoder gate: testing {len(direct_jobs):,} passing direct candidates")
            gate_results = {}
            with concurrent.futures.ThreadPoolExecutor(max_workers=direct_survival_workers) as executor:
                future_map = {
                    executor.submit(_validate_direct_media_survival, entry, direct_survival_settings): key
                    for key, entry in direct_jobs.items()
                }
                for future in concurrent.futures.as_completed(future_map):
                    key = future_map[future]
                    try:
                        gate_results[key] = future.result()
                    except Exception as exc:
                        gate_results[key] = {"status": "failed", "detail": f"Direct media decoder gate error: {str(exc)[:160]}", "continuity_failure": "direct-media-decoder-error"}
            direct_passed = 0
            direct_failed = 0
            for key, gate in gate_results.items():
                current = results[key]
                if gate.get("status") == "ok":
                    current["direct_media_survival"] = gate.get("direct_media_survival")
                    current["direct_media_survival_detail"] = gate.get("detail")
                    direct_passed += 1
                else:
                    results[key] = {
                        **current,
                        "status": "failed",
                        "detail": gate.get("detail") or "Direct media survival failed",
                        "continuity_failure": gate.get("continuity_failure", "direct-media-survival-failed"),
                        "direct_media_survival": gate.get("direct_media_survival"),
                        "stable": False,
                    }
                    direct_failed += 1
            log(f"Direct-media decoder gate: passed={direct_passed:,}, failed={direct_failed:,}")
    selected_by_url = {row["url"]: row for row in priority_recovery["selected"]}
    for key in forced_keys:
        health = results.get(key, {"status": "failed", "detail": "No probe result"})
        health["priority_recovery"] = True
        health["priority_recovery_reason"] = "bounded retry for a previously missing channel"
        results[key] = health
        selected = selected_by_url.get(key[0])
        if selected is not None:
            selected["status"] = health.get("status", "failed")
            selected["detail"] = health.get("detail")

    for key, linked_entries in refs.items():
        health = results.get(key, {"status": "failed", "detail": "No probe result"})
        for entry in linked_entries:
            entry["health"] = dict(health)
            entry["effective_url"] = str(health.get("final_url") or entry.get("url") or "")
            entry["provider_host"] = stream_provider_host(entry)
            for field in (
                "measured_width",
                "measured_height",
                "measured_quality",
                "quality_source",
                "available_resolutions",
            ):
                if health.get(field) is not None:
                    entry[field] = health[field]

    stability_state = apply_stream_stability(
        entries,
        config,
        enabled=True,
        previous=previous_stability,
    )
    stability_state["priority_recovery"] = priority_recovery

    for entry in entries:
        rows.append({
            "name": entry.get("name"),
            "tvg_id": entry.get("tvg_id"),
            "source": entry.get("source"),
            "family": entry.get("family"),
            "source_type": entry.get("source_type"),
            "quality": quality_label(entry),
            "declared_quality": entry.get("quality") or "Unknown",
            "measured_quality": entry.get("measured_quality"),
            "measured_width": entry.get("measured_width"),
            "measured_height": entry.get("measured_height"),
            "quality_source": entry.get("quality_source"),
            "available_resolutions": entry.get("available_resolutions") or [],
            "url": entry.get("url"),
            "effective_url": stream_effective_url(entry),
            "provider_host": stream_provider_host(entry),
            "health": entry.get("health"),
            "stability": entry.get("stability"),
            "discovery_issue": entry.get("discovery_issue"),
            "content_note": entry.get("content_note"),
        })

    counts = defaultdict(int)
    for row in rows:
        counts[(row.get("health") or {}).get("status", "not-tested")] += 1
    log(
        "Candidate stream health: "
        + ", ".join(f"{status}={count}" for status, count in sorted(counts.items()))
    )

    if not require_passed_only:
        return entries, rows, stability_state

    healthy = [
        entry
        for entry in entries
        if (entry.get("health") or {}).get("status") == "ok"
    ]
    log(f"Passed-only policy: keeping {len(healthy):,}/{len(entries):,} curated candidates")
    return healthy, rows, stability_state

def _load_previous_source_scoreboard(config: dict[str, Any]) -> dict[str, Any]:
    settings = config.get("source_scoreboard") or {}
    if not settings.get("enabled", True):
        return {}
    state_url = str(settings.get("previous_state_url") or "").strip()
    public_base = derive_public_base()
    if not state_url and public_base:
        state_url = f"{public_base}/diagnostics/source-family-scoreboard.json"
    if not state_url:
        return {}
    separator = "&" if "?" in state_url else "?"
    cache_bust = urllib.parse.quote(os.getenv("GITHUB_RUN_ID") or str(int(time.time())))
    req = urllib.request.Request(
        f"{state_url}{separator}build={cache_bust}",
        headers={"User-Agent": USER_AGENT, "Accept": "application/json", "Cache-Control": "no-cache"},
    )
    try:
        with urllib.request.urlopen(req, timeout=int(settings.get("load_timeout_seconds", 8))) as resp:
            payload = json.loads(resp.read(4_000_000).decode("utf-8"))
        if isinstance(payload, dict) and isinstance(payload.get("sources"), list):
            return payload
    except urllib.error.HTTPError as exc:
        if exc.code != 404:
            log(f"WARNING: could not load prior source scoreboard: HTTP {exc.code}")
    except Exception as exc:
        log(f"WARNING: could not load prior source scoreboard: {str(exc)[:160]}")
    return {}


def build_source_family_scoreboard(
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
    channel_changes: dict[str, Any],
    discovery_results: list[dict[str, Any]],
    config: dict[str, Any],
    *,
    enabled: bool = True,
    previous: dict[str, Any] | None = None,
) -> dict[str, Any]:
    """Measure which source families produce distinct, passing, selected streams over time."""
    settings = config.get("source_scoreboard") or {}
    history_size = max(1, int(settings.get("history_size", 5)))
    remove_min = max(2, int(settings.get("remove_review_min_builds", 3)))
    prior = previous if previous is not None else (_load_previous_source_scoreboard(config) if enabled else {})
    prior_by_key = {row.get("key"): row for row in (prior.get("sources") or []) if row.get("key")}
    metrics: dict[str, dict[str, Any]] = {}

    def source_key(family: Any, source: Any) -> str:
        return f"{str(family or 'unknown').strip()}::{str(source or family or 'unknown').strip()}"

    for row in health_rows:
        family = row.get("family") or "unknown"
        source = row.get("source") or family
        key = source_key(family, source)
        metric = metrics.setdefault(key, {
            "key": key, "family": family, "source": source, "candidates": 0,
            "effective_urls": set(), "passing": 0, "failed": 0, "cooldown": 0,
            "selected_streams": 0, "selected_channels": set(), "provider_hosts": set(),
            "discovery_events": 0, "discovered_candidates": 0,
        })
        metric["candidates"] += 1
        metric["effective_urls"].add(row.get("effective_url") or row.get("url"))
        status = (row.get("health") or {}).get("status", "not-tested")
        if status == "ok":
            metric["passing"] += 1
        elif status == "failed":
            metric["failed"] += 1
        elif status == "cooldown":
            metric["cooldown"] += 1
        host = row.get("provider_host")
        if host:
            metric["provider_hosts"].add(host)

    for channel in channels:
        for stream in channel.get("streams") or []:
            family = stream.get("family") or "unknown"
            source = stream.get("source") or family
            key = source_key(family, source)
            metric = metrics.setdefault(key, {
                "key": key, "family": family, "source": source, "candidates": 0,
                "effective_urls": set(), "passing": 0, "failed": 0, "cooldown": 0,
                "selected_streams": 0, "selected_channels": set(), "provider_hosts": set(),
                "discovery_events": 0, "discovered_candidates": 0,
            })
            metric["selected_streams"] += 1
            metric["selected_channels"].add(channel.get("tvg_id"))
            host = stream_provider_host(stream)
            if host:
                metric["provider_hosts"].add(host)

    for event in discovery_results:
        family = event.get("family") or event.get("source_family")
        source = event.get("source") or event.get("name") or family
        if not family:
            continue
        key = source_key(family, source)
        metric = metrics.setdefault(key, {
            "key": key, "family": family, "source": source, "candidates": 0,
            "effective_urls": set(), "passing": 0, "failed": 0, "cooldown": 0,
            "selected_streams": 0, "selected_channels": set(), "provider_hosts": set(),
            "discovery_events": 0, "discovered_candidates": 0,
        })
        metric["discovery_events"] += 1
        metric["discovered_candidates"] += int(event.get("candidates_found", 0) or 0)

    recovered = {canonical_tvg_id(row.get("tvg_id", "")) for row in channel_changes.get("recovered_this_build", [])}
    built_at = dt.datetime.now(UTC).isoformat(timespec="seconds").replace("+00:00", "Z")
    rows = []
    for key in sorted(set(metrics) | set(prior_by_key)):
        metric = metrics.get(key) or {
            "key": key, "family": prior_by_key[key].get("family"), "source": prior_by_key[key].get("source"),
            "candidates": 0, "effective_urls": set(), "passing": 0, "failed": 0, "cooldown": 0,
            "selected_streams": 0, "selected_channels": set(), "provider_hosts": set(),
            "discovery_events": 0, "discovered_candidates": 0,
        }
        selected_ids = {canonical_tvg_id(x or "") for x in metric["selected_channels"]}
        snapshot = {
            "built_at": built_at, "candidates": metric["candidates"],
            "unique_effective_urls": len({x for x in metric["effective_urls"] if x}),
            "passing": metric["passing"], "failed": metric["failed"], "cooldown": metric["cooldown"],
            "selected_streams": metric["selected_streams"], "selected_channels": len(selected_ids),
            "provider_hosts": len(metric["provider_hosts"]), "recovered_channels": len(selected_ids & recovered),
        }
        old_history = list((prior_by_key.get(key) or {}).get("history") or [])
        history = (old_history + [snapshot])[-history_size:]
        sampled = [item for item in history if int(item.get("candidates", 0)) > 0]
        total_passing = sum(int(item.get("passing", 0)) for item in history)
        total_selected = sum(int(item.get("selected_streams", 0)) for item in history)
        if total_selected:
            recommendation = "keep"
        elif total_passing:
            recommendation = "observe"
        elif len(sampled) >= remove_min:
            recommendation = "remove-review"
        else:
            recommendation = "collect-more-data"
        rows.append({
            "key": key, "family": metric["family"], "source": metric["source"],
            "current": snapshot, "history": history, "recommendation": recommendation,
            "discovery_events": metric["discovery_events"],
            "discovered_candidates": metric["discovered_candidates"],
        })
    return {
        "version": 1, "updated_at": built_at, "history_size": history_size,
        "source_state_built_at": prior.get("updated_at"),
        "summary": {
            "sources": len(rows),
            "productive": sum(1 for row in rows if row["current"]["passing"] or row["current"]["selected_streams"]),
            "zero_yield": sum(1 for row in rows if not row["current"]["passing"] and not row["current"]["selected_streams"]),
            "remove_review": sum(1 for row in rows if row["recommendation"] == "remove-review"),
        },
        "sources": rows,
    }


def _runner_or_geo_blocked_health(health: dict[str, Any]) -> bool:
    """Return True when the builder failure looks egress/geo/access related."""
    if not isinstance(health, dict) or health.get("status") == "ok":
        return False
    try:
        status_code = int(health.get("http_status"))
    except (TypeError, ValueError):
        status_code = 0
    if status_code in {401, 403, 429, 451}:
        return True
    blob = json.dumps(health, sort_keys=True, default=str)
    return bool(
        re.search(
            r"(?:HTTP\\s+(?:401|403|429|451)|geo(?:graphy|blocked)?|forbidden|access denied|region(?:al)?(?:ly)? blocked)",
            blob,
            re.I,
        )
    )


def build_local_test_channels(
    probed_entries: list[dict[str, Any]],
    published_channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> tuple[list[dict[str, Any]], dict[str, Any]]:
    """Expose runner/geo-blocked candidates in a separate Stremio test catalog.

    These streams never participate in the trusted channel lineup, stability
    promotion, or final Primary QA. They are surfaced only so a user can test
    whether a URL blocked from GitHub Actions works from their own network.
    """
    settings = config.get("local_test_catalog") or {}
    enabled = bool(settings.get("enabled", False))
    max_streams = max(1, int(settings.get("max_streams_per_channel", 8)))
    built_at = iso_z(dt.datetime.now(UTC))
    if not enabled:
        return [], {
            "version": 1,
            "enabled": False,
            "built_at": built_at,
            "summary": {"channels": 0, "streams": 0},
            "channels": [],
        }

    published_ids = {
        canonical_tvg_id(str(channel.get("tvg_id") or ""))
        for channel in published_channels
        if channel.get("tvg_id")
    }
    pinned_target_ids = {
        canonical_tvg_id(str(value))
        for value in settings.get("target_ids", [])
        if str(value).strip()
    }
    candidates: list[dict[str, Any]] = []
    for entry in probed_entries:
        tvg_id = str(entry.get("tvg_id") or "")
        marker = canonical_tvg_id(tvg_id)
        if not marker or marker in published_ids:
            continue
        health = entry.get("health") or {}
        explicitly_pinned = bool(pinned_target_ids and marker in pinned_target_ids)
        if explicitly_pinned:
            if health.get("status") == "ok":
                continue
        elif not _runner_or_geo_blocked_health(health):
            continue
        item = dict(entry)
        item["local_test"] = True
        item["local_test_reason"] = str(health.get("detail") or "Runner/geo access failure")
        candidates.append(item)

    channels = merge_channels(
        candidates,
        config.get("favorites", []),
        max_streams,
        config,
    )
    rows: list[dict[str, Any]] = []
    for channel in channels:
        original_id = channel["id"]
        digest = hashlib.sha1(
            f"local-test:{canonical_tvg_id(str(channel.get('tvg_id') or channel.get('name') or original_id))}".encode("utf-8")
        ).hexdigest()[:14]
        channel["id"] = f"stremiotv_localtest_{digest}"
        channel["original_channel_id"] = original_id
        channel["name"] = f"🧪 {channel['name']}"
        channel["category"] = "Local Test"
        channel["service_type"] = "Runner-blocked candidates"
        channel["favorite"] = False
        channel["local_test"] = True
        provider_hosts: list[str] = []
        for index, stream in enumerate(channel.get("streams") or [], start=1):
            stream["display_role"] = f"LOCAL TEST {index}"
            stream["local_test"] = True
            stream["local_test_reason"] = str(
                (stream.get("health") or {}).get("detail")
                or stream.get("local_test_reason")
                or "Runner/geo access failure"
            )
            host = stream_provider_host(stream)
            if host and host not in provider_hosts:
                provider_hosts.append(host)
        rows.append({
            "id": channel["id"],
            "tvg_id": channel.get("tvg_id"),
            "name": channel.get("name"),
            "stream_count": len(channel.get("streams") or []),
            "provider_hosts": provider_hosts,
            "sources": [
                stream.get("source")
                for stream in channel.get("streams") or []
            ],
        })

    report = {
        "version": 1,
        "enabled": True,
        "built_at": built_at,
        "summary": {
            "channels": len(channels),
            "streams": sum(len(channel.get("streams") or []) for channel in channels),
            "max_streams_per_channel": max_streams,
            "pinned_target_ids": len(pinned_target_ids),
        },
        "channels": rows,
    }
    log(
        "Local Test catalog: "
        f"{report['summary']['channels']:,} runner-blocked channels, "
        f"{report['summary']['streams']:,} candidate streams"
    )
    return channels, report


def build_missing_source_diagnostics(
    candidate_coverage: list[dict[str, Any]],
    passing_coverage: list[dict[str, Any]],
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> dict[str, Any]:
    """Turn broad missing-source counts into mutually exclusive work queues."""
    passing_by_id = {canonical_tvg_id(row.get("pattern", "")): row for row in passing_coverage}
    health_by_id: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for row in health_rows:
        health_by_id[canonical_tvg_id(row.get("tvg_id", ""))].append(row)
    channels_by_id = {canonical_tvg_id(channel.get("tvg_id", "")): channel for channel in channels}
    target = config.get("stream_selection") or {}
    target_hd = int(target.get("target_hd", 2))
    target_sd = int(target.get("target_sd", 1))
    private_only = _private_connector_only_ids(config)

    rows: list[dict[str, Any]] = []
    bucket_counts: dict[str, int] = defaultdict(int)
    without_passing_counts: dict[str, int] = defaultdict(int)
    for candidate in candidate_coverage:
        channel_id = candidate.get("pattern", "")
        key = canonical_tvg_id(channel_id)
        passing = passing_by_id.get(key) or {}
        channel = channels_by_id.get(key)
        probes = health_by_id.get(key, [])
        statuses = [(row.get("health") or {}).get("status", "not-tested") for row in probes]
        details = [str((row.get("health") or {}).get("detail") or "") for row in probes]
        candidate_count = int(candidate.get("candidate_count", 0) or 0)
        passing_count = int(passing.get("candidate_count", 0) or 0)
        failure_kind: str | None = None

        public_search_eligible = key not in private_only

        if passing_count == 0 and not public_search_eligible:
            bucket = "private_connector_required"
            action = "Use the authorized HDHomeRun/provider connector; skip recurring public searches."
            failure_kind = "private_or_authorized_access"
        elif candidate_count == 0:
            bucket = "no_candidate"
            action = "Find an exact-ID public source."
        elif passing_count == 0:
            blocked = any(
                re.search(r"(?:HTTP\s+(?:401|403|451|429)|geo|forbidden|access denied)", detail, re.I)
                for detail in details
            )
            if statuses and all(status == "cooldown" for status in statuses):
                bucket = "quarantined_or_cooldown"
                action = "Wait for the dead-source cooldown, then review the next scheduled retry."
                failure_kind = "dead_source_cooldown"
            elif blocked:
                bucket = "runner_or_geo_blocked"
                action = "Compare the URL in Stremio with the GitHub runner result before replacing it."
                failure_kind = "access_or_geo_signal"
            else:
                bucket = "failed_probes"
                action = "Replace or repair candidates that failed playlist/media validation."
                failure_kind = "playback_validation"
        else:
            if channel is None:
                bucket = "primary_playback_qa_failed"
                action = "Replace or promote a candidate that passes decoded Primary playback QA."
                failure_kind = "decoded_primary_playback"
            else:
                streams = channel.get("streams", [])
                measured = [stream for stream in streams if stream.get("measured_height")]
                hd_count = sum(1 for stream in streams if is_hd_stream(stream))
                sd_count = sum(1 for stream in streams if is_sd_stream(stream))
                if streams and not measured:
                    bucket = "missing_quality_metadata"
                    action = "Add or expose an HLS master playlist with RESOLUTION metadata."
                elif hd_count < target_hd or sd_count < target_sd:
                    bucket = "insufficient_quality_mix"
                    action = "Find the missing measured HD/SD rendition or backup family."
                else:
                    bucket = "complete"
                    action = "No source work required."
                failure_kind = None

        bucket_counts[bucket] += 1
        if passing_count == 0:
            without_passing_counts[bucket] += 1
        row = {
            "group": candidate.get("group"),
            "tvg_id": channel_id,
            "bucket": bucket,
            "candidate_count": candidate_count,
            "passing_candidate_count": passing_count,
            "probe_statuses": dict(Counter(statuses)),
            "sources": candidate.get("sources") or [],
            "public_search_eligible": public_search_eligible,
            "action": action,
        }
        if failure_kind:
            row["failure_kind"] = failure_kind
        rows.append(row)

    rows.sort(key=lambda row: (row["bucket"], row.get("group", ""), row["tvg_id"]))
    return {
        "summary": {
            "expected_channels": len(rows),
            "channels_without_passing_stream": sum(
                1
                for row in rows
                if row["passing_candidate_count"] == 0
                or row["bucket"] == "primary_playback_qa_failed"
            ),
            "channels_without_published_stream": sum(
                1
                for row in rows
                if row["bucket"] in {
                    "no_candidate",
                    "private_connector_required",
                    "quarantined_or_cooldown",
                    "runner_or_geo_blocked",
                    "failed_probes",
                    "primary_playback_qa_failed",
                }
            ),
            "bucket_counts": dict(sorted(bucket_counts.items())),
            "without_passing_bucket_counts": dict(sorted(without_passing_counts.items())),
        },
        "channels": rows,
    }


def build_philly_recovery_diagnostics(
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
    config: dict[str, Any],
) -> dict[str, Any]:
    settings = ((config.get("discovery") or {}).get("philly_local_sources") or {})
    target_ids = settings.get("target_ids") or []
    client_results = config.get("stremio_client_results") or {}
    verification_settings = config.get("stremio_client_verification") or {}
    now = dt.datetime.now(UTC)
    health_by_id: dict[str, list[dict[str, Any]]] = defaultdict(list)
    for row in health_rows:
        health_by_id[canonical_tvg_id(row.get("tvg_id", ""))].append(row)
    channels_by_id = {canonical_tvg_id(channel.get("tvg_id", "")): channel for channel in channels}

    rows: list[dict[str, Any]] = []
    for tvg_id in target_ids:
        key = canonical_tvg_id(tvg_id)
        probes = health_by_id.get(key, [])
        statuses = [(row.get("health") or {}).get("status", "not-tested") for row in probes]
        channel = channels_by_id.get(key)
        selected = (channel or {}).get("streams", [])
        client_ready = [
            stream for stream in selected
            if str(stream.get("url") or "").lower().startswith("https://") and not stream.get("headers")
        ]
        runner_status = (
            "passing" if "ok" in statuses
            else "cooldown" if statuses and all(status == "cooldown" for status in statuses)
            else "failing" if probes
            else "no-candidate"
        )
        configured_client = client_results.get(tvg_id) or {}
        if isinstance(configured_client, str):
            configured_client = {"status": configured_client}
        client_verification = evaluate_client_verification(
            configured_client,
            verification_settings,
            now,
        )
        client_status = client_verification["status"]
        if runner_status == "passing" and client_status == "not-tested":
            next_action = "Test the Primary URL in Stremio and record pass/fail in stremio_client_results."
        elif runner_status == "passing" and client_status == "expired":
            next_action = "Retest the Primary URL in Stremio; the prior client result has expired."
        elif client_status == "invalid":
            next_action = "Record a valid passed/failed client result with an ISO tested_at timestamp."
        elif runner_status == "passing" and client_status == "failed":
            next_action = "Investigate a Stremio-versus-runner playback mismatch."
        elif runner_status == "no-candidate":
            next_action = "Continue exact-station source discovery; do not substitute a news/FAST lookalike."
        elif runner_status == "cooldown":
            next_action = "Wait for the scheduled retry or add a new exact-station candidate."
        else:
            next_action = "Review failed probe details and compare any plausible URL in Stremio."
        rows.append({
            "tvg_id": tvg_id,
            "name": (channel or {}).get("name") or tvg_id,
            "candidate_count": len(probes),
            "probe_statuses": dict(Counter(statuses)),
            "runner_status": runner_status,
            "selected_streams": len(selected),
            "stremio_ready_selected_streams": len(client_ready),
            "stremio_client_status": client_status,
            "stremio_client_recorded_status": client_verification["recorded_status"],
            "stremio_client_tested_at": client_verification["tested_at"],
            "stremio_client_expires_at": client_verification["expires_at"],
            "stremio_client_age_hours": client_verification["age_hours"],
            "stremio_client_note": configured_client.get("note"),
            "verification_required": runner_status == "passing" and client_status not in {"passed", "failed"},
            "next_action": next_action,
        })
    return {
        "summary": {
            "targets": len(rows),
            "runner_passing": sum(1 for row in rows if row["runner_status"] == "passing"),
            "client_tested": sum(1 for row in rows if row["stremio_client_status"] in {"passed", "failed"}),
            "client_expired": sum(1 for row in rows if row["stremio_client_status"] == "expired"),
            "verification_required": sum(1 for row in rows if row["verification_required"]),
            "client_runner_mismatches": sum(
                1 for row in rows
                if row["runner_status"] == "passing" and row["stremio_client_status"] == "failed"
            ),
        },
        "channels": rows,
    }


def evaluate_client_verification(
    result: dict[str, Any],
    settings: dict[str, Any],
    now: dt.datetime | None = None,
) -> dict[str, Any]:
    """Normalize and expire a manually recorded Stremio playback observation."""
    now = now or dt.datetime.now(UTC)
    aliases = {
        "pass": "passed",
        "passed": "passed",
        "ok": "passed",
        "fail": "failed",
        "failed": "failed",
        "not-tested": "not-tested",
        "": "not-tested",
    }
    raw_status = str(result.get("status") or "not-tested").strip().lower()
    recorded_status = aliases.get(raw_status, "invalid")
    ttl_hours = max(1, int(result.get("ttl_hours") or settings.get("default_ttl_hours", 168)))
    tested_at = _parse_state_datetime(result.get("tested_at"))
    if recorded_status == "not-tested":
        return {
            "status": "not-tested",
            "recorded_status": "not-tested",
            "tested_at": None,
            "expires_at": None,
            "age_hours": None,
            "ttl_hours": ttl_hours,
        }
    if recorded_status == "invalid" or tested_at is None or tested_at > now + dt.timedelta(minutes=5):
        return {
            "status": "invalid",
            "recorded_status": recorded_status,
            "tested_at": iso_z(tested_at) if tested_at else result.get("tested_at"),
            "expires_at": None,
            "age_hours": None,
            "ttl_hours": ttl_hours,
        }
    expires_at = tested_at + dt.timedelta(hours=ttl_hours)
    age_hours = max(0.0, (now - tested_at).total_seconds() / 3600)
    return {
        "status": "expired" if now >= expires_at else recorded_status,
        "recorded_status": recorded_status,
        "tested_at": iso_z(tested_at),
        "expires_at": iso_z(expires_at),
        "age_hours": round(age_hours, 1),
        "ttl_hours": ttl_hours,
    }


def build_quality_measurements(
    health_rows: list[dict[str, Any]],
    channels: list[dict[str, Any]],
) -> dict[str, Any]:
    selected_streams = [
        stream
        for channel in channels
        for stream in channel.get("streams", [])
    ]
    selected_urls = {
        stream.get("url")
        for stream in selected_streams
        if stream.get("url")
    }
    rows: list[dict[str, Any]] = []
    for stream in health_rows:
        health = stream.get("health") or {}
        if health.get("status") != "ok":
            continue
        rows.append({
            "name": stream.get("name"),
            "tvg_id": stream.get("tvg_id"),
            "source": stream.get("source"),
            "family": stream.get("family"),
            "selected": stream.get("url") in selected_urls,
            "declared_quality": stream.get("declared_quality") or "Unknown",
            "measured_quality": stream.get("measured_quality"),
            "measured_width": stream.get("measured_width"),
            "measured_height": stream.get("measured_height"),
            "available_resolutions": stream.get("available_resolutions") or [],
            "quality_source": stream.get("quality_source"),
            "url": stream.get("url"),
        })
    return {
        "summary": {
            "passing_candidates": len(rows),
            "measured_passing_candidates": sum(1 for row in rows if row["measured_height"]),
            "selected_streams": len(selected_streams),
            "measured_selected_streams": sum(1 for stream in selected_streams if stream.get("measured_height")),
            "selected_hd": sum(1 for stream in selected_streams if is_hd_stream(stream)),
            "selected_sd": sum(1 for stream in selected_streams if is_sd_stream(stream)),
            "selected_quality_unknown": sum(
                1 for stream in selected_streams
                if not stream.get("measured_height") and not stream.get("quality")
            ),
        },
        "streams": rows,
    }


def channel_meta(channel: dict[str, Any], programmes: list[dict[str, Any]] | None = None) -> dict[str, Any]:
    programmes = sorted(programmes or [], key=lambda p: p.get("startTime", ""))

    # Native Stremio Live TV renders current/next programme data from videos.
    # Keep the visible channel description clean and reserve operational details
    # for structured diagnostics and playback-option metadata.
    description_bits: list[str] = []
    if channel.get("philly"):
        description_bits.append("Philadelphia")
    service_type = channel.get("service_type")
    category = channel.get("category", "Live TV")
    if service_type:
        description_bits.append(service_type)
    else:
        description_bits.append(category)

    # De-duplicate while preserving order.
    description = " • ".join(dict.fromkeys(bit for bit in description_bits if bit))

    version = channel.get("build_version")
    built_at = channel.get("built_at")
    build_id = channel.get("build_id")

    meta: dict[str, Any] = {
        "id": channel["id"],
        "type": "tv",
        "name": channel["name"],
        "description": description,
        "genres": [category],
        "posterShape": "square",
        "behaviorHints": {"isLive": True},
        "stremioTvDiagnostics": {
            "epg_status": channel.get(
                "epg_status",
                "matched" if channel.get("epg_matched") else "unmatched",
            ),
            "epg_matched": bool(channel.get("epg_matched")),
            "source_count": len(channel.get("streams", [])),
            "category": category,
            "service_type": service_type,
            "philly": bool(channel.get("philly")),
        },
    }
    if channel.get("logo"):
        meta["poster"] = channel["logo"]
        meta["background"] = channel["logo"]
    if channel.get("epg_matched"):
        meta["behaviorHints"]["hasScheduledVideos"] = True
        meta["videos"] = programmes
    if build_id:
        meta["stremioTvBuild"] = {
            "build_id": build_id,
            "version": version,
            "built_at": built_at,
            "source_revision": channel.get("build_revision"),
        }
    return meta

def stream_object(
    stream: dict[str, Any],
    index: int,
    build_info: dict[str, Any] | None = None,
) -> dict[str, Any]:
    quality = quality_label(stream, "Live")
    source = _short_source_name(stream)
    role = stream.get("display_role") or ("Primary" if index == 1 else f"Backup {index - 1}")
    url = stream["url"]
    health = stream.get("health") or {}
    health_status = health.get("status", "not-tested")

    description_parts = [source]
    if stream.get("measured_quality"):
        description_parts.append(f"Measured {quality}")
    elif stream.get("quality"):
        description_parts.append(f"Declared {quality}")
    else:
        description_parts.append("Quality unknown")
    if health_status == "ok":
        description_parts.append("Builder check: OK")
    elif health_status == "failed":
        description_parts.append("Builder check failed — may still work locally")
    else:
        description_parts.append("Not health-tested")
    if stream.get("local_test"):
        host = stream_provider_host(stream)
        description_parts.append("LOCAL TEST: runner/geo blocked on GitHub")
        if host:
            description_parts.append(host)
    stability = (stream.get("stability") or {}).get("classification")
    if stability and stability != "Unknown":
        description_parts.append(stability)
    if build_info:
        description_parts.append(f"Build {build_info.get('build_id')}")

    obj: dict[str, Any] = {
        "name": f"{role} • {quality}",
        "description": " • ".join(description_parts),
        "url": url,
    }
    if build_info:
        obj["stremioTvBuild"] = {
            "build_id": build_info.get("build_id"),
            "version": build_info.get("version"),
            "built_at": build_info.get("built_at"),
            "source_revision": build_info.get("source_revision"),
        }
    headers = stream.get("headers") or {}
    not_web_ready = (not url.lower().startswith("https://")) or bool(headers)
    if not_web_ready:
        obj["behaviorHints"] = {"notWebReady": True}
        if headers:
            obj["behaviorHints"]["proxyHeaders"] = {"request": headers}
    return obj


RESOURCE_CACHE_HINTS = {
    "cacheMaxAge": 300,
    "staleRevalidate": 1800,
    "staleError": 604800,
}


def cached_resource(payload: dict[str, Any]) -> dict[str, Any]:
    """Attach Stremio addon-protocol cache hints to a resource response."""
    return {**payload, **RESOURCE_CACHE_HINTS}


def write_paginated_catalog(
    base_dir: Path,
    catalog_id: str,
    channels: list[dict[str, Any]],
    page_size: int,
    discover_only: bool = False,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / catalog_id
    catalog_dir.mkdir(parents=True, exist_ok=True)
    previews = [channel_preview(c) for c in channels]

    first = cached_resource({"metas": previews[:page_size]})
    # Keep the plain endpoint for debugging/compatibility. For discover-only
    # catalogs the manifest requires genre=All, so Stremio's Board/Home will
    # not request this endpoint.
    write_json(base_dir / "catalog" / "tv" / f"{catalog_id}.json", first)
    write_json(catalog_dir / "skip=0.json", first)

    if discover_only:
        write_json(catalog_dir / "genre=All.json", first)
        write_json(catalog_dir / "genre=All&skip=0.json", first)
        write_json(catalog_dir / "skip=0&genre=All.json", first)

    skip = page_size
    while skip < len(previews):
        page = cached_resource({"metas": previews[skip : skip + page_size]})
        write_json(catalog_dir / f"skip={skip}.json", page)
        if discover_only:
            write_json(catalog_dir / f"genre=All&skip={skip}.json", page)
            write_json(catalog_dir / f"skip={skip}&genre=All.json", page)
        skip += page_size

    # Stremio advances skip by the number of items actually received.
    # The terminal empty page therefore belongs at the exact item count,
    # not at the next page-size boundary (for example 111, not 200).
    final_skip = len(previews)
    empty = cached_resource({"metas": []})
    write_json(catalog_dir / f"skip={final_skip}.json", empty)
    if discover_only:
        write_json(catalog_dir / f"genre=All&skip={final_skip}.json", empty)
        write_json(catalog_dir / f"skip={final_skip}&genre=All.json", empty)

def write_epg_pages(
    base_dir: Path,
    channels: list[dict[str, Any]],
    schedules: dict[str, dict[str, list[dict[str, Any]]]],
    dates: Iterable[str],
    page_size: int,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / "channels"
    catalog_dir.mkdir(parents=True, exist_ok=True)

    for day in sorted(dates):
        detailed = [channel_meta(c, schedules.get(c["id"], {}).get(day, [])) for c in channels]
        first = detailed[:page_size]
        write_json(catalog_dir / f"date={day}.json", cached_resource({"metasDetailed": first}))
        write_json(catalog_dir / f"date={day}&skip=0.json", cached_resource({"metasDetailed": first}))
        write_json(catalog_dir / f"skip=0&date={day}.json", cached_resource({"metasDetailed": first}))

        skip = page_size
        while skip < len(detailed):
            page = detailed[skip : skip + page_size]
            write_json(catalog_dir / f"date={day}&skip={skip}.json", cached_resource({"metasDetailed": page}))
            write_json(catalog_dir / f"skip={skip}&date={day}.json", cached_resource({"metasDetailed": page}))
