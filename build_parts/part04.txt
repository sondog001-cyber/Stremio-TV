== 0.0):
                return epg_id, "fuzzy-name", best_score

    return None, "unmatched", 0.0


def parse_xmltv_time(value: str) -> dt.datetime | None:
    value = (value or "").strip()
    if not value:
        return None
    # XMLTV often uses: YYYYMMDDHHMMSS ±HHMM
    formats = ["%Y%m%d%H%M%S %z", "%Y%m%d%H%M %z", "%Y%m%d%H%M%S", "%Y%m%d%H%M"]
    for fmt in formats:
        try:
            parsed = dt.datetime.strptime(value, fmt)
            if parsed.tzinfo is None:
                parsed = parsed.replace(tzinfo=UTC)
            return parsed.astimezone(UTC)
        except ValueError:
            continue
    return None


def iso_z(value: dt.datetime) -> str:
    return value.astimezone(UTC).isoformat(timespec="seconds").replace("+00:00", "Z")


def date_range_for_program(start: dt.datetime, end: dt.datetime) -> list[str]:
    if end <= start:
        return []
    cursor = start.date()
    last = (end - dt.timedelta(microseconds=1)).date()
    result: list[str] = []
    while cursor <= last:
        result.append(cursor.isoformat())
        cursor += dt.timedelta(days=1)
    return result


def build_program_index(
    epg_path: Path,
    epg_to_channels: dict[str, list[str]],
) -> tuple[dict[str, dict[str, list[dict[str, Any]]]], set[str]]:
    schedules: dict[str, dict[str, list[dict[str, Any]]]] = defaultdict(lambda: defaultdict(list))
    all_dates: set[str] = set()
    matched_programmes = 0

    with gzip.open(epg_path, "rb") as f:
        for event, elem in ET.iterparse(f, events=("end",)):
            if elem.tag != "programme":
                if elem.tag == "channel":
                    elem.clear()
                continue

            epg_id = elem.attrib.get("channel", "")
            channel_ids = epg_to_channels.get(epg_id)
            if not channel_ids:
                elem.clear()
                continue

            start = parse_xmltv_time(elem.attrib.get("start", ""))
            end = parse_xmltv_time(elem.attrib.get("stop", ""))
            if not start or not end or end <= start:
                elem.clear()
                continue

            title = extract_text(elem, "title") or "Untitled"
            desc = extract_text(elem, "desc")
            categories = [
                (x.text or "").strip()
                for x in elem.findall("category")
                if (x.text or "").strip()
            ]
            icon_elem = elem.find("icon")
            thumbnail = icon_elem.attrib.get("src", "") if icon_elem is not None else ""
            date_text = extract_text(elem, "date")
            runtime_min = max(1, round((end - start).total_seconds() / 60))
            start_iso = iso_z(start)
            end_iso = iso_z(end)

            for channel_id in channel_ids:
                programme: dict[str, Any] = {
                    "id": f"{channel_id}:epg:{start_iso}",
                    "title": title,
                    "released": start_iso,
                    "startTime": start_iso,
                    "endTime": end_iso,
                    "runtime": f"{runtime_min} min",
                }
                if desc:
                    programme["overview"] = desc
                if thumbnail:
                    programme["thumbnail"] = thumbnail
                if categories:
                    programme["genres"] = categories[:8]
                if date_text[:4].isdigit():
                    programme["releaseInfo"] = date_text[:4]

                for day in date_range_for_program(start, end):
                    schedules[channel_id][day].append(programme)
                    all_dates.add(day)
            matched_programmes += 1
            elem.clear()

    for channel_days in schedules.values():
        for day, programmes in channel_days.items():
            programmes.sort(key=lambda p: p["startTime"])

    log(f"Matched {matched_programmes:,} XMLTV programmes to playlist channels")
    return schedules, all_dates

def _display_time(value: str) -> str:
    parsed = dt.datetime.fromisoformat(value.replace("Z", "+00:00")).astimezone(PHILLY_TZ)
    return parsed.strftime("%I:%M %p").lstrip("0")


def _short_source_name(stream: dict[str, Any]) -> str:
    family = (stream.get("family") or "").lower()
    mapping = {
        "iptv-org": "IPTV-org",
        "free-tv": "Free-TV",
        "freecasthub": "FreeCastHub",
        "localbtv-relay": "OTA Relay",
        "moveonjoy": "MoveOnJoy",
        "tvpass": "TVPass",
        "insolence": "InsolenceTV",
        "manual": "Manual",
    }
    if family in mapping:
        return mapping[family]
    source = stream.get("source") or "IPTV"
    source = re.sub(r"\s*\(Unverified\)\s*$", "", source, flags=re.I)
    return source


def _probe_stream(stream: dict[str, Any], timeout: int, verify_segment: bool) -> dict[str, Any]:
    url = stream.get("url") or ""
    if not re.match(r"^https?://", url, re.I):
        return {"status": "not-tested", "detail": "Non-HTTP stream"}

    headers = {
        "User-Agent": USER_AGENT,
        "Accept": "*/*",
        "Accept-Encoding": "identity",
    }
    headers.update(stream.get("headers") or {})
    started = time.monotonic()

    def fetch(target: str, byte_limit: int = 262144) -> tuple[int, str, str, bytes]:
        req = urllib.request.Request(target, headers=headers)
        with urllib.request.urlopen(req, timeout=timeout) as resp:
            status = int(getattr(resp, "status", 200) or 200)
            final_url = resp.geturl()
            content_type = (resp.headers.get("Content-Type") or "").lower()
            body = resp.read(byte_limit)
            return status, final_url, content_type, body

    try:
        status, final_url, content_type, body = fetch(url)
        elapsed_ms = round((time.monotonic() - started) * 1000)
        text = body.decode("utf-8", errors="ignore")
        is_hls = (
            "#EXTM3U" in text
            or ".m3u8" in urllib.parse.urlsplit(final_url).path.lower()
            or "mpegurl" in content_type
        )

        if is_hls:
            if "#EXTM3U" not in text:
                return {
                    "status": "failed",
                    "detail": "HLS URL did not return a valid playlist",
                    "http_status": status,
                    "latency_ms": elapsed_ms,
                }

            media_text = text
            media_url = final_url
            lines = [line.strip() for line in text.splitlines() if line.strip() and not line.startswith("#")]
            if "#EXT-X-STREAM-INF" in text and lines:
                candidate = urllib.parse.urljoin(final_url, lines[0])
                child_status, child_url, _child_type, child_body = fetch(candidate)
                media_text = child_body.decode("utf-8", errors="ignore")
                media_url = child_url
                if "#EXTM3U" not in media_text:
                    return {
                        "status": "failed",
                        "detail": "Master playlist child was invalid",
                        "http_status": child_status,
                        "latency_ms": round((time.monotonic() - started) * 1000),
                    }

            if verify_segment:
                segments = [
                    line.strip()
                    for line in media_text.splitlines()
                    if line.strip() and not line.startswith("#")
                ]
                if segments:
                    segment_url = urllib.parse.urljoin(media_url, segments[0])
                    seg_headers = dict(headers)
                    seg_headers["Range"] = "bytes=0-1023"
                    req = urllib.request.Request(segment_url, headers=seg_headers)
                    with urllib.request.urlopen(req, timeout=timeout) as resp:
                        resp.read(1)

            return {
                "status": "ok",
                "detail": "Valid HLS playlist" + (" + segment" if verify_segment else ""),
                "http_status": status,
                "latency_ms": round((time.monotonic() - started) * 1000),
            }

        if "text/html" in content_type:
            return {
                "status": "failed",
                "detail": "Returned HTML instead of media",
                "http_status": status,
                "latency_ms": elapsed_ms,
            }

        return {
            "status": "ok",
            "detail": "Reachable direct media URL",
            "http_status": status,
            "latency_ms": elapsed_ms,
        }

    except urllib.error.HTTPError as exc:
        return {
            "status": "failed",
            "detail": f"HTTP {exc.code}",
            "http_status": exc.code,
            "latency_ms": round((time.monotonic() - started) * 1000),
        }
    except Exception as exc:
        return {
            "status": "failed",
            "detail": str(exc)[:180],
            "latency_ms": round((time.monotonic() - started) * 1000),
        }


def probe_selected_streams(
    channels: list[dict[str, Any]],
    config: dict[str, Any],
    enabled: bool = True,
) -> list[dict[str, Any]]:
    settings = config.get("stream_health") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    timeout = int(settings.get("timeout_seconds", 6))
    max_workers = int(settings.get("max_workers", 16))

    rows: list[dict[str, Any]] = []
    if not enabled:
        for channel in channels:
            for index, stream in enumerate(channel.get("streams", [])):
                stream["health"] = {"status": "not-tested", "detail": "Health checks disabled"}
                stream["display_role"] = "Primary" if index == 0 else f"Backup {index}"
        return rows

    jobs: dict[tuple[str, tuple[tuple[str, str], ...]], tuple[dict[str, Any], bool]] = {}
    refs: dict[tuple[str, tuple[tuple[str, str], ...]], list[tuple[dict[str, Any], dict[str, Any]]]] = defaultdict(list)

    for channel in channels:
        priority_channel = bool(channel.get("philly") or channel.get("favorite"))
        for stream in channel.get("streams", []):
            headers_key = tuple(sorted((stream.get("headers") or {}).items()))
            key = (stream.get("url") or "", headers_key)
            jobs.setdefault(key, (stream, priority_channel))
            refs[key].append((channel, stream))

    results: dict[tuple[str, tuple[tuple[str, str], ...]], dict[str, Any]] = {}
    with concurrent.futures.ThreadPoolExecutor(max_workers=max_workers) as executor:
        future_map = {
            executor.submit(
                _probe_stream,
                stream,
                timeout,
                bool(priority_channel and settings.get("verify_segment_for_priority_channels", True)),
            ): key
            for key, (stream, priority_channel) in jobs.items()
        }
        for future in concurrent.futures.as_completed(future_map):
            key = future_map[future]
            try:
                results[key] = future.result()
            except Exception as exc:
                results[key] = {"status": "failed", "detail": str(exc)[:180]}

    rank = {"ok": 0, "not-tested": 1, "failed": 2}
    for key, linked in refs.items():
        health = results.get(key, {"status": "not-tested", "detail": "No result"})
        for channel, stream in linked:
            stream["health"] = dict(health)

    for channel in channels:
        channel["streams"] = sorted(
            channel.get("streams", []),
            key=lambda s: rank.get((s.get("health") or {}).get("status", "not-tested"), 1),
        )
        for index, stream in enumerate(channel.get("streams", [])):
            stream["display_role"] = "Primary" if index == 0 else f"Backup {index}"
            rows.append({
                "channel_id": channel["id"],
                "channel_name": channel["name"],
                "tvg_id": channel.get("tvg_id"),
                "role": stream["display_role"],
                "source": stream.get("source"),
                "family": stream.get("family"),
                "quality": stream.get("quality") or "Unknown",
                "url": stream.get("url"),
                "health": stream.get("health"),
            })

    counts = defaultdict(int)
    for row in rows:
        counts[(row.get("health") or {}).get("status", "not-tested")] += 1
    log(
        "Stream health: "
        + ", ".join(f"{status}={count}" for status, count in sorted(counts.items()))
    )
    return rows


def channel_meta(channel: dict[str, Any], programmes: list[dict[str, Any]] | None = None) -> dict[str, Any]:
    programmes = sorted(programmes or [], key=lambda p: p.get("startTime", ""))
    now = dt.datetime.now(UTC)

    current: dict[str, Any] | None = None
    upcoming: dict[str, Any] | None = None
    for programme in programmes:
        try:
            p_start = dt.datetime.fromisoformat(programme["startTime"].replace("Z", "+00:00"))
            p_end = dt.datetime.fromisoformat(programme["endTime"].replace("Z", "+00:00"))
        except (KeyError, ValueError):
            continue
        if p_start <= now < p_end:
            current = programme
            continue
        if p_start >= now and upcoming is None:
            upcoming = programme

    description_lines: list[str] = []
    if current:
        description_lines.append(
            f"NOW: {current.get('title', 'Untitled')} • "
            f"{_display_time(current['startTime'])}–{_display_time(current['endTime'])}"
        )
    if upcoming:
        label = "NEXT" if current else "UP NEXT"
        description_lines.append(
            f"{label}: {upcoming.get('title', 'Untitled')} • "
            f"{_display_time(upcoming['startTime'])}–{_display_time(upcoming['endTime'])}"
        )
    if not current and not upcoming and channel.get("epg_matched"):
        description_lines.append("LIVE TV • No current programme data")
    elif not channel.get("epg_matched"):
        description_lines.append("LIVE TV • EPG unavailable")

    info_bits = []
    if channel.get("philly"):
        info_bits.append("Philadelphia local")
    info_bits.append(channel.get("category", "Live TV"))
    info_bits.append(
        f"{len(channel.get('streams', []))} stream source"
        + ("s" if len(channel.get("streams", [])) != 1 else "")
    )
    description_lines.append("")
    description_lines.append(" • ".join(info_bits))

    version = channel.get("build_version")
    built_at = channel.get("built_at")
    if version or built_at:
        build_bits = []
        if version:
            build_bits.append(f"Stremio TV v{version}")
        if built_at:
            build_bits.append(f"built {_display_time(built_at)} ET")
        description_lines.append(" • ".join(build_bits))

    meta: dict[str, Any] = {
        "id": channel["id"],
        "type": "tv",
        "name": channel["name"],
        "description": "\n".join(description_lines),
        "genres": [channel.get("category", "Live TV")],
        "posterShape": "square",
        "behaviorHints": {"isLive": True},
    }
    if channel.get("logo"):
        meta["poster"] = channel["logo"]
        meta["background"] = channel["logo"]
    return meta


def stream_object(stream: dict[str, Any], index: int) -> dict[str, Any]:
    quality = stream.get("quality") or "Live"
    source = _short_source_name(stream)
    role = stream.get("display_role") or ("Primary" if index == 1 else f"Backup {index - 1}")
    url = stream["url"]
    health = stream.get("health") or {}
    health_status = health.get("status", "not-tested")

    description_parts = [source, quality]
    if health_status == "ok":
        description_parts.append("Builder check: OK")
    elif health_status == "failed":
        description_parts.append("Builder check failed — may still work locally")
    else:
        description_parts.append("Not health-tested")

    obj: dict[str, Any] = {
        "name": f"{role} • {quality} • {source}",
        "description": " • ".join(description_parts),
        "url": url,
    }
    headers = stream.get("headers") or {}
    not_web_ready = (not url.lower().startswith("https://")) or bool(headers)
    if not_web_ready:
        obj["behaviorHints"] = {"notWebReady": True}
        if headers:
            obj["behaviorHints"]["proxyHeaders"] = {"request": headers}
    return obj


def write_paginated_catalog(
    base_dir: Path,
    catalog_id: str,
    channels: list[dict[str, Any]],
    page_size: int,
    discover_only: bool = False,
) -> None:
    catalog_dir = base_dir / "catalog" / "tv" / catalog_id
    catalog_dir.mkdir(parents=True, exist_ok=True)
    previews = [channel_preview(c) for c in channels]

    first = {"metas": previews[:page_size]}
    # Keep the plain endpoint for debugging/compatibility. For discover-only
    # catalogs the manifest requires genre=All, so Stremio's Board/Home will
    # not request this endpoint.
    write_json(base_dir / "catalog" / "tv" / f"{catalog_id}.json", first)
    write_json(catalog_dir / "skip=0.json", first)

    if discover_only:
        write_json(catalog_dir / "genre=All.json", first)
        write_json(catalog_dir / "genre=All&skip=0.json", first)
        write_json(catalog_dir / "skip=0&genre=All.json", first)

    skip = page_size
    while skip < len(previews):
        page = {"metas": previews[skip : skip + page_size]}
        write_json(catalog_dir / f"skip={skip}.json", page)
        if discover_only:
            write_json(catalog_dir / f"genre=All&skip={skip}.json", page)
            write_json(catalog_dir / f"skip={skip}&genre=All.json", page)
        skip += page_size

    # Stremio continues until it receives an empty page.
    final_skip = ((len(previews) + page_size - 1) // page_size) * page_size
    if final_skip == 0:
        final_skip = page_size
    empty = {"metas": []}
    write_json(catalog_dir / f"skip={final_skip}.json", empty)
    if discover_only:
        write_json(catalog_dir / f"genre=All&skip={final_skip}.json", empty)
        write_json(catalog_dir / f"skip={final_skip}&genre=All.json", empty)
