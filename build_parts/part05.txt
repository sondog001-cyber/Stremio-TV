            skip += page_size

        # EPG pagination follows the same exact-count rule as normal catalogs:
        # after a short final page Stremio requests skip=<total channels>.
        final_skip = len(detailed)
        write_json(catalog_dir / f"date={day}&skip={final_skip}.json", cached_resource({"metasDetailed": []}))
        write_json(catalog_dir / f"skip={final_skip}&date={day}.json", cached_resource({"metasDetailed": []}))


def make_build_info(
    config: dict[str, Any],
    now: dt.datetime | None = None,
) -> dict[str, Any]:
    """Create one immutable identity shared by every artifact in this build."""
    built_at = iso_z(now or dt.datetime.now(UTC))
    version = str((config.get("addon") or {}).get("version") or "0.0.0")
    revision = os.getenv("GITHUB_SHA", "").strip()
    compact_stamp = built_at.replace("-", "").replace(":", "")
    build_id = f"v{version}-{compact_stamp}"
    if revision:
        build_id += f"-{revision[:12]}"

    info: dict[str, Any] = {
        "build_id": build_id,
        "version": version,
        "built_at": built_at,
        "builder": "stremio-tv-static",
    }
    optional_env = {
        "source_revision": "GITHUB_SHA",
        "repository": "GITHUB_REPOSITORY",
        "workflow": "GITHUB_WORKFLOW",
        "workflow_run_id": "GITHUB_RUN_ID",
        "workflow_run_number": "GITHUB_RUN_NUMBER",
        "workflow_run_attempt": "GITHUB_RUN_ATTEMPT",
    }
    for key, env_name in optional_env.items():
        value = os.getenv(env_name, "").strip()
        if value:
            info[key] = value
    return info

def derive_public_base() -> str:
    explicit = os.getenv("PUBLIC_BASE_URL", "").strip().rstrip("/")
    if explicit:
        return explicit
    repo = os.getenv("GITHUB_REPOSITORY", "").strip()
    if "/" in repo:
        owner, name = repo.split("/", 1)
        if name.lower() == f"{owner.lower()}.github.io":
            return f"https://{owner}.github.io"
        return f"https://{owner}.github.io/{name}"
    return ""


def _canonical_logo_url(url: str) -> str:
    """Turn common GitHub blob image links into directly downloadable assets."""
    value = str(url or "").strip()
    match = re.match(
        r"^https://github\.com/([^/]+)/([^/]+)/blob/([^/]+)/(.*?)(?:\?.*)?$",
        value,
        re.I,
    )
    if match:
        owner, repository, revision, path = match.groups()
        return f"https://raw.githubusercontent.com/{owner}/{repository}/{revision}/{path}"
    return value


def _fallback_logo_label(name: str) -> str:
    tokens = re.findall(r"[A-Za-z0-9]+", str(name or ""))
    if len(tokens) >= 2:
        return "".join(token[0] for token in tokens[:4]).upper()
    if tokens:
        return tokens[0][:4].upper()
    return "TV"


def normalize_channel_logos(
    site: Path,
    channels: list[dict[str, Any]],
    public_base: str,
    config: dict[str, Any],
    *,
    enabled: bool = True,
) -> dict[str, Any]:
    """Cache every channel logo as a consistently padded local square PNG."""
    settings = config.get("logo_normalization") or {}
    enabled = enabled and bool(settings.get("enabled", True))
    canvas_size = max(128, int(settings.get("canvas_size", 512)))
    max_logo_size = max(64, min(canvas_size, int(settings.get("max_logo_size", 384))))
    timeout = max(1, int(settings.get("timeout_seconds", 8)))
    byte_limit = max(64_000, int(settings.get("byte_limit", 5_000_000)))
    rows: list[dict[str, Any]] = []

    if not enabled:
        return {
            "version": 1,
            "enabled": False,
            "summary": {
                "channels": len(channels),
                "normalized": 0,
                "fallback": 0,
                "failed": 0,
            },
            "channels": [],
        }

    try:
        from PIL import Image, ImageDraw, ImageFont
    except Exception as exc:
        log(f"WARNING: Pillow unavailable; channel-logo normalization skipped: {exc}")
        return {
            "version": 1,
            "enabled": False,
            "error": f"Pillow unavailable: {exc}",
            "summary": {
                "channels": len(channels),
                "normalized": 0,
                "fallback": 0,
                "failed": len(channels),
            },
            "channels": [],
        }

    asset_dir = site / "assets" / "channel-logos"
    asset_dir.mkdir(parents=True, exist_ok=True)

    def make_fallback(name: str):
        canvas = Image.new("RGBA", (canvas_size, canvas_size), (0, 0, 0, 0))
        draw = ImageDraw.Draw(canvas)
        pad = max(20, canvas_size // 8)
        draw.rounded_rectangle(
            (pad, pad, canvas_size - pad, canvas_size - pad),
            radius=max(18, canvas_size // 12),
            fill=(28, 31, 43, 255),
            outline=(255, 255, 255, 72),
            width=max(2, canvas_size // 128),
        )
        label = _fallback_logo_label(name)
        font_size = max(28, int(canvas_size * (0.22 if len(label) <= 3 else 0.16)))
        try:
            font = ImageFont.truetype("DejaVuSans-Bold.ttf", font_size)
        except Exception:
            font = ImageFont.load_default()
        box = draw.textbbox((0, 0), label, font=font)
        width = box[2] - box[0]
        height = box[3] - box[1]
        draw.text(
            ((canvas_size - width) / 2, (canvas_size - height) / 2 - box[1]),
            label,
            font=font,
            fill=(245, 247, 250, 255),
        )
        return canvas

    for channel in channels:
        source_url = _canonical_logo_url(str(channel.get("logo") or ""))
        destination = asset_dir / f"{channel['id']}.png"
        status = "fallback"
        error = None
        original_size = None
        try:
            if source_url and re.match(r"^https?://", source_url, re.I):
                req = urllib.request.Request(
                    source_url,
                    headers={
                        "User-Agent": USER_AGENT,
                        "Accept": "image/avif,image/webp,image/apng,image/svg+xml,image/*,*/*;q=0.8",
                        "Accept-Encoding": "identity",
                    },
                )
                with urllib.request.urlopen(req, timeout=timeout) as resp:
                    data = resp.read(byte_limit + 1)
                if len(data) > byte_limit:
                    raise ValueError(f"logo exceeded {byte_limit} byte limit")
                with Image.open(io.BytesIO(data)) as image:
                    image.load()
                    original_size = list(image.size)
                    logo = image.convert("RGBA")
                    alpha_box = logo.getchannel("A").getbbox()
                    if alpha_box:
                        logo = logo.crop(alpha_box)
                    logo.thumbnail((max_logo_size, max_logo_size), Image.Resampling.LANCZOS)
                    canvas = Image.new("RGBA", (canvas_size, canvas_size), (0, 0, 0, 0))
                    x = (canvas_size - logo.width) // 2
                    y = (canvas_size - logo.height) // 2
                    canvas.alpha_composite(logo, (x, y))
                    canvas.save(destination, format="PNG", optimize=True)
                    status = "normalized"
            else:
                make_fallback(channel.get("name", "TV")).save(destination, format="PNG", optimize=True)
        except Exception as exc:
            error = str(exc)[:240]
            make_fallback(channel.get("name", "TV")).save(destination, format="PNG", optimize=True)
            status = "fallback"

        local_path = f"assets/channel-logos/{channel['id']}.png"
        local_url = f"{public_base}/{local_path}" if public_base else local_path
        channel["logo_source"] = source_url
        if public_base:
            channel["logo"] = local_url
        rows.append({
            "id": channel.get("id"),
            "tvg_id": channel.get("tvg_id"),
            "name": channel.get("name"),
            "source_url": source_url or None,
            "local_url": local_url,
            "status": status,
            "original_size": original_size,
            "canvas_size": [canvas_size, canvas_size],
            "max_logo_size": max_logo_size,
            "error": error,
        })

    summary = {
        "channels": len(rows),
        "normalized": sum(1 for row in rows if row["status"] == "normalized"),
        "fallback": sum(1 for row in rows if row["status"] == "fallback"),
        "failed": sum(1 for row in rows if not (asset_dir / f"{row['id']}.png").exists()),
    }
    log(
        "Channel logos: "
        f"normalized={summary['normalized']:,}, "
        f"fallback={summary['fallback']:,}, "
        f"failed={summary['failed']:,}"
    )
    return {
        "version": 1,
        "enabled": True,
        "summary": summary,
        "channels": rows,
    }

def load_manual_streams(config: dict[str, Any], config_path: Path) -> list[dict[str, Any]]:
    manual_name = config.get("manual_sources_file")
    if not manual_name:
        return []
    path = config_path.parent / manual_name
    if not path.exists():
        return []
    data = load_json(path)
    rows = data.get("streams", []) if isinstance(data, dict) else []
    result: list[dict[str, Any]] = []
    for row in rows:
        if not isinstance(row, dict) or not row.get("url"):
            continue
        item = dict(row)
        item.setdefault("source", "User")
        item.setdefault("family", "manual")
        item.setdefault("priority", 1)

        item.setdefault("headers", {})

        item.setdefault("language", "")
        item.setdefault("country", "US")
        item.setdefault("tvg_id", "")
        item.setdefault("name", item.get("tvg_id") or "Manual Channel")
        item.setdefault("name_raw", item["name"])
        if not item.get("quality"):
            item["quality"] = extract_quality(item.get("name_raw", ""), item.get("tvg_id", ""))
        result.append(item)
    return result


def load_private_connector_streams(config: dict[str, Any], config_path: Path) -> list[dict[str, Any]]:
    """Load a generated private overlay only when explicitly enabled locally."""
    configured_path = os.getenv("STREMIO_TV_PRIVATE_SOURCES_FILE", "").strip()
    if not configured_path:
        return []
    if os.getenv("GITHUB_ACTIONS", "").strip().lower() == "true":
        raise RuntimeError("Private connector overlays are blocked in GitHub Actions public builds")
    path = Path(configured_path).expanduser()
    if not path.is_absolute():
        path = config_path.parent / path
    if not path.is_file():
        raise FileNotFoundError("Configured private connector overlay does not exist")

    data = load_json(path)
    rows = data.get("streams", []) if isinstance(data, dict) else []
    allowed_ids = {canonical_tvg_id(item) for item in _exact_allowlist_ids(config)}
    result: list[dict[str, Any]] = []
    for row in rows:
        if not isinstance(row, dict) or not row.get("url") or not row.get("tvg_id"):
            continue
        if canonical_tvg_id(str(row["tvg_id"])) not in allowed_ids:
            continue
        item = dict(row)
        item.setdefault("source", "Private connector")
        item.setdefault("family", "private-connector")
        item.setdefault("priority", -30)
        item.setdefault("philly", False)
        item.setdefault("headers", {})
        item.setdefault("group", "")
        item.setdefault("logo", "")
        item.setdefault("language", "English")
        item.setdefault("country", "US")
        item.setdefault("name", item["tvg_id"])
        item.setdefault("name_raw", item["name"])
        item.setdefault("source_type", "private-authorized-connector")
        item["private_connector"] = True
        if not item.get("quality"):
            item["quality"] = extract_quality(item.get("name_raw", ""), item.get("tvg_id", ""))
        result.append(item)
    return result


def build(config_path: Path, aliases_path: Path, output: Path, fixtures: Path | None = None) -> dict[str, Any]:
    _PUBLIC_TEXT_CACHE.clear()
    config = load_json(config_path)
    build_info = make_build_info(config)
    public_base = derive_public_base()
    if public_base:
        build_info["public_base_url"] = public_base
    aliases = load_json(aliases_path) if aliases_path.exists() else {"by_tvg_id": {}, "by_name": {}}
    page_size = int(config.get("page_size", 100))
    max_streams = int(config.get("max_streams_per_channel", 3))

    if output.exists():
        shutil.rmtree(output)
    output.mkdir(parents=True, exist_ok=True)

    temp_dir = Path(tempfile.mkdtemp(prefix="stremio-tv-"))
    try:
        entries: list[dict[str, Any]] = []
        source_results: list[dict[str, Any]] = []
        for idx, src in enumerate(config["sources"]["playlists"]):
            required = bool(src.get("required", True))
            result = {
                "name": src.get("name", f"Source {idx + 1}"),
                "family": src.get("family", ""),
                "url": src.get("url", ""),
                "required": required,
                "status": "pending",
                "streams": 0,
            }
            try:
                if fixtures:
                    fixture_name = src.get("fixture")
                    if not fixture_name:
                        if required:
                            raise RuntimeError(f"No fixture declared for required source {src['name']}")
                        result["status"] = "skipped-no-fixture"
                        source_results.append(result)
                        continue
                    playlist_path = fixtures / fixture_name
                    if not playlist_path.exists():
                        if required:
                            raise RuntimeError(f"Missing required fixture: {playlist_path}")
                        result["status"] = "skipped-missing-fixture"
